//===== TERMUXFILE: Testmate/app/api/tests/targeted/route.ts =====
import { NextRequest, NextResponse } from 'next/server';
import { getServiceClient } from '@/lib/supabase';
import { getCurrentUserFromRequest } from '@/lib/auth';
import { TARGETED_REPAIR_KIND } from '@/lib/test-origin';

// POST /api/tests/targeted
// Body: {
//   intervention_id: string,       // Checkmate's GapTaskLedger concept id
//   chapter: string,
//   topic?: string,
//   question_count?: number,       // default 15 for pool 'NEW'; omitted/0/negative means
//                                   // "no cap" for WRONG / SKIPPED / WRONG_SKIPPED — returns
//                                   // every matching question instead of truncating at 15
//   pool?: 'WRONG' | 'SKIPPED' | 'WRONG_SKIPPED' | 'NEW',  // default 'WRONG_SKIPPED'
//   duration_seconds?: number      // default question_count * 90
//   external_questions?: Array<{   // NEW — external-report pathway (see doc below)
//     question_text: string,
//     options?: Record<string, string>,
//     correct_answer?: string,
//     explanation?: string
//   }>
// }
//
// external_questions (NEW): for a report Checkmate parsed locally but that was never
// actually taken on Testmate — an AI-recreated or hand-written markdown import has no
// `questions`/`responses` rows here to look up by chapter. When this array is non-empty,
// it REPLACES the entire "student's own Testmate history" lookup below: Checkmate has
// already done its own wrong/skipped filtering on data this endpoint has no other way to
// see, so these questions are used as-is — no backfill from the chapter's Testmate
// question bank, no by-chapter staleness comparison (meaningless here; see that check's
// own note). Idempotency-by-intervention_id still applies the same as the native path.
//
// P0b — the "Testmate owns question selection, Checkmate owns why" endpoint. Builds a
// student-specific test out of THIS student's own wrong/skipped history for one
// chapter/topic (falling back to never-attempted questions from the same chapter/topic
// if there aren't enough), then immediately creates and joins a solo session against it —
// so Checkmate never has to also call POST /api/sessions or implement question selection
// itself. Auth via getCurrentUserFromRequest (device-token compatible), unlike
// POST /api/sessions, which currently only accepts the cookie session — Checkmate has no
// browser session, so that route was never callable from the app for this.
//
// Idempotent on intervention_id (see db/migrations/009_targeted_tests.sql, refined by
// 010_targeted_tests_reusable_intervention.sql): calling this again with the same
// intervention_id for the same user, while a LIVE session for it still exists, returns
// that SAME test/session rather than generating a new test every retry — UNLESS the
// chapter/topic has been re-imported since that session's test was built, in which case
// the old session is superseded (status -> 'ended') and a fresh one is generated from the
// current question bank (see the staleness check below). A targeted test's question set is
// frozen at creation time, so without this an unfinished repair test could never pick up
// newly imported questions for its own chapter.
//
// KNOWN LIMITATION: "SKIPPED" only catches a question the student visited and left
// unanswered (a `responses` row with a null selected_answer) — a question in a past test
// the student never even navigated to (no `responses` row at all) can't be detected this
// way without diffing every past test's full question list against responses, which this
// endpoint doesn't do. Good enough for v1; revisit if that gap turns out to matter.
//
// BUGFIX (Aug 2026): the WRONG/SKIPPED lookup used to pull the student's ENTIRE response
// history (every chapter, every subject, since day one) via `.eq('user_id', ...)`, then
// filter it down to this chapter in memory. supabase-js sends `.in()` filters as URL query
// params, and that unbounded history was stuffed into a single GET request's `.in('id', ...)`
// filter — fine while history was small, but it grows without bound as the student keeps
// using the app, and eventually exceeds the HTTP gateway's URL-length limit. That produces
// a raw, non-JSON 400 from the gateway itself (not a normal PostgREST error), which
// supabase-js surfaces as a generic `"Bad Request"` message — indistinguishable from a real
// query error, and unrelated to which chapter/concept was being requested (hence "fails on
// every chapter, even a brand new one" once history crossed the threshold).
//
// Fixed by flipping the query order: first fetch THIS CHAPTER's (optionally topic's)
// question-id pool from `questions` — bounded by how many questions exist for one
// chapter, not by the student's lifetime history — then filter `responses` down to only
// those ids. That pool is batched defensively (CHUNK_SIZE) in case a single chapter's
// question bank is ever large enough to risk the same URL-length problem on its own.
//
// TAG_SOURCE ISOLATION: coaching_module (canonical Q-bank supply, see
// app/api/qbank/practice/route.ts's POOL SOURCE note) and both copy tag_sources
// (targeted_copy, qbank_copy) are excluded from every candidate query this route runs
// (WRONG/SKIPPED chapter-pool lookup and the NEW/backfill query) — see each query's own
// comment. Invariant: targeted-repair never selects a Q-bank-owned row or a generated
// copy, full stop, no per-pool exception. This does NOT stop a coaching_module attempt
// from counting toward mastery/retention (lib/daily-target-engine.ts's canonical total
// already includes it) — it only keeps that supply from being copied into a repair test
// and, as a side effect, keeps a freshly-generated copy (which always looks "never
// attempted" until it's actually answered) from resurfacing as its own backfill
// candidate.
export async function POST(req: NextRequest) {
  const user = await getCurrentUserFromRequest(req);
  if (!user) {
    return NextResponse.json({ error: 'Not authenticated' }, { status: 401 });
  }

  let body: Record<string, unknown>;
  try {
    body = await req.json();
  } catch {
    return NextResponse.json({ error: 'Invalid JSON body' }, { status: 400 });
  }

  const interventionId = body.intervention_id;
  const chapter = body.chapter;
  const topic = typeof body.topic === 'string' && body.topic.trim() ? body.topic.trim() : null;
  const pool = (typeof body.pool === 'string' ? body.pool.toUpperCase() : 'WRONG_SKIPPED') as
    | 'WRONG'
    | 'SKIPPED'
    | 'WRONG_SKIPPED'
    | 'NEW';

  // NEW (external-report pathway): Checkmate's own local wrong/skipped set for a
  // report it parsed but that was never actually taken on Testmate (so there is no
  // `responses` history here to look up by chapter). When present and non-empty,
  // this REPLACES the entire "build candidate pool from this student's own
  // Testmate history" section below — Checkmate is the source of truth for which
  // questions to serve, not a `.eq('chapter', ...)` lookup against data that was
  // never actually written to `questions`/`responses` for this student. `pool` is
  // still accepted/validated for logging consistency but has no effect on
  // selection in this path — Checkmate already filtered to wrong/skipped on its
  // side before sending these.
  const rawExternalQuestions = Array.isArray(body.external_questions) ? body.external_questions : null;
  const usingExternalQuestions = !!rawExternalQuestions && rawExternalQuestions.length > 0;

  type ExternalQuestion = {
    question_text: string;
    options: Record<string, string> | null;
    correct_answer: string | null;
    explanation: string | null;
  };
  let externalQuestions: ExternalQuestion[] = [];
  if (usingExternalQuestions) {
    for (const [i, raw] of rawExternalQuestions!.entries()) {
      if (typeof raw !== 'object' || raw === null) {
        return NextResponse.json({ error: `external_questions[${i}] is not an object` }, { status: 400 });
      }
      const q = raw as Record<string, unknown>;
      if (typeof q.question_text !== 'string' || !q.question_text.trim()) {
        return NextResponse.json({ error: `external_questions[${i}].question_text is required` }, { status: 400 });
      }
      externalQuestions.push({
        question_text: q.question_text,
        options: typeof q.options === 'object' && q.options !== null ? (q.options as Record<string, string>) : null,
        correct_answer: typeof q.correct_answer === 'string' ? q.correct_answer : null,
        explanation: typeof q.explanation === 'string' ? q.explanation : null
      });
    }
  }

  // FIX (repair-test hard limit): question_count as a positive number is still a hard cap,
  // same as always — this is what a retention check (Checkmate's RetentionCheckManager)
  // keeps using to stay a short probe. But omitted/0/negative now means "no cap" for
  // WRONG / SKIPPED / WRONG_SKIPPED — a repair test should pull in EVERY wrong/skipped
  // question this student has for the chapter/topic, not just the first 15 the old default
  // silently truncated to. 'NEW' still needs a real ceiling here since its candidate pool is
  // the chapter's ENTIRE question bank, not bounded by the student's own history — an
  // uncapped request for 'NEW' falls back to the same default of 15 it always had.
  const rawQuestionCount = typeof body.question_count === 'number' ? Math.floor(body.question_count) : null;
  const uncapped = pool !== 'NEW' && (rawQuestionCount === null || rawQuestionCount <= 0);
  const questionCap = uncapped
    ? Number.POSITIVE_INFINITY
    : rawQuestionCount !== null && rawQuestionCount > 0
      ? rawQuestionCount
      : 15;

  if (typeof interventionId !== 'string' || !interventionId.trim()) {
    return NextResponse.json({ error: 'intervention_id is required' }, { status: 400 });
  }
  if (typeof chapter !== 'string' || !chapter.trim()) {
    return NextResponse.json({ error: 'chapter is required' }, { status: 400 });
  }
  if (!['WRONG', 'SKIPPED', 'WRONG_SKIPPED', 'NEW'].includes(pool)) {
    return NextResponse.json(
      { error: "pool must be one of 'WRONG', 'SKIPPED', 'WRONG_SKIPPED', 'NEW'" },
      { status: 400 }
    );
  }

  const supabase = getServiceClient();

  // Batches any `.in('id', [...])` filter so a single request's URL can't grow unbounded —
  // see the BUGFIX note above. 150 ids/request comfortably stays under gateway URL limits
  // even with 36-char UUIDs.
  const CHUNK_SIZE = 150;
  function chunk<T>(arr: T[], size: number): T[][] {
    const out: T[][] = [];
    for (let i = 0; i < arr.length; i += size) out.push(arr.slice(i, i + size));
    return out;
  }

  // ── Idempotency: same student + same intervention_id => same LIVE test/session ──────
  // BUGFIX (Aug 2026, reintroduced by 010's re-migration): this query used to have no
  // status filter, which was only safe under 009's original unique index (one row per
  // intervention_id, ever). 010_targeted_tests_reusable_intervention.sql rescoped that
  // index to `status = 'live'` so a later, genuinely new occurrence of the same concept
  // could get a fresh session once the earlier one ended — but this route wasn't updated
  // to match, so once an old (ended) session and a new (live) session both existed for the
  // same intervention_id, `.maybeSingle()` found more than one row and threw "JSON object
  // requested, multiple (or no) rows returned" (HTTP 500). When only the old ended session
  // existed, this also silently handed it back as "existing" instead of creating a new
  // one — the cause of a fresh round's "Take Test" landing on a previous round's result
  // page. Filtering to status = 'live' here matches the DB index and fixes both.
  const { data: existingSession, error: idempotencyError } = await supabase
    .from('test_sessions')
    .select('id, test_id, intervention_id')
    .eq('intervention_id', interventionId)
    .eq('created_by', user.id)
    .eq('status', 'live')
    .maybeSingle();

  if (idempotencyError) {
    console.error('[targeted] idempotency check failed:', idempotencyError.message);
    return NextResponse.json({ error: idempotencyError.message }, { status: 500 });
  }

  if (existingSession) {
    const { data: existingTest, error: existingTestError } = await supabase
      .from('tests')
      .select('id, title, status, duration_seconds, created_at')
      .eq('id', existingSession.test_id)
      .single();
    if (existingTestError) {
      console.error('[targeted] existing-test lookup failed:', existingTestError.message);
      return NextResponse.json({ error: existingTestError.message }, { status: 500 });
    }

    // BUGFIX (stale repair test never refreshed): a targeted test freezes its question set
    // at creation time (see the copy-into-`questions` step below), so blindly reusing a
    // still-`live` session forever meant a student stuck on an unfinished repair test never
    // saw questions from a LATER re-import of the same chapter — reproduced live on
    // Biomolecules, stuck on a 282-question set built well before several subsequent
    // imports. Compare this test's created_at against the newest question row for this
    // chapter/topic; if the chapter was (re-)imported after this test was built, the old
    // session is stale and gets superseded instead of handed back.
    // NEW (external-report pathway): the date-based check below compares against
    // `questions.created_at` for this chapter — meaningless here, since external
    // questions aren't looked up from that table at all. Checkmate only resends
    // external_questions when it actually has something to serve (a fresh
    // import, or the once-per-day re-verify with unchanged content), so treat
    // its presence itself as the freshness signal and always rebuild — capped
    // to once/day by the client's own `activeTestmateSessionCheckedDay` guard,
    // same as the native path's effective cadence.
    let isStale = usingExternalQuestions;
    if (!usingExternalQuestions) {
      let newestQuestionQuery = supabase
        .from('questions')
        .select('created_at')
        .eq('chapter', chapter)
        .order('created_at', { ascending: false })
        .limit(1);
      if (topic) newestQuestionQuery = newestQuestionQuery.eq('topic', topic);
      const { data: newestQuestionRows, error: newestQError } = await newestQuestionQuery;
      if (newestQError) {
        console.error('[targeted] staleness check failed, reusing existing session as-is:', newestQError.message);
      }

      isStale =
        !newestQError &&
        newestQuestionRows?.[0]?.created_at != null &&
        new Date(newestQuestionRows[0].created_at).getTime() > new Date(existingTest.created_at).getTime();
    }

    if (!isStale) {
      return NextResponse.json({
        test: existingTest,
        session: existingSession,
        question_count: uncapped ? null : questionCap,
        reused: true
      });
    }

    console.log(
      `[targeted] session ${existingSession.id} (intervention_id=${interventionId}) is stale — ` +
      `chapter "${chapter}" was re-imported after this test was built. Superseding and rebuilding.`
    );
    const { error: supersedeError } = await supabase
      .from('test_sessions')
      .update({ status: 'ended' }) // reuses the existing enum value — already outside the `status = 'live'` unique index, no migration needed
      .eq('id', existingSession.id);
    if (supersedeError) {
      // Couldn't mark it superseded — reusing the (still technically live, if stale) session
      // beats erroring the whole request or, worse, ending up with two live sessions for the
      // same intervention_id once a new one is inserted below.
      console.error('[targeted] failed to supersede stale session, reusing it instead:', supersedeError.message);
      return NextResponse.json({
        test: existingTest,
        session: existingSession,
        question_count: uncapped ? null : questionCap,
        reused: true
      });
    }
    // fall through — builds a fresh test/session below, same path as "no existing session"
  }

  // ── Build the candidate question pool from this student's own history ──────────────
  type CandidateRow = {
    id: string;
    question_text: string;
    question_image_url: string | null;
    question_data: unknown;
    options: unknown;
    option_images: unknown;
    correct_answer: string | null;
    marks: number | null;
    negative_marks: number | null;
    chapter: string | null;
    topic: string | null;
    difficulty: string | null;
    question_type: string | null;
    explanation: string | null;
    answered_at?: string | null;
  };

  const QUESTION_SELECT =
    'id, question_text, question_image_url, question_data, options, option_images, correct_answer, marks, negative_marks, chapter, topic, difficulty, question_type, explanation';

  const chosen: CandidateRow[] = [];
  const chosenIds = new Set<string>();

  // NEW (external-report pathway): Checkmate already did its own wrong/skipped
  // filtering locally (on data this endpoint has no other way to see — these
  // questions were never taken on Testmate, so there's no `questions`/
  // `responses` row to look up). Map the payload straight into `chosen` and skip
  // the whole history-lookup + backfill section below entirely — mixing in
  // never-attempted Testmate-bank questions for a chapter Checkmate is reporting
  // on independently would defeat the point of sending explicit content.
  if (usingExternalQuestions) {
    for (const [i, eq] of externalQuestions.entries()) {
      chosen.push({
        id: `external-${interventionId}-${i}`,
        question_text: eq.question_text,
        question_image_url: null,
        question_data: null,
        options: eq.options,
        option_images: null,
        correct_answer: eq.correct_answer,
        marks: 4,
        negative_marks: 1,
        chapter: typeof chapter === 'string' ? chapter : null,
        topic,
        difficulty: null,
        question_type: null,
        explanation: eq.explanation
      });
      chosenIds.add(chosen[chosen.length - 1].id);
    }
  } else if (pool === 'WRONG' || pool === 'SKIPPED' || pool === 'WRONG_SKIPPED') {
    // Scope by CHAPTER first — bounded by this chapter's question bank, not by the
    // student's entire lifetime response history (see BUGFIX note above).
    //
    // TAG_SOURCE EXCLUSION (repair-test/Q-bank isolation): canonical Q-bank supply
    // (tag_source = 'coaching_module', per app/api/qbank/practice/route.ts's POOL
    // SOURCE note) plus both copy tag_sources ('targeted_copy', 'qbank_copy') are
    // excluded here so targeted-repair can never select a Q-bank-owned row or a
    // generated copy — coaching_module attempts still count toward mastery/retention
    // (lib/daily-target-engine.ts counts them in its canonical total), they're just
    // never eligible to be copied INTO a repair test. Since chapterQuestionIds below
    // is the only source `responses` gets matched against, this one filter also keeps
    // a coaching_module/copy row from ever reaching the candidate-question lookup
    // further down — no second filter needed there.
    let chapterQuestionsQuery = supabase
      .from('questions')
      .select('id')
      .eq('chapter', chapter)
      .not('tag_source', 'in', '(coaching_module,targeted_copy,qbank_copy)');
    if (topic) chapterQuestionsQuery = chapterQuestionsQuery.eq('topic', topic);
    const { data: chapterQuestions, error: chapterQError } = await chapterQuestionsQuery;
    if (chapterQError) {
      console.error('[targeted] chapter question-pool lookup failed:', chapterQError.message);
      return NextResponse.json({ error: chapterQError.message }, { status: 500 });
    }
    const chapterQuestionIds = (chapterQuestions ?? []).map((q) => q.id);

    if (chapterQuestionIds.length > 0) {
      // Fetch this student's responses restricted to THIS chapter's question ids only,
      // batched defensively in case one chapter's bank is itself very large.
      const responseRows: { question_id: string; selected_answer: string | null; is_correct: boolean | null; answered_at: string | null }[] = [];
      for (const idsBatch of chunk(chapterQuestionIds, CHUNK_SIZE)) {
        const { data: batch, error: rError } = await supabase
          .from('responses')
          .select('question_id, selected_answer, is_correct, answered_at')
          .eq('user_id', user.id)
          .in('question_id', idsBatch);
        if (rError) {
          console.error('[targeted] responses lookup failed:', rError.message);
          return NextResponse.json({ error: rError.message }, { status: 500 });
        }
        if (batch) responseRows.push(...batch);
      }

      if (responseRows.length > 0) {
        const uniqueAnsweredIds = Array.from(new Set(responseRows.map((r) => r.question_id)));
        const questionById = new Map<string, CandidateRow>();
        for (const idsBatch of chunk(uniqueAnsweredIds, CHUNK_SIZE)) {
          const { data: qBatch, error: qError } = await supabase
            .from('questions')
            .select(QUESTION_SELECT)
            .in('id', idsBatch);
          if (qError) {
            console.error('[targeted] candidate-question lookup failed:', qError.message);
            return NextResponse.json({ error: qError.message }, { status: 500 });
          }
          for (const q of qBatch ?? []) questionById.set(q.id, q as CandidateRow);
        }

        const wantWrong = pool === 'WRONG' || pool === 'WRONG_SKIPPED';
        const wantSkipped = pool === 'SKIPPED' || pool === 'WRONG_SKIPPED';

        // Most-recent-first, and de-duped by question_id (a question can have been
        // attempted in more than one past test session) — keep only the latest attempt
        // at each one.
        const sorted = [...responseRows].sort(
          (a, b) => new Date(b.answered_at ?? 0).getTime() - new Date(a.answered_at ?? 0).getTime()
        );

        for (const r of sorted) {
          if (chosen.length >= questionCap) break;
          if (chosenIds.has(r.question_id)) continue;
          const q = questionById.get(r.question_id);
          if (!q) continue;

          const isWrong = r.is_correct === false;
          const isSkipped = r.selected_answer === null || r.selected_answer === undefined;
          if ((wantWrong && isWrong) || (wantSkipped && isSkipped)) {
            chosen.push(q);
            chosenIds.add(r.question_id);
          }
        }
      }
    }
  }

  // ── Backfill (or serve entirely, for pool === 'NEW') from never-attempted questions ──
  // Skipped entirely when `uncapped`: an uncapped WRONG/SKIPPED/WRONG_SKIPPED request wants
  // exactly this student's actual wrong/skipped set, not padded up to some target size with
  // questions they've never even seen.
  if (!uncapped && !usingExternalQuestions && chosen.length < questionCap) {
    // Only needs to know WHICH of this chapter's questions the student has already
    // attempted, so scope this the same bounded way — not the student's full history.
    // Same TAG_SOURCE EXCLUSION as the WRONG/SKIPPED pool lookup above (see that
    // query's own doc) — this is the pool === 'NEW' backfill's own candidate query,
    // so without this exclusion a never-attempted coaching_module row (real Q-bank
    // supply) or a freshly-generated copy (which always looks "unattempted" — it has
    // no mastery history of its own yet) could get pulled straight into a repair test.
    let backfillQuery = supabase
      .from('questions')
      .select(QUESTION_SELECT)
      .eq('chapter', chapter)
      .not('tag_source', 'in', '(coaching_module,targeted_copy,qbank_copy)')
      .limit(questionCap * 3); // overfetch — some will be filtered out below
    if (topic) backfillQuery = backfillQuery.eq('topic', topic);

    const { data: backfillCandidates, error: bError } = await backfillQuery;
    if (bError) {
      console.error('[targeted] backfill lookup failed:', bError.message);
      return NextResponse.json({ error: bError.message }, { status: 500 });
    }

    const backfillIds = (backfillCandidates ?? []).map((q) => q.id);
    const attemptedIds = new Set<string>();
    for (const idsBatch of chunk(backfillIds, CHUNK_SIZE)) {
      const { data: attempted, error: aError } = await supabase
        .from('responses')
        .select('question_id')
        .eq('user_id', user.id)
        .in('question_id', idsBatch);
      if (aError) {
        console.error('[targeted] attempted-ids lookup failed:', aError.message);
        return NextResponse.json({ error: aError.message }, { status: 500 });
      }
      for (const r of attempted ?? []) attemptedIds.add(r.question_id);
    }

    for (const q of backfillCandidates ?? []) {
      if (chosen.length >= questionCap) break;
      if (chosenIds.has(q.id)) continue;
      if (attemptedIds.has(q.id)) continue; // "NEW" means genuinely never attempted
      chosen.push(q as CandidateRow);
      chosenIds.add(q.id);
    }
  }

  if (chosen.length === 0) {
    return NextResponse.json(
      { error: `No questions available for chapter "${chapter}"${topic ? ` / topic "${topic}"` : ''} yet.` },
      { status: 422 }
    );
  }

  // Computed from the actual selected set (chosen.length), not the nominal cap — matters
  // once `pool` is uncapped, since there is no fixed target question_count left to multiply
  // by 90 in that case.
  const durationSeconds =
    typeof body.duration_seconds === 'number' && body.duration_seconds > 0
      ? body.duration_seconds
      : chosen.length * 90;

  // ── Create the test, copy the chosen questions into it, start a solo session ────────
  const { data: newTest, error: testError } = await supabase
    .from('tests')
    .insert({
      title: `Repair: ${chapter}${topic ? ` — ${topic}` : ''}`,
      status: 'published', // system-generated from already-tagged questions — skip draft/review
      owner_user_id: user.id,
      generated_from: {
        // NEW (Q-bank/targeted-repair origin split — see lib/test-origin.ts):
        // explicit going forward. Rows created before this change have no
        // `kind` at all; isTargetedRepair() only matches the explicit tag,
        // so old rows are simply invisible to the new origin-scoped queries
        // rather than misclassified — see that file's own note on why no
        // backfill migration was done for this.
        kind: TARGETED_REPAIR_KIND,
        intervention_id: interventionId,
        chapter,
        topic,
        pool,
        question_count: chosen.length,
        external: usingExternalQuestions
      },
      duration_seconds: durationSeconds
    })
    .select()
    .single();
  if (testError) {
    console.error('[targeted] test insert failed:', testError.message);
    return NextResponse.json({ error: testError.message }, { status: 500 });
  }

  const questionRows = chosen.map((q, i) => ({
    test_id: newTest.id,
    question_number: i + 1,
    question_text: q.question_text,
    question_image_url: q.question_image_url,
    question_data: q.question_data,
    options: q.options,
    option_images: q.option_images,
    correct_answer: q.correct_answer,
    marks: q.marks ?? 4,
    negative_marks: q.negative_marks ?? 1,
    chapter: q.chapter,
    topic: q.topic,
    difficulty: q.difficulty,
    question_type: q.question_type,
    explanation: q.explanation,
    // NEW (external-report pathway): `q.id` for an external question is a
    // synthetic id fabricated client-side (see the usingExternalQuestions branch
    // above) — it doesn't reference any real row in `questions`. Sending it here
    // would either silently mislabel provenance or violate a FK on this column if
    // one exists (migration not in scope here to confirm either way) — null is
    // the honest value: there IS no origin question for these.
    origin_question_id: usingExternalQuestions ? null : q.id,
    tag_source: 'targeted_copy'
  }));

  const { error: qInsertError } = await supabase.from('questions').insert(questionRows);
  if (qInsertError) {
    console.error('[targeted] question copy insert failed:', qInsertError.message);
    return NextResponse.json({ error: qInsertError.message }, { status: 500 });
  }

  const { data: session, error: sessionError } = await supabase
    .from('test_sessions')
    .insert({
      test_id: newTest.id,
      mode: 'solo',
      server_start_time: new Date().toISOString(),
      duration_seconds: durationSeconds,
      status: 'live',
      created_by: user.id,
      intervention_id: interventionId
    })
    .select()
    .single();
  if (sessionError) {
    console.error('[targeted] session insert failed:', sessionError.message);
    return NextResponse.json({ error: sessionError.message }, { status: 500 });
  }

  const { error: joinError } = await supabase
    .from('session_participants')
    .insert({ session_id: session.id, user_id: user.id });
  if (joinError) {
    console.error('[targeted] session join failed:', joinError.message);
    return NextResponse.json({ error: joinError.message }, { status: 500 });
  }

  return NextResponse.json(
    { test: newTest, session, question_count: questionRows.length, reused: false },
    { status: 201 }
  );
}

// ── Guardian lock gate for Work Mode settings ─────────────────────────────────
//
// Wraps any Work Mode-related settings (Blocked Apps, Blocked Websites, the
// Focus Cycle toggles) that must not be editable by the student while
// blocking is enforced — either because a focus session is running or
// because the hardcoded 19:00-02:00 window (WorkModeSchedule) is live.
// Shows a PIN-unlock prompt instead of [content] until a correct guardian
// PIN is entered, reusing UninstallGuard's existing PIN/hash/lockout flow
// so there's still exactly one PIN, and the student never sees it either
// way.
@Composable
fun WorkModeLockGate(content: @Composable () -> Unit) {
    val context = LocalContext.current

    // Poll every second - settingsLocked() depends on wall-clock time (the
    // schedule) and on UninstallGuard's unlock-window expiry, neither of
    // which trigger recomposition on their own.
    var tick by remember { mutableStateOf(0L) }
    LaunchedEffect(Unit) {
        while (true) {
            delay(1000)
            tick = System.currentTimeMillis()
        }
    }
    val locked = remember(tick) { WorkModeManager.settingsLocked() }

    // NOTE: previously this was `if (!locked) { content(); return }` — a bare early
    // `return` placed before any UI in a @Composable body. Google's own compose-runtime
    // tracker (b/203576696) flags exactly this shape — an early return that leaves one
    // branch of the function emitting nothing and the other branch emitting a full UI
    // subtree — as a trigger for composer group-stack corruption ("Index -1 out of
    // bounds" / IndexOutOfBoundsException in Stack.pop / ComposerImpl.exitGroup) once a
    // recomposition flips `locked` across that branch. Restructured as a plain if/else so
    // both branches are ordinary sibling groups instead of an early function exit.
    if (!locked) {
        content()
    } else {
        var pin by remember { mutableStateOf("") }
        var status by remember { mutableStateOf<String?>(null) }
        var statusColor by remember { mutableStateOf(AccentRed) }

        Column(modifier = Modifier.padding(horizontal = 14.dp, vertical = 12.dp)) {
            Row(verticalAlignment = Alignment.CenterVertically) {
                Icon(Icons.Default.Lock, null, tint = AccentAmber, modifier = Modifier.size(18.dp))
                Spacer(Modifier.width(8.dp))
                Text("Locked by guardian", fontSize = 13.sp, color = White90, fontWeight = FontWeight.Medium)
            }
            Text(
                "Blocked Apps, Blocked Websites, Focus Cycle, and the Guardian Telegram Chat ID " +
                    "are locked once a guardian PIN is set — not just during a focus session or the " +
                    "daily ${WorkModeSchedule.LABEL} window. Enter the PIN to make changes.",
                fontSize = 11.sp, color = White60,
                modifier = Modifier.padding(top = 4.dp, bottom = 10.dp)
            )
            OutlinedTextField(
                value         = pin,
                onValueChange = { if (it.length <= 6) { pin = it; status = null } },
                label         = { Text("Guardian PIN", color = White30) },
                singleLine    = true,
                modifier      = Modifier.fillMaxWidth(),
                visualTransformation = PasswordVisualTransformation(),
                leadingIcon   = { Icon(Icons.Default.LockOpen, null, tint = White60) },
                trailingIcon  = {
                    IconButton(onClick = {
                        when (val result = UninstallGuard.unlockWithPin(pin)) {
                            is UninstallGuard.UnlockResult.Success -> {
                                status = "Unlocked for 2 minutes"
                                statusColor = AccentGreen
                                pin = ""
                                tick = System.currentTimeMillis()
                            }
                            is UninstallGuard.UnlockResult.WrongPin -> {
                                status = "Incorrect PIN"
                                statusColor = AccentRed
                            }
                            is UninstallGuard.UnlockResult.NoPinConfigured -> {
                                status = "No PIN generated yet - see Security settings"
                                statusColor = AccentAmber
                            }
                            is UninstallGuard.UnlockResult.LockedOut -> {
                                status = "Too many wrong attempts - locked for ${result.secondsLeft / 60}m"
                                statusColor = AccentRed
                                if (result.justTriggered) GuardianNotifier.notifyPinBruteForce(context)
                            }
                        }
                    }) {
                        Icon(Icons.Default.LockOpen, null, tint = AccentGreen)
                    }
                },
                colors = OutlinedTextFieldDefaults.colors(
                    focusedBorderColor   = AccentGreen,
                    unfocusedBorderColor = White30,
                    cursorColor          = AccentGreen,
                    focusedTextColor     = White90,
                    unfocusedTextColor   = White90
                )
            )
            status?.let {
                Text(it, fontSize = 11.sp, color = statusColor, modifier = Modifier.padding(top = 6.dp))
            }
        }
    }
}

                IconButton(onClick = {
                    val trimmed = syncCode.trim()
                    if (trimmed.isBlank()) CheckmatePrefs.remove("sync_code")
                    else CheckmatePrefs.putString("sync_code", trimmed)
                    syncCode = trimmed

                    // immediately rather than waiting for the next Planner screen

                    }

                        if (saved) Icons.Default.Check else Icons.Default.Save,
                        null,
                        tint = if (saved) AccentGreen else White60
                    )
                }
            },

                focusedBorderColor   = AccentGreen,
                unfocusedBorderColor = White30,
                cursorColor          = AccentGreen,
                focusedTextColor     = White90,
                unfocusedTextColor   = White90

            Text(
                if (syncCode.isBlank()) "Sync turned off ✓" else "Saved ✓",

                modifier = Modifier.padding(top = 4.dp)
            )
        }
    }

/** "3h ago" / "2d ago" style label for [GapTaskManager.PREF_TESTMATE_LAST_ERROR_AT]. */
private fun relativeTimeAgo(epochMillis: Long): String {
    val deltaMin = (System.currentTimeMillis() - epochMillis) / 60_000L
    return when {
        deltaMin < 1L  -> "just now"
        deltaMin < 60L -> "${deltaMin}m ago"
        deltaMin < 1440L -> "${deltaMin / 60L}h ago"
        else -> "${deltaMin / 1440L}d ago"
    }
}

                            if (tokenSaved) Icons.Default.Check else Icons.Default.Save,
                            null,
                            tint = if (tokenSaved) AccentGreen else White60
                        )
                    }
                }
            },
            colors = OutlinedTextFieldDefaults.colors(
                focusedBorderColor   = AccentGreen,
                unfocusedBorderColor = White30,
                cursorColor          = AccentGreen,
                focusedTextColor     = White90,
                unfocusedTextColor   = White90
            )
        )
        if (tokenSaved) {
            Text("Saved ✓", fontSize = 11.sp, color = AccentGreen, modifier = Modifier.padding(top = 4.dp))
        }
    }
}

@Composable
private fun SettingSection(title: String, content: @Composable ColumnScope.() -> Unit) {
    Column(modifier = Modifier.padding(horizontal = 16.dp)) {
        Text(
            title,
            fontSize      = 11.sp,
            fontWeight    = FontWeight.Bold,
            letterSpacing = androidx.compose.ui.unit.TextUnit(2f, androidx.compose.ui.unit.TextUnitType.Sp),
            color         = White60,
            modifier      = Modifier.padding(top = 16.dp, bottom = 8.dp)
        )
        Surface(
            shape  = RoundedCornerShape(12.dp),
            color  = BgCard,
            border = BorderStroke(0.5.dp, White10)
        ) {
            Column(content = content)
        }
    }
}

@Composable
private fun SettingTile(title: String, subtitle: String, icon: ImageVector, onClick: () -> Unit) {
    Surface(onClick = onClick, color = androidx.compose.ui.graphics.Color.Transparent) {
        Row(
            modifier          = Modifier.fillMaxWidth().padding(14.dp),
            verticalAlignment = Alignment.CenterVertically
        ) {
            Icon(icon, null, tint = AccentGreen, modifier = Modifier.size(20.dp))

            Column(Modifier.weight(1f)) {
                Text(title, fontSize = 14.sp, color = White90)
                Text(subtitle, fontSize = 11.sp, color = White60)
            }
            Icon(Icons.Default.ChevronRight, null, tint = White30, modifier = Modifier.size(16.dp))
        }
    }
}
