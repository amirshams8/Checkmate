//===== TERMUXFILE: Testmate/lib/markdown-export.ts =====
export interface ReportQuestionRow {
  question_number: number;
  question_text: string;
  chapter: string | null;
  topic: string | null;
  correct_answer: string | null;
  selected_answer: string | null;
  is_correct: boolean | null;
  time_spent_seconds: number;
  // ADDED: Checkmate's ErrorEngine (learning/engine/ErrorEngine.kt) can only
  // classify a wrong answer as CARELESS/TIME_PRESSURE by timing today — it has
  // no evidence about *why* a specific wrong option was chosen, because this
  // report previously carried only the letter ("Your answer: c"), never what
  // option c actually said. Full option text is the missing evidence an LLM
  // classifier needs to distinguish e.g. a sign-convention mix-up from a
  // formula-selection error from a careless misread. Null for question_type
  // "numeric"/"subjective", which have no fixed option set — those still fall
  // back to the old letter-only rendering below.
  options: Record<string, string> | null;
}

export interface ReportWeakArea {
  accuracy_pct: number;
  attempted: number;
  chapter?: string;
  topic?: string;
}

export interface TestReportInput {
  testTitle: string;
  score: number;
  totalMarks: number;
  accuracyPct: number;
  attempted: number;
  correct: number;
  incorrect: number;
  skipped: number;
  weakChapters: ReportWeakArea[];
  weakTopics: ReportWeakArea[];
  breakdown: ReportQuestionRow[];
}

// ADDED: renders "c) <option text>" when the option set and the letter are
// both available, falling back to the bare letter (or "—") otherwise — same
// fallback shape the report already used for numeric/subjective questions and
// for skipped questions (selected_answer null).
function formatAnswer(letter: string | null, options: Record<string, string> | null): string {
  if (!letter) return '—';
  const text = options?.[letter];
  return text ? `${letter}) ${text}` : letter;
}

// Matches spec section 4 exactly — a single markdown block a student
// copy-pastes alongside their rough work into an LLM, not a file the LLM
// has to parse from PDF. Kept as a pure function (no fetching, no DOM) so
// it's testable on its own and reusable from both the results page and,
// if Checkmate wants it later, a plain API response.
export function buildTestReportMarkdown(input: TestReportInput): string {
  const lines: string[] = [];

  lines.push(`# Test Report — ${input.testTitle}`);
  lines.push(
    `Score: ${input.score}/${input.totalMarks} (${input.accuracyPct}%) | Attempted: ${input.attempted} | Correct: ${input.correct} | Wrong: ${input.incorrect} | Skipped: ${input.skipped}`
  );
  lines.push('');

  lines.push('## Weak chapters (by accuracy)');
  if (input.weakChapters.length === 0) {
    lines.push('Not enough attempted questions to break this down.');
  } else {
    input.weakChapters.forEach((c, i) => {
      const correctCount = Math.round((c.accuracy_pct / 100) * c.attempted);
      lines.push(`${i + 1}. ${c.chapter} — ${c.accuracy_pct}% (${correctCount}/${c.attempted})`);
    });
  }
  lines.push('');

  lines.push('## Weak topics');
  if (input.weakTopics.length === 0) {
    lines.push('Not enough attempted questions to break this down.');
  } else {
    input.weakTopics.forEach((t, i) => {
      const correctCount = Math.round((t.accuracy_pct / 100) * t.attempted);
      lines.push(`${i + 1}. ${t.topic} — ${t.accuracy_pct}% (${correctCount}/${t.attempted})`);
    });
  }
  lines.push('');

  lines.push('## Question-by-question');
  for (const q of input.breakdown) {
    const status = q.selected_answer ? (q.is_correct ? 'CORRECT' : 'WRONG') : 'SKIPPED';
    const timeSuffix = q.time_spent_seconds ? ` — time ${q.time_spent_seconds}s` : '';
    lines.push(`Q${q.question_number} [${q.chapter ?? '—'}/${q.topic ?? '—'}] — ${status}${timeSuffix}`);
    lines.push(`  Q: ${q.question_text}`);
    // ADDED: full option set, only when this question actually has one (mcq
    // type — numeric/subjective questions have no options and skip this
    // line). Sorted by label rather than object insertion order, since
    // Supabase's jsonb column doesn't guarantee key order survives a round
    // trip.
    if (q.options) {
      const optionsLine = Object.keys(q.options)
        .sort()
        .map((label) => `${label}) ${q.options![label]}`)
        .join(' | ');
      lines.push(`  Options: ${optionsLine}`);
    }
    lines.push(
      `  Your answer: ${formatAnswer(q.selected_answer, q.options)} | Correct: ${formatAnswer(q.correct_answer, q.options)}`
    );
  }

  return lines.join('\n');
}


//===== TERMUXFILE: Testmate/app/results/[id]/page.tsx =====
import { redirect } from 'next/navigation';
import { getCurrentUser } from '@/lib/auth';
import { getServiceClient } from '@/lib/supabase';
import { buildTestReportMarkdown } from '@/lib/markdown-export';
import ReportActions from './ReportActions';

interface WeakAreaRow {
  accuracy_pct: number;
  attempted: number;
  chapter?: string;
  topic?: string;
}

export default async function ResultsPage({ params }: { params: { id: string } }) {
  const user = await getCurrentUser();
  if (!user) {
    redirect(`/login?next=/results/${params.id}`);
  }

  const supabase = getServiceClient();

  const { data: result } = await supabase
    .from('result_summaries')
    .select('*')
    .eq('session_id', params.id)
    .eq('user_id', user.id)
    .single();

  if (!result) {
    return (
      <main className="results-shell">
        <p>No results yet. Have you submitted this test?</p>
        <p>
          <a href={`/test/${params.id}`}>Back to test</a>
        </p>
      </main>
    );
  }

  const { data: session } = await supabase
    .from('test_sessions')
    .select('test_id, mode')
    .eq('id', params.id)
    .single();

  const { data: test } = await supabase
    .from('tests')
    .select('title')
    .eq('id', session?.test_id)
    .single();

  // CHANGED: added `options` to this select — was previously
  // 'id, question_number, question_text, correct_answer, chapter, topic'.
  // Nothing downstream of Testmate could ever see answer-choice text (only
  // the correct/selected letter), which meant Checkmate's ErrorEngine had no
  // evidence to diagnose *why* a wrong option was picked. `options` already
  // existed on this table (populated at question-upload time — see
  // TestDetailEditor.tsx) and was simply never selected here.
  const { data: questions } = await supabase
    .from('questions')
    .select('id, question_number, question_text, correct_answer, chapter, topic, options')
    .eq('test_id', session?.test_id)
    .order('question_number', { ascending: true });

  const { data: responses } = await supabase
    .from('responses')
    .select('question_id, selected_answer, is_correct, time_spent_seconds')
    .eq('session_id', params.id)
    .eq('user_id', user.id);

  const responseByQuestion = new Map((responses ?? []).map((r) => [r.question_id, r]));
  const breakdown = (questions ?? []).map((q) => {
    const r = responseByQuestion.get(q.id);
    return {
      question_number: q.question_number,
      question_text: q.question_text,
      chapter: q.chapter,
      topic: q.topic,
      correct_answer: q.correct_answer,
      selected_answer: r?.selected_answer ?? null,
      is_correct: r?.is_correct ?? null,
      time_spent_seconds: r?.time_spent_seconds ?? 0,
      // ADDED: passed straight through to buildTestReportMarkdown — see that
      // file for the rendering change and ReportQuestionRow's doc for why.
      options: (q.options as Record<string, string> | null) ?? null
    };
  });

  const weakChapters = (result.weak_chapters ?? []) as WeakAreaRow[];
  const weakTopics = (result.weak_topics ?? []) as WeakAreaRow[];
  const testTitle = test?.title ?? 'Test';

  const reportMarkdown = buildTestReportMarkdown({
    testTitle,
    score: result.score,
    totalMarks: result.total_marks,
    accuracyPct: result.accuracy_pct,
    attempted: result.attempted_count,
    correct: result.correct_count,
    incorrect: result.incorrect_count,
    skipped: result.skipped_count,
    weakChapters,
    weakTopics,
    breakdown
  });

  return (
    <main className="results-shell">
      <h1 className="results-title">{testTitle} — Results</h1>

      {session?.mode === 'group' && (
        <p className="results-subtitle">
          <a href={`/sessions/${params.id}/leaderboard`}>View leaderboard →</a>
        </p>
      )}

      <div className="score-seal">
        <span className="score-seal-label">Score</span>
        <span className="score-seal-value">{result.score}</span>
        <span className="score-seal-total">/ {result.total_marks}</span>
      </div>

      <ReportActions markdown={reportMarkdown} testTitle={testTitle} />

      <div className="stat-strip">
        <Stat label="Accuracy" value={`${result.accuracy_pct}%`} />
        <Stat label="Attempted" value={String(result.attempted_count)} />
        <Stat label="Correct" value={String(result.correct_count)} />
        <Stat label="Wrong" value={String(result.incorrect_count)} />
        <Stat label="Skipped" value={String(result.skipped_count)} />
        <Stat label="Avg time/Q" value={`${result.avg_time_per_question}s`} />
      </div>

      <section className="sheet-section">
        <h2 className="sheet-heading">Weak chapters</h2>
        {weakChapters.length === 0 ? (
          <p className="sheet-empty">Not enough attempted questions to break this down.</p>
        ) : (
          <ul className="sheet-list">
            {weakChapters.map((item, i) => (
              <li key={i} className="sheet-row">
                <span className="sheet-row-name">{item.chapter}</span>
                <span className="sheet-row-score">
                  {item.accuracy_pct}% ({Math.round((item.accuracy_pct / 100) * item.attempted)}/
                  {item.attempted})
                </span>
              </li>
            ))}
          </ul>
        )}
      </section>

      <section className="sheet-section">
        <h2 className="sheet-heading">Weak topics</h2>
        {weakTopics.length === 0 ? (
          <p className="sheet-empty">Not enough attempted questions to break this down.</p>
        ) : (
          <ul className="sheet-list">
            {weakTopics.map((item, i) => (
              <li key={i} className="sheet-row">
                <span className="sheet-row-name">{item.topic}</span>
                <span className="sheet-row-score">
                  {item.accuracy_pct}% ({Math.round((item.accuracy_pct / 100) * item.attempted)}/
                  {item.attempted})
                </span>
              </li>
            ))}
          </ul>
        )}
      </section>

      <section className="sheet-section">
        <h2 className="sheet-heading">Question-by-question</h2>
        <ol className="sheet-list">
          {breakdown.map((q) => {
            const status = q.selected_answer ? (q.is_correct ? 'CORRECT' : 'WRONG') : 'SKIPPED';
            const tagClass =
              status === 'CORRECT'
                ? 'status-tag status-tag--correct'
                : status === 'WRONG'
                ? 'status-tag status-tag--wrong'
                : 'status-tag status-tag--skipped';
            return (
              <li key={q.question_number} className="qrow">
                <div className="qrow-head">
                  <span>Q{q.question_number}</span>
                  <span className={tagClass}>{status}</span>
                  <span className="qrow-detail">
                    [{q.chapter ?? '—'}/{q.topic ?? '—'}]
                    {q.time_spent_seconds ? ` · ${q.time_spent_seconds}s` : ''}
                  </span>
                </div>
                <div className="qrow-detail">
                  Your answer: {q.selected_answer ?? '—'} · Correct: {q.correct_answer ?? '—'}
                </div>
              </li>
            );
          })}
        </ol>
      </section>
    </main>
  );
}

function Stat({ label, value }: { label: string; value: string }) {
  return (
    <div className="stat-cell">
      <div className="stat-cell-label">{label}</div>
      <div className="stat-cell-value">{value}</div>
    </div>
  );
}

import { NextRequest, NextResponse } from 'next/server';
import { getServiceClient } from '@/lib/supabase';
import { getCurrentUserFromRequest } from '@/lib/auth';

// GET /api/sessions/:id/results
// Result summary for the current user plus a question-by-question
// breakdown (built by diffing the test's full question list against saved
// responses, so skipped questions show up too — a plain join on `responses`
// alone would silently drop them since skipped questions have no row).
//
// Auth accepts either the browser session cookie (web app) or
// `Authorization: Bearer <device token>` (Checkmate / other external
// clients — Phase 6, test-platform-blueprint.md §5) via
// getCurrentUserFromRequest.
//
// Response shape: the result_summaries row's columns spread at the top
// level, plus `test_title` (joined in — result_summaries itself has no
// title) and `breakdown`. This was changed from the previous
// `{ result: {...}, breakdown }` nested shape; nothing else in this repo
// consumes this API route (app/results/[id]/page.tsx queries Supabase
// directly), so the change is safe.
//
// CHANGED: `breakdown` rows now include `options` — this route is the one
// external clients (Checkmate) actually hit, since the page above renders
// server-side HTML, not JSON. Without this, any Checkmate integration that
// switches from report.md copy/paste to calling this route directly would
// have hit the exact same missing-option-text gap markdown-export.ts just
// fixed, just via a different path.
export async function GET(req: NextRequest, { params }: { params: { id: string } }) {
  const user = await getCurrentUserFromRequest(req);
  if (!user) {
    return NextResponse.json({ error: 'Not authenticated' }, { status: 401 });
  }

  const supabase = getServiceClient();

  const { data: result, error } = await supabase
    .from('result_summaries')
    .select('*')
    .eq('session_id', params.id)
    .eq('user_id', user.id)
    .single();

  if (error || !result) {
    return NextResponse.json({ error: 'No result yet — submit the test first.' }, { status: 404 });
  }

  const { data: session } = await supabase
    .from('test_sessions')
    .select('test_id')
    .eq('id', params.id)
    .single();

  const { data: test } = await supabase
    .from('tests')
    .select('title')
    .eq('id', session?.test_id)
    .single();

  const { data: questions } = await supabase
    .from('questions')
    .select('id, question_number, question_text, correct_answer, chapter, topic, options')
    .eq('test_id', session?.test_id)
    .order('question_number', { ascending: true });

  const { data: responses } = await supabase
    .from('responses')
    .select('question_id, selected_answer, is_correct, time_spent_seconds')
    .eq('session_id', params.id)
    .eq('user_id', user.id);

  const responseByQuestion = new Map((responses ?? []).map((r) => [r.question_id, r]));
  const breakdown = (questions ?? []).map((q) => {
    const r = responseByQuestion.get(q.id);
    return {
      question_number: q.question_number,
      question_text: q.question_text,
      chapter: q.chapter,
      topic: q.topic,
      correct_answer: q.correct_answer,
      selected_answer: r?.selected_answer ?? null,
      is_correct: r?.is_correct ?? null,
      time_spent_seconds: r?.time_spent_seconds ?? 0,
      options: (q.options as Record<string, string> | null) ?? null
    };
  });

  return NextResponse.json({
    ...result,
    test_title: test?.title ?? 'Test',
    breakdown
  });
}
