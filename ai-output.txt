/**
 * TelegramAlertBot — sends screenshot + caption to guardian via the Cloudflare Worker relay.
 *
 * The Telegram bot token no longer lives in the app. It is a Worker secret; the app only
 * talks to the Worker's /tg/send and /tg/photo routes, authenticated with a rotatable relay
 * key (relay_key in local.properties -> BuildConfig.RELAY_KEY). Public API is unchanged, so
 * no call site needs to change.
 *
 * Guardian onboarding:
 *   1. Guardian opens Telegram -> searches the bot -> sends /start
 *   2. Guardian messages @userinfobot to get their chat_id
 *   3. Student enters that chat_id in Settings -> Guardian Telegram Chat ID
 *
 * Call sendAlert() / uploadPhotoAndGetFileId() from a background thread — both block on network.
 */
object TelegramAlertBot {

    private const val TAG = "TelegramAlertBot"
    private const val RELAY_BASE = "https://steep-band-1bd0.amirshamse8.workers.dev"
    private val RELAY_KEY get() = BuildConfig.RELAY_KEY

    private val client = OkHttpClient.Builder()
        .connectTimeout(15, TimeUnit.SECONDS)
        .readTimeout(30, TimeUnit.SECONDS)
        .writeTimeout(30, TimeUnit.SECONDS)
        .build()

    // ── Public API ────────────────────────────────────────────────────────────

    /**
     * Send photo + caption to guardian.
     * Falls back to text-only if screenshotUri is null or upload fails.
     * Must be called from a background thread.
     */
    fun sendAlert(context: Context, caption: String, screenshotUri: Uri? = null) {
        if (RELAY_KEY.isBlank()) {
            Log.e(TAG, "relay_key not set in local.properties — skipping Telegram alert")
            return
        }
        val chatId = getChatId() ?: run {
            Log.w(TAG, "telegram_chat_id not set — skipping alert")
            return
        }

        if (screenshotUri != null) {
            val tmp = copyUriToTempFile(context, screenshotUri)
            if (tmp != null) {
                val sent = sendPhoto(chatId, caption, tmp) != null
                tmp.delete()
                if (sent) return
                Log.w(TAG, "sendPhoto failed — falling back to text")
            }
        }

        sendText(chatId, caption)
    }

    /**
     * Uploads a screenshot to the guardian's chat with the given caption and
     * returns Telegram's file_id for the largest photo size, or null on failure.
     * The file_id is reusable by the Worker to re-send the same image.
     *
     * Must be called from a background thread.
     */
    fun uploadPhotoAndGetFileId(context: Context, caption: String, screenshotUri: Uri): String? {
        if (RELAY_KEY.isBlank()) {
            Log.e(TAG, "relay_key not set in local.properties — skipping status photo upload")
            return null
        }
        val chatId = getChatId() ?: run {
            Log.w(TAG, "telegram_chat_id not set — skipping status photo upload")
            return null
        }

        val tmp = copyUriToTempFile(context, screenshotUri) ?: return null
        return try {
            sendPhoto(chatId, caption, tmp)
        } finally {
            tmp.delete()
        }
    }

    // ── Internal ──────────────────────────────────────────────────────────────

    /** Returns the file_id on success (may be empty string if Worker omitted it), null on failure. */
    private fun sendPhoto(chatId: String, caption: String, photo: File): String? {
        return try {
            val body = MultipartBody.Builder()
                .setType(MultipartBody.FORM)
                .addFormDataPart("chat_id", chatId)
                .addFormDataPart("caption", caption)
                .addFormDataPart("photo", photo.name, photo.asRequestBody("image/png".toMediaType()))
                .build()

            client.newCall(
                Request.Builder()
                    .url("$RELAY_BASE/tg/photo")
                    .addHeader("X-Relay-Key", RELAY_KEY)
                    .post(body)
                    .build()
            ).execute().use { response ->
                val bodyStr = response.body?.string()
                Log.d(TAG, "sendPhoto: ${response.code}")
                if (!response.isSuccessful || bodyStr.isNullOrBlank()) return null
                val json = JSONObject(bodyStr)
                if (!json.optBoolean("ok", false)) return null
                json.optString("fileId", "")
            }
        } catch (e: Exception) {
            Log.e(TAG, "sendPhoto exception: ${e.message}")
            null
        }
    }

    private fun sendText(chatId: String, text: String) {
        try {
            val payload = JSONObject().apply {
                put("chat_id", chatId)
                put("text", text)
            }
            client.newCall(
                Request.Builder()
                    .url("$RELAY_BASE/tg/send")
                    .addHeader("X-Relay-Key", RELAY_KEY)
                    .post(payload.toString().toRequestBody("application/json".toMediaType()))
                    .build()
            ).execute().use { Log.d(TAG, "sendText: ${it.code}") }
        } catch (e: Exception) {
            Log.e(TAG, "sendText exception: ${e.message}")
        }
    }

    private fun copyUriToTempFile(context: Context, uri: Uri): File? {
        return try {
            val tmp = File(context.cacheDir, "tg_${System.currentTimeMillis()}.png")
            context.contentResolver.openInputStream(uri)?.use { input ->
                FileOutputStream(tmp).use { output -> input.copyTo(output) }
            }
            tmp
        } catch (e: Exception) {
            Log.e(TAG, "copyUriToTempFile: ${e.message}")
            null
        }
    }

    fun getChatId(): String? {
        val id = CheckmatePrefs.getString("telegram_chat_id", null)
        return if (id.isNullOrBlank()) null else id.trim()
    }
}

export default {
  async fetch(request, env) {
    const url = new URL(request.url);

    // ── Consistency stats snapshot backup: same TASKS_KV binding, "stats:<code>"
    // prefix. Powers StatsSyncManager on the Android side (backs up the
    // Consistency tab's computed numbers — streak, completion %, calendar,
    // attention/pause stats, focus score. Screen-usage-time is deliberately
    // NOT included here — that stays on-device only).
    if (url.pathname === "/stats" && request.method === "POST") {
      const body = await request.json();
      if (!body.code) return new Response("missing code", { status: 400 });
      await env.TASKS_KV.put(`stats:${body.code}`, JSON.stringify(body));
      return new Response("ok");
    }
    if (url.pathname === "/stats" && request.method === "GET") {
      const code = url.searchParams.get("code");
      if (!code) return new Response("missing code", { status: 400 });
      const cached = await env.TASKS_KV.get(`stats:${code}`);
      if (!cached) {
        return new Response(JSON.stringify({ updatedAt: 0, stats: null }), {
          headers: { "content-type": "application/json" }
        });
      }
      return new Response(cached, { headers: { "content-type": "application/json" } });
    }

    // ── Setup Profile backup: same TASKS_KV binding, "profile:<code>" prefix
    // so it can't collide with "tasks:<code>" keys. Powers ProfileSyncManager
    // on the Android side (backs up exam type/date, study window, guardian
    // number, tts, subjects — NOT today's task list, that's still /tasks).
    if (url.pathname === "/profile" && request.method === "POST") {
      const body = await request.json();
      if (!body.code) return new Response("missing code", { status: 400 });
      await env.TASKS_KV.put(`profile:${body.code}`, JSON.stringify(body));
      return new Response("ok");
    }
    if (url.pathname === "/profile" && request.method === "GET") {
      const code = url.searchParams.get("code");
      if (!code) return new Response("missing code", { status: 400 });
      const cached = await env.TASKS_KV.get(`profile:${code}`);
      if (!cached) {
        return new Response(JSON.stringify({ updatedAt: 0, profile: null }), {
          headers: { "content-type": "application/json" }
        });
      }
      return new Response(cached, { headers: { "content-type": "application/json" } });
    }

    // ── Outcome Ledger backup: same TASKS_KV binding, "outcomes:<code>" prefix.
    // Powers OutcomeLedgerSyncManager (Android) — whole-table push/pull of
    // resolved intervention transactions (Step 12). The client always sends its
    // FULL local table on push (see OutcomeLedgerSyncManager.pushLedger), so
    // this is a flat overwrite, same as /profile and /stats above — no
    // read-modify-write needed. This route was previously missing entirely,
    // so every pushLedger()/pullLedgerIfLocalEmpty() call has been a 404 until now.
    if (url.pathname === "/outcomes" && request.method === "POST") {
      const body = await request.json();
      if (!body.code) return new Response("missing code", { status: 400 });
      await env.TASKS_KV.put(`outcomes:${body.code}`, JSON.stringify(body));
      return new Response("ok");
    }
    if (url.pathname === "/outcomes" && request.method === "GET") {
      const code = url.searchParams.get("code");
      if (!code) return new Response("missing code", { status: 400 });
      const cached = await env.TASKS_KV.get(`outcomes:${code}`);
      if (!cached) {
        return new Response(JSON.stringify({ updatedAt: 0, entries: [] }), {
          headers: { "content-type": "application/json" }
        });
      }
      return new Response(cached, { headers: { "content-type": "application/json" } });
    }

    // ── Behavior Ledger backup: same TASKS_KV binding, "behavior:<code>" prefix.
    // Powers BehaviorLedgerSyncManager (Android) — whole-table push/pull of the
    // psyche module's behavior event log (capped at 200 rows on the client, so
    // this stays small regardless of how long the student's used the app). Flat
    // overwrite, same reasoning as /outcomes above. Also previously missing.
    if (url.pathname === "/behavior" && request.method === "POST") {
      const body = await request.json();
      if (!body.code) return new Response("missing code", { status: 400 });
      await env.TASKS_KV.put(`behavior:${body.code}`, JSON.stringify(body));
      return new Response("ok");
    }
    if (url.pathname === "/behavior" && request.method === "GET") {
      const code = url.searchParams.get("code");
      if (!code) return new Response("missing code", { status: 400 });
      const cached = await env.TASKS_KV.get(`behavior:${code}`);
      if (!cached) {
        return new Response(JSON.stringify({ updatedAt: 0, events: [] }), {
          headers: { "content-type": "application/json" }
        });
      }
      return new Response(cached, { headers: { "content-type": "application/json" } });
    }

    // ── Test report backup: same TASKS_KV binding, "reports:<code>" prefix.
    // Powers LearningReportSyncManager (Android). UNLIKE every route above,
    // this is an APPEND-ONLY LIST, not a flat overwrite — each report.md the
    // student imports is one immutable historical event, deduped by reportId
    // (a content hash the app computes), not a "current state" blob two
    // devices might race on. POST reads the existing array, appends only if
    // reportId isn't already present, writes it back.
    if (url.pathname === "/reports" && request.method === "POST") {
      const body = await request.json();
      if (!body.code || !body.reportId || !body.reportText) {
        return new Response("missing code/reportId/reportText", { status: 400 });
      }
      const key = `reports:${body.code}`;
      const existingRaw = await env.TASKS_KV.get(key);
      const existing = existingRaw ? JSON.parse(existingRaw) : [];
      if (!existing.some((r) => r.reportId === body.reportId)) {
        existing.push({
          reportId: body.reportId,
          pushedAt: body.pushedAt || Date.now(),
          reportText: body.reportText
        });
        await env.TASKS_KV.put(key, JSON.stringify(existing));
      }
      return new Response("ok");
    }
    if (url.pathname === "/reports" && request.method === "GET") {
      const code = url.searchParams.get("code");
      if (!code) return new Response("missing code", { status: 400 });
      const cached = await env.TASKS_KV.get(`reports:${code}`);
      const reports = cached ? JSON.parse(cached) : [];
      return new Response(JSON.stringify({ reports }), {
        headers: { "content-type": "application/json" }
      });
    }

    // ── Bounded "current state" backup: same TASKS_KV binding, "studystate:<code>"
    // prefix. Powers StudyStateSyncManager (Android) — Mentor chat history,
    // checklist template, coaching-test countdown entries. Flat overwrite, same
    // reasoning as /profile.
    if (url.pathname === "/studystate" && request.method === "POST") {
      const body = await request.json();
      if (!body.code) return new Response("missing code", { status: 400 });
      await env.TASKS_KV.put(`studystate:${body.code}`, JSON.stringify(body));
      return new Response("ok");
    }
    if (url.pathname === "/studystate" && request.method === "GET") {
      const code = url.searchParams.get("code");
      if (!code) return new Response("missing code", { status: 400 });
      const cached = await env.TASKS_KV.get(`studystate:${code}`);
      if (!cached) {
        return new Response(JSON.stringify({ updatedAt: 0, state: {} }), {
          headers: { "content-type": "application/json" }
        });
      }
      return new Response(cached, { headers: { "content-type": "application/json" } });
    }

    // ── Full day-by-day history backup: same TASKS_KV binding, "dayhistory:<code>"
    // prefix. Powers DayHistorySyncManager (Android) — plan_<day>/checklist_<day>/
    // daily_checkin_<day> for every day the app has locally. The client always
    // sends its FULL locally-known set on push (see DayHistorySyncManager.pushHistory),
    // so this is a flat overwrite like /profile/​/stats — the per-day MERGE logic
    // (only fill in days the device doesn't already have) lives entirely on the
    // Android side in pullMissingDays(), not here.
    if (url.pathname === "/dayhistory" && request.method === "POST") {
      const body = await request.json();
      if (!body.code) return new Response("missing code", { status: 400 });
      await env.TASKS_KV.put(`dayhistory:${body.code}`, JSON.stringify(body));
      return new Response("ok");
    }
    if (url.pathname === "/dayhistory" && request.method === "GET") {
      const code = url.searchParams.get("code");
      if (!code) return new Response("missing code", { status: 400 });
      const cached = await env.TASKS_KV.get(`dayhistory:${code}`);
      if (!cached) {
        return new Response(JSON.stringify({ updatedAt: 0, plans: {}, checklists: {}, checkins: {} }), {
          headers: { "content-type": "application/json" }
        });
      }
      return new Response(cached, { headers: { "content-type": "application/json" } });
    }

    // ── Task Sync: must be checked before the generic "non-POST = health
    // check" fallback below, since GET is a valid method for pulling tasks.
    if (url.pathname === "/tasks" && request.method === "POST") {
      const body = await request.json();
      if (!body.code) return new Response("missing code", { status: 400 });
      await env.TASKS_KV.put(`tasks:${body.code}`, JSON.stringify(body));
      return new Response("ok");
    }
    if (url.pathname === "/tasks" && request.method === "GET") {
      const code = url.searchParams.get("code");
      if (!code) return new Response("missing code", { status: 400 });
      const cached = await env.TASKS_KV.get(`tasks:${code}`);
      if (!cached) {
        return new Response(JSON.stringify({ dayKey: "", updatedAt: 0, tasks: [] }), {
          headers: { "content-type": "application/json" }
        });
      }
      return new Response(cached, { headers: { "content-type": "application/json" } });
    }

    // ── Telegram relay: app sends alerts through here; BOT_TOKEN never leaves the Worker ──
    if (url.pathname === "/tg/send" || url.pathname === "/tg/photo") {
      return handleTg(request, env, url);
    }

    // Browser visit / health check
    if (request.method !== "POST") {
      return new Response("Bot is running ✅");
    }

    // ── App pushes status + screenshot info ───────────────────
    if (url.pathname === "/status") {
      const body = await request.json();
      await env.STATUS_KV.put(body.chatId, JSON.stringify(body), { expirationTtl: 600 });
      return new Response("ok");
    }

    // ── Telegram webhook (parent messages the bot) ────────────
    // Telegram sends the secret_token set via setWebhook in this header. Reject forged updates.
    if (!env.WEBHOOK_SECRET ||
        request.headers.get("X-Telegram-Bot-Api-Secret-Token") !== env.WEBHOOK_SECRET) {
      return new Response("unauthorized", { status: 401 });
    }
    const update = await request.json();
    const chatId = update?.message?.chat?.id;
    const text = (update?.message?.text || "").toLowerCase();

    if (!chatId) return new Response("ok");

    const BOT_TOKEN = env.BOT_TOKEN;

    if (text.includes("status") || text.includes("doing")) {
      const cached = await env.STATUS_KV.get(chatId.toString());

      if (!cached) {
        await sendMessage(BOT_TOKEN, chatId,
          "No recent data — phone may be off or app closed.");
        return new Response("ok");
      }

      const s = JSON.parse(cached);
      const mins = Math.floor(s.secondsLeft / 60);
      const ageMin = Math.floor((Date.now() - s.timestamp) / 60000);

      let caption =
        `📚 Phase: ${s.phase}\n` +
        `⏱ ${mins}m left in this block\n` +
        `🎯 Cycle ${s.cycleIndex}\n` +
        `⚠️ Checks missed: ${s.checksMissed}\n\n` +
        `(updated ${ageMin}m ago)`;

      if (s.fileId) {
        const shotAge = Math.floor((Date.now() - s.screenshotTimestamp) / 60000);
        caption += `\n📸 Screenshot from ${shotAge}m ago`;
        await sendPhoto(BOT_TOKEN, chatId, s.fileId, caption);
      } else {
        caption += `\n📸 No recent screenshot available.`;
        await sendMessage(BOT_TOKEN, chatId, caption);
      }
    } else {
      // Default: reply with chat ID for setup
      await sendMessage(
        BOT_TOKEN,
        chatId,
        `Your Checkmate Chat ID is:\n${chatId}\n\nPaste this into Settings → Guardian Telegram Chat ID.`
      );
    }

    return new Response("ok");
  }
};

async function sendMessage(token, chatId, text) {
  await fetch(`https://api.telegram.org/bot${token}/sendMessage`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ chat_id: chatId, text })
  });
}

async function sendPhoto(token, chatId, fileId, caption) {
  await fetch(`https://api.telegram.org/bot${token}/sendPhoto`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ chat_id: chatId, photo: fileId, caption })
  });
}

async function handleTg(request, env, url) {
  if (request.method !== "POST") return new Response("method", { status: 405 });
  if (!env.RELAY_KEY || request.headers.get("X-Relay-Key") !== env.RELAY_KEY) {
    return new Response("unauthorized", { status: 401 });
  }
  const allowed = (env.ALLOWED_CHAT_IDS || "").split(",").map(x => x.trim()).filter(Boolean);
  const api = `https://api.telegram.org/bot${env.BOT_TOKEN}`;

  if (url.pathname === "/tg/send") {
    const { chat_id, text } = await request.json();
    if (!allowed.includes(String(chat_id))) return new Response("chat not allowed", { status: 403 });
    const r = await fetch(`${api}/sendMessage`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ chat_id, text: String(text || "").slice(0, 4096) })
    });
    return new Response(await r.text(), { status: r.status });
  }

  // /tg/photo — multipart in (chat_id, caption, photo), JSON out {ok, fileId}
  const form = await request.formData();
  const chat_id = String(form.get("chat_id") || "");
  if (!allowed.includes(chat_id)) return new Response("chat not allowed", { status: 403 });
  const out = new FormData();
  out.append("chat_id", chat_id);
  out.append("caption", String(form.get("caption") || "").slice(0, 1024));
  out.append("photo", form.get("photo"));
  const r = await fetch(`${api}/sendPhoto`, { method: "POST", body: out });
  const j = await r.json().catch(() => ({}));
  const sizes = j?.result?.photo || [];
  const fileId = sizes.length ? sizes[sizes.length - 1].file_id : "";
  return new Response(JSON.stringify({ ok: !!j.ok, fileId }), {
    status: r.ok ? 200 : 502,
    headers: { "Content-Type": "application/json" }
  });
}
