/**
 * Checkmate MCP gateway — read-projection push (FINAL_PLAN "Option C — hybrid").
 *
 * Checkmate stays the source of truth for the student model, the intervention history and
 * today's plan. This pushes curated SNAPSHOTS of those three things to Testmate's
 * `PUT /api/gateway/projection`, using the SAME base URL + device token the rest of
 * Checkmate already uses for Testmate (Settings -> Test Platform). The MCP gateway on the
 * Testmate side only ever reads those stored snapshots — it never reaches into this phone,
 * and keeps answering (with a freshness stamp) while the phone is offline.
 *
 * Six snapshots: student_model, interventions, today_plan, plus knowledge_graph (the seeded
 * prerequisite edges, re-sent at most once a day unless they change), behavior (AGGREGATE
 * numbers only) and task_history (the last two weeks of saved daily plans).
 *
 * What is deliberately NOT pushed: the raw behavior event ledger (only counts derived from
 * it), which app distracted the student, screen/app-usage data, profile/consultation text,
 * API keys, anything outside the snapshots below. The server re-normalizes every field it
 * accepts, so this file is the first filter, not the only one.
 *
 * Fire-and-forget background sync: a periodic WorkManager job (network required) plus one
 * immediate run at app start. Each snapshot is pushed independently — one failing (e.g. an
 * empty student model on a fresh install) never blocks the others.
 */
object GatewaySync {

    private const val TAG = "GatewaySync"
    private const val MAX_INTERVENTIONS = 300
    private const val MAX_GRAPH_EDGES = 4000
    private const val MAX_GRAPH_NODES = 1500
    private const val HISTORY_DAYS = 14
    private const val MAX_TASKS_PER_DAY = 80
    // The prerequisite graph is seeded once and rarely changes: skip re-sending an identical
    // payload for this long (any change goes out on the next sync regardless).
    private const val GRAPH_RESEND_MS = 24L * 60 * 60 * 1000
    private const val PREF_GRAPH_HASH = "gateway_kg_hash"
    private const val PREF_GRAPH_PUSHED_MS = "gateway_kg_pushed_ms"
    private const val DAY_MS = 24L * 60 * 60 * 1000

    private val client = OkHttpClient.Builder()
        .connectTimeout(15, TimeUnit.SECONDS)
        .writeTimeout(30, TimeUnit.SECONDS)
        .readTimeout(30, TimeUnit.SECONDS)
        .build()

    enum class PushResult { OK, SKIPPED_NOT_CONFIGURED, REJECTED, RETRYABLE_FAILURE }

    /** Pushes every snapshot. Returns the worst outcome so the Worker can decide to retry. */
    suspend fun pushAll(context: Context): PushResult = withContext(Dispatchers.IO) {
        val base = baseUrl()
        val token = token()
        if (base == null || token == null) return@withContext PushResult.SKIPPED_NOT_CONFIGURED

        // `map` is inline, so the suspend builders are legal here (a lambda stored in a
        // listOf(...) is NOT a suspend lambda — that was the compile error).
        val kinds = listOf("student_model", "interventions", "today_plan", "knowledge_graph", "behavior", "task_history")
        val results = kinds.map { kind ->
            try {
                val payload = when (kind) {
                    "student_model" -> buildStudentModel(context)
                    "interventions" -> buildInterventions(context)
                    "today_plan" -> buildTodayPlan()
                    "knowledge_graph" -> buildKnowledgeGraph(context)
                    "behavior" -> buildBehavior(context)
                    else -> buildTaskHistory()
                }
                if (kind == "knowledge_graph") pushKnowledgeGraph(base, token, payload) else push(base, token, kind, payload)
            } catch (e: CancellationException) {
                throw e
            } catch (e: Exception) {
                Log.w(TAG, "push $kind failed to build/send: ${e.message}")
                PushResult.RETRYABLE_FAILURE
            }
        }
        when {
            PushResult.REJECTED in results -> PushResult.REJECTED
            PushResult.RETRYABLE_FAILURE in results -> PushResult.RETRYABLE_FAILURE
            else -> PushResult.OK
        }
    }

    // ── config (same prefs + allow-list as TestmateApi; never accepts an arbitrary host) ──
    private fun baseUrl(): String? {
        val saved = CheckmatePrefs.getString(TestmateApi.PREF_BASE_URL, null)?.trim()?.trimEnd('/')
            ?.takeIf { it.isNotBlank() } ?: return null
        if (!TestmateApi.isAllowedBaseUrl(saved)) {
            Log.w(TAG, "Stored Testmate base URL failed the allow-list check — not syncing")
            return null
        }
        return saved
    }

    private fun token(): String? =
        CheckmatePrefs.getString(TestmateApi.PREF_TOKEN, null)?.trim()?.takeIf { it.isNotBlank() }

    private fun push(base: String, token: String, kind: String, payload: JSONObject): PushResult {
        payload.put("generated_ms", System.currentTimeMillis())
        val body = payload.toString().toRequestBody("application/json".toMediaType())
        val request = Request.Builder()
            .url("$base/api/gateway/projection?kind=$kind")
            .addHeader("Authorization", "Bearer $token")
            .put(body)
            .build()
        client.newCall(request).execute().use { resp ->
            Log.d(TAG, "push $kind: HTTP ${resp.code}")
            return when {
                resp.isSuccessful -> PushResult.OK
                // 400/401/403/413: retrying the identical payload/credentials can't succeed.
                resp.code in 400..499 && resp.code != 408 && resp.code != 429 -> PushResult.REJECTED
                else -> PushResult.RETRYABLE_FAILURE
            }
        }
    }

    // org.json throws on NaN/Infinity — skip non-finite numbers instead of failing the whole push.
    private fun JSONObject.putNum(key: String, v: Double?): JSONObject {
        if (v != null && v.isFinite()) put(key, v)
        return this
    }

    private fun JSONObject.putStr(key: String, v: String?): JSONObject {
        if (!v.isNullOrBlank() && v != "null") put(key, v)
        return this
    }

    // ── student_model ─────────────────────────────────────────────────────────
    private suspend fun buildStudentModel(context: Context): JSONObject {
        val model: StudentModel = StudentModelBuilder.build(context)
        val root = JSONObject()
        root.putStr("exam", runCatching { ConsultationProfile.load().examTarget }.getOrNull())
        root.put("model_version", model.modelVersion)
        root.put(
            "overall",
            JSONObject().apply {
                put("concepts_tracked", model.overall.conceptsTracked)
                put("concepts_mastered", model.overall.conceptsMastered)
                put("concepts_weak", model.overall.conceptsWeak)
                putNum("average_mastery", model.overall.averageMastery)
                put("total_attempts", model.overall.totalAttempts)
                put("unresolved_error_count", model.overall.unresolvedErrorCount)
            }
        )

        val concepts = JSONArray()
        for (c in model.concepts.values) {
            val issues = JSONArray()
            for (p in c.prerequisiteIssues) {
                issues.put(JSONObject().putStr("subject", p.subject).putStr("chapter", p.chapter).putStr("topic", p.topic))
            }
            concepts.put(
                JSONObject().apply {
                    putStr("concept_id", c.conceptId)
                    putStr("subject", c.subject)
                    putStr("chapter", c.chapter)
                    putStr("topic", c.topic)
                    putNum("mastery", c.mastery)
                    putNum("forgetting_risk", c.forgettingRisk)
                    put("retention_decision", c.retentionDecision.name)
                    put("attempt_count", c.attemptCount)
                    putNum("recent_accuracy", c.recentAccuracy)
                    putNum("lifetime_accuracy", c.lifetimeAccuracy)
                    put("error_count", c.errorCount)
                    c.lastSeen?.let { put("last_seen_ms", it) }
                    put("weak", c.mastery < MasteryEngine.MASTERY_THRESHOLD)
                    put("prerequisite_issues", issues)
                }
            )
        }
        root.put("concepts", concepts)

        val errors = JSONArray()
        for (e in model.unresolvedErrors) {
            val concept = model.concepts[e.conceptId]
            errors.put(
                JSONObject().apply {
                    putStr("concept_id", e.conceptId)
                    putStr("subject", concept?.subject)
                    putStr("chapter", concept?.chapter)
                    putStr("topic", concept?.topic)
                    putStr("error_type", e.errorType)
                    put("occurrences", e.occurrences)
                    put("first_seen_ms", e.firstSeen)
                    put("last_seen_ms", e.lastSeen)
                }
            )
        }
        root.put("unresolved_errors", errors)
        return root
    }

    // ── interventions (Outcome Ledger) ────────────────────────────────────────
    private suspend fun buildInterventions(context: Context): JSONObject {
        val all = InterventionDatabase.getInstance(context).outcomeLedgerDao().getAll()
        val entries = JSONArray()
        all.sortedByDescending { it.resolvedAt }.take(MAX_INTERVENTIONS).forEach { e ->
            entries.put(
                JSONObject().apply {
                    putStr("transaction_id", e.transactionId)
                    putStr("task_id", e.taskId)
                    put("trigger_type", e.triggerType.name)
                    put("terminal_state", e.terminalState.name)
                    put("provenance", e.provenance.name)
                    put("attempt_count", e.attemptCount)
                    put("resolved_ms", e.resolvedAt)
                    putStr("outcome", e.outcome)
                    putStr("failure_reason", e.failureReason)
                }
            )
        }
        return JSONObject().put("entries", entries)
    }

    // ── today_plan ────────────────────────────────────────────────────────────
    private fun buildTodayPlan(): JSONObject {
        val tasks = JSONArray()
        PlanStore.getTodayTasksSnapshot_Sync().forEach { t ->
            tasks.put(
                JSONObject().apply {
                    putStr("id", t.id)
                    putStr("subject", t.subject)
                    putStr("topic", t.topic)
                    put("duration_minutes", t.durationMinutes)
                    put("state", t.state.name)
                    put("task_type", t.taskType.name)
                    putStr("scheduled_start_time", t.scheduledStartTime)
                    putStr("completed_status", t.completedStatus)
                    putStr("learning_intent", t.learningIntent)
                    putStr("concept_id", t.conceptId)
                    putStr("rationale", t.rationale)
                    put("focus_minutes", t.focusMinutes)
                }
            )
        }
        return JSONObject().apply {
            putStr("day_key", PlanStore.currentDayKey())
            put("completion_percent", PlanStore.getTodayCompletionPercent())
            put("tasks", tasks)
        }
    }

    // ── knowledge_graph (prerequisite edges) ──────────────────────────────────
    private suspend fun buildKnowledgeGraph(context: Context): JSONObject {
        val db = LearningDatabase.getInstance(context)
        val edges = db.conceptDependencyDao().getAll().take(MAX_GRAPH_EDGES)
        val ids = edges.flatMap { listOf(it.conceptId, it.prerequisiteConceptId) }.distinct().take(MAX_GRAPH_NODES)
        // chunked: SQLite caps the number of bound variables in one IN (...) query.
        val concepts = ids.chunked(400).flatMap { db.conceptDao().getByIds(it) }
        val known = concepts.map { it.id }.toSet()

        val nodes = JSONArray()
        for (c in concepts) {
            nodes.put(
                JSONObject().apply {
                    putStr("concept_id", c.id)
                    putStr("subject", c.subject)
                    putStr("chapter", c.chapter)
                    putStr("topic", c.topic)
                }
            )
        }
        val edgeArr = JSONArray()
        for (e in edges) {
            if (e.conceptId in known && e.prerequisiteConceptId in known) {
                edgeArr.put(JSONObject().put("concept_id", e.conceptId).put("prerequisite_id", e.prerequisiteConceptId))
            }
        }
        return JSONObject().put("nodes", nodes).put("edges", edgeArr)
    }

    /** Skips an unchanged graph for [GRAPH_RESEND_MS]; records the hash only after a successful push. */
    private fun pushKnowledgeGraph(base: String, token: String, payload: JSONObject): PushResult {
        val hash = payload.toString().hashCode().toString()
        val lastHash = CheckmatePrefs.getString(PREF_GRAPH_HASH, null)
        val lastPushed = CheckmatePrefs.getLong(PREF_GRAPH_PUSHED_MS, 0L)
        if (hash == lastHash && System.currentTimeMillis() - lastPushed < GRAPH_RESEND_MS) return PushResult.OK
        val result = push(base, token, "knowledge_graph", payload)
        if (result == PushResult.OK) {
            CheckmatePrefs.putString(PREF_GRAPH_HASH, hash)
            CheckmatePrefs.putLong(PREF_GRAPH_PUSHED_MS, System.currentTimeMillis())
        }
        return result
    }

    // ── behavior (aggregates only: counts and rates, never individual events) ──
    private suspend fun buildBehavior(context: Context): JSONObject {
        val snap = BehaviorLedger.getSnapshot()
        val events = BehaviorDatabase.getInstance(context).behaviorEventDao().getAll()
        val now = System.currentTimeMillis()
        fun count(state: String, days: Int) = events.count { it.state == state && it.timestamp > now - days * DAY_MS }

        val patterns = JSONArray()
        for (p in snap.subjectPatterns) {
            patterns.put(JSONObject().putStr("subject", p.subject).putStr("task_type", p.taskType).put("occurrences", p.occurrences))
        }
        return JSONObject().apply {
            put("streak_days", snap.streakDays)
            put("consecutive_missed_days", PlanStore.getConsecutiveMissedDays())
            put("week_completion_percent", PlanStore.getWeekCompletionPercent())
            put("recent_skip_rate_percent", snap.recentSkipRatePercent)
            put("total_skips_7d", snap.totalSkips7d)
            put("today_completed", snap.todayCompleted.size)
            put(
                "attention",
                JSONObject()
                    .put("checks_passed", snap.attentionChecksPassed)
                    .put("checks_missed", snap.attentionChecksMissed)
                    .put("avg_focus_minutes", snap.avgFocusMinutes)
            )
            put("skip_patterns", patterns)
            put(
                "ledger",
                JSONObject()
                    .put("events", events.size)
                    .put("done_7d", count("DONE", 7))
                    .put("skipped_7d", count("SKIPPED", 7))
                    .put("done_30d", count("DONE", 30))
                    .put("skipped_30d", count("SKIPPED", 30))
                    .put("focus_minutes_7d", events.filter { it.state == "DONE" && it.timestamp > now - 7 * DAY_MS }.sumOf { it.focusMinutes })
            )
        }
    }

    // ── task_history (saved daily plans, newest first, today included) ────────
    private fun buildTaskHistory(): JSONObject {
        val fmt = SimpleDateFormat("yyyy-MM-dd", Locale.US)
        val cal = Calendar.getInstance()
        val days = JSONArray()
        repeat(HISTORY_DAYS) {
            val tasks = PlanStore.loadDay(PlanStore.keyForDay(cal))
            val rows = JSONArray()
            tasks.take(MAX_TASKS_PER_DAY).forEach { t ->
                rows.put(
                    JSONObject().apply {
                        putStr("id", t.id)
                        putStr("subject", t.subject)
                        putStr("topic", t.topic)
                        put("task_type", t.taskType.name)
                        put("state", t.state.name)
                        put("duration_minutes", t.durationMinutes)
                        put("focus_minutes", t.focusMinutes)
                        putStr("learning_intent", t.learningIntent)
                        putStr("completed_status", t.completedStatus)
                    }
                )
            }
            days.put(
                JSONObject().apply {
                    put("date", fmt.format(cal.time))
                    put("planned", tasks.size)
                    put("done", tasks.count { it.state == TaskState.DONE })
                    put("skipped", tasks.count { it.state == TaskState.SKIPPED })
                    put("unfinished", tasks.count { it.state != TaskState.DONE && it.state != TaskState.SKIPPED })
                    put("focus_minutes", tasks.sumOf { it.focusMinutes })
                    put("tasks", rows)
                }
            )
            cal.add(Calendar.DAY_OF_YEAR, -1)
        }
        return JSONObject().put("days", days)
    }
}

/** WorkManager entry point. Network-constrained; only retries failures that could succeed on retry. */
class GatewaySyncWorker(
    context: Context,
    params: WorkerParameters
) : CoroutineWorker(context, params) {

    override suspend fun doWork(): Result {
        val pushed = GatewaySync.pushAll(applicationContext)

        // After the snapshot push, so the server's view is fresh when the AI reads an outcome.
        // Independent of the push result and never fails the job: a request-queue problem must not
        // block snapshots, and anything unanswered is simply picked up on the next run.
        try {
            GatewayRequestSync.processPending(applicationContext)
        } catch (e: kotlinx.coroutines.CancellationException) {
            throw e
        } catch (e: Exception) {
            android.util.Log.w("GatewaySyncWorker", "request queue pass failed: ${e.message}")
        }

        return when (pushed) {
            GatewaySync.PushResult.RETRYABLE_FAILURE -> if (runAttemptCount < 3) Result.retry() else Result.success()
            // Not configured / rejected: nothing to gain from retrying; the next periodic run re-checks.
            else -> Result.success()
        }
    }
}

object GatewaySyncScheduler {

    private const val PERIODIC_WORK = "checkmate_gateway_sync"
    private const val STARTUP_WORK = "checkmate_gateway_sync_startup"

    /** WorkManager's periodic floor is 15 min; 30 keeps the snapshot fresh without wasting battery. */
    private const val INTERVAL_MINUTES = 30L

    fun schedule(context: Context) {
        val constraints = Constraints.Builder()
            .setRequiredNetworkType(NetworkType.CONNECTED)
            .build()
        val wm = WorkManager.getInstance(context)

        wm.enqueueUniquePeriodicWork(
            PERIODIC_WORK,
            ExistingPeriodicWorkPolicy.KEEP,
            PeriodicWorkRequestBuilder<GatewaySyncWorker>(INTERVAL_MINUTES, TimeUnit.MINUTES)
                .setConstraints(constraints)
                .setBackoffCriteria(BackoffPolicy.EXPONENTIAL, 5, TimeUnit.MINUTES)
                .build()
        )
        // One immediate run per app start so a fresh install / new token has data right away.
        wm.enqueueUniqueWork(
            STARTUP_WORK,
            ExistingWorkPolicy.KEEP,
            OneTimeWorkRequestBuilder<GatewaySyncWorker>()
                .setConstraints(constraints)
                .setBackoffCriteria(BackoffPolicy.EXPONENTIAL, 5, TimeUnit.MINUTES)
                .build()
        )
    }

    fun cancel(context: Context) {
        val wm = WorkManager.getInstance(context)
        wm.cancelUniqueWork(PERIODIC_WORK)
        wm.cancelUniqueWork(STARTUP_WORK)
    }
}

// Scope vocabulary from FINAL_PLAN. read + analysis are the default; checkmate:plan (Phase 3)
// unlocks the request tools, which only QUEUE a request for Checkmate to accept or reject.
// checkmate:behavior unlocks AGGREGATE behavior/adherence summaries only (never raw events) and is

export const ALL_SCOPES = [
  'checkmate:read',
  'checkmate:analysis',
  'checkmate:behavior',
  'checkmate:plan',
  'checkmate:write',
  'checkmate:admin'
] as const;
export type Scope = (typeof ALL_SCOPES)[number];

// What a token gets when no scopes are asked for: read-only, exactly as in v1.
export const DEFAULT_SCOPES: Scope[] = ['checkmate:read', 'checkmate:analysis'];
// What POST /api/gateway/tokens may mint when asked explicitly. checkmate:plan and
// checkmate:behavior are OFF by default and must be requested; write/admin can't be issued at all.
export const MINTABLE_SCOPES: Scope[] = ['checkmate:read', 'checkmate:analysis', 'checkmate:behavior', 'checkmate:plan'];
// Kept so older imports keep compiling.
export const V1_MINTABLE_SCOPES: Scope[] = DEFAULT_SCOPES;

export interface GatewayPrincipal {
  userId: string;
  tokenId: string;
  label: string;
  scopes: Scope[];
}

export interface ToolContext {
  principal: GatewayPrincipal;

}

export interface ToolDef {
  name: string;
  title: string;
  description: string;
  inputSchema: Record<string, unknown>;
  scope: Scope;
  /** Queues or changes a request (so it is not read-only). Drives the MCP annotations. */
  mutates?: boolean;
  /** Asks for something to be removed/dismissed; hosts may prompt before running it. */
  destructive?: boolean;
  handler: (args: Record<string, unknown>, ctx: ToolContext) => Promise<unknown>;
}

/** Bad arguments from the model — reported as a tool error it can read and fix, not a 500. */

// ─────────────────────────────────────────────────────────────────────────────
// Gateway read-projection (FINAL_PLAN "Option C — hybrid").
//
// The Android app stays the source of truth for the student model, interventions and
// today's plan. It PUSHES curated snapshots here (PUT /api/gateway/projection, device-token
// auth); the MCP gateway only ever READS these rows. The gateway never gets arbitrary SQL
// against anything, and keeps working while the phone is offline (with a visible
// freshness stamp so the model can say "as of 3h ago").
//
// Every pushed payload is re-normalized field by field below — unknown fields are dropped,
// strings bounded, arrays capped — so the phone can't (accidentally) smuggle internals,
// secrets, or megabytes into what an external model eventually reads.
// ─────────────────────────────────────────────────────────────────────────────

export const PROJECTION_KINDS = ['student_model', 'interventions', 'today_plan', 'knowledge_graph', 'behavior', 'task_history'] as const;
export type ProjectionKind = (typeof PROJECTION_KINDS)[number];
export const MAX_PROJECTION_BYTES = 800_000;
const SCHEMA_VERSION = 1;

export class ProjectionError extends Error {}

function str(v: unknown, max = 200): string | null {
  if (typeof v !== 'string') return null;
  const t = v.trim();
  return t ? t.slice(0, max) : null;
}
function num(v: unknown): number | null {
  return typeof v === 'number' && Number.isFinite(v) ? v : null;
}
function int(v: unknown): number {
  const n = num(v);
  return n === null ? 0 : Math.trunc(n);
}
function bool(v: unknown): boolean {
  return v === true;
}
function arr(v: unknown, max: number): any[] {
  return Array.isArray(v) ? v.slice(0, max) : [];
}
function epochToIso(v: unknown): string | null {
  const n = num(v);
  return n === null || n <= 0 ? null : new Date(n).toISOString();
}

function normalizeStudentModel(b: any) {
  const o = b?.overall ?? {};
  const ref = (p: any) => ({ subject: str(p?.subject), chapter: str(p?.chapter), topic: str(p?.topic) });
  return {
    exam: str(b?.exam, 60),
    model_version: int(b?.model_version) || 1,
    overall: {
      concepts_tracked: int(o.concepts_tracked),
      concepts_mastered: int(o.concepts_mastered),
      concepts_weak: int(o.concepts_weak),
      average_mastery: num(o.average_mastery),
      total_attempts: int(o.total_attempts),
      unresolved_error_count: int(o.unresolved_error_count)
    },
    concepts: arr(b?.concepts, 3000).map((c) => ({
      concept_id: str(c?.concept_id, 120),
      subject: str(c?.subject),
      chapter: str(c?.chapter),
      topic: str(c?.topic),
      mastery: num(c?.mastery),
      forgetting_risk: num(c?.forgetting_risk),
      retention_decision: ['REVIEW', 'TEACH', 'MOVE_ON'].includes(c?.retention_decision) ? c.retention_decision : null,
      attempt_count: int(c?.attempt_count),
      recent_accuracy: num(c?.recent_accuracy),
      lifetime_accuracy: num(c?.lifetime_accuracy),
      error_count: int(c?.error_count),
      last_seen_at: epochToIso(c?.last_seen_ms),
      weak: bool(c?.weak),
      prerequisite_issues: arr(c?.prerequisite_issues, 10).map(ref)
    })),
    unresolved_errors: arr(b?.unresolved_errors, 500).map((e) => ({
      concept_id: str(e?.concept_id, 120),
      subject: str(e?.subject),
      chapter: str(e?.chapter),
      topic: str(e?.topic),
      error_type: str(e?.error_type, 40),
      occurrences: int(e?.occurrences),
      first_seen_at: epochToIso(e?.first_seen_ms),
      last_seen_at: epochToIso(e?.last_seen_ms)
    }))
  };
}

function normalizeInterventions(b: any) {
  return {
    entries: arr(b?.entries, 300).map((e) => ({
      transaction_id: str(e?.transaction_id, 80),
      task_id: str(e?.task_id, 80),
      trigger_type: str(e?.trigger_type, 60),
      terminal_state: str(e?.terminal_state, 60),
      provenance: str(e?.provenance, 60),
      attempt_count: int(e?.attempt_count),
      resolved_at: epochToIso(e?.resolved_ms),
      outcome: str(e?.outcome, 300),
      failure_reason: str(e?.failure_reason, 300)
    }))
  };
}

function normalizeTodayPlan(b: any) {
  return {
    day_key: str(b?.day_key, 20),
    completion_percent: num(b?.completion_percent),
    tasks: arr(b?.tasks, 100).map((t) => ({
      id: str(t?.id, 80),
      subject: str(t?.subject, 80),
      topic: str(t?.topic, 200),
      duration_minutes: int(t?.duration_minutes),
      state: str(t?.state, 20),
      task_type: str(t?.task_type, 20),
      scheduled_start_time: str(t?.scheduled_start_time, 10),
      completed_status: str(t?.completed_status, 10),
      learning_intent: str(t?.learning_intent, 40),
      concept_id: str(t?.concept_id, 120),
      rationale: str(t?.rationale, 400),
      focus_minutes: int(t?.focus_minutes)
    }))
  };
}


// Static-ish prerequisite graph: nodes are concepts (same ids as the student model), edges say
// "prerequisite_id must be understood before concept_id". Dangling edges are dropped.
function normalizeKnowledgeGraph(b: any) {
  const nodes = arr(b?.nodes, 1500)
    .map((n) => ({ concept_id: str(n?.concept_id, 120), subject: str(n?.subject), chapter: str(n?.chapter), topic: str(n?.topic) }))
    .filter((n) => n.concept_id);
  const ids = new Set(nodes.map((n) => n.concept_id));
  const edges = arr(b?.edges, 4000)
    .map((e) => ({ concept_id: str(e?.concept_id, 120), prerequisite_id: str(e?.prerequisite_id, 120) }))
    .filter((e) => e.concept_id && e.prerequisite_id && e.concept_id !== e.prerequisite_id && ids.has(e.concept_id as string) && ids.has(e.prerequisite_id as string));
  return { nodes, edges };
}

// AGGREGATES ONLY (FINAL_PLAN: never raw behavior events). No app names, no timestamps of
// individual events, no free text.
function normalizeBehavior(b: any) {
  const a = b?.attention ?? {};
  const l = b?.ledger ?? {};
  return {
    streak_days: int(b?.streak_days),
    consecutive_missed_days: int(b?.consecutive_missed_days),
    week_completion_percent: int(b?.week_completion_percent),
    recent_skip_rate_percent: int(b?.recent_skip_rate_percent),
    total_skips_7d: int(b?.total_skips_7d),
    today_completed: int(b?.today_completed),
    attention: { checks_passed: int(a.checks_passed), checks_missed: int(a.checks_missed), avg_focus_minutes: int(a.avg_focus_minutes) },
    skip_patterns: arr(b?.skip_patterns, 20).map((s) => ({ subject: str(s?.subject, 80), task_type: str(s?.task_type, 20), occurrences: int(s?.occurrences) })),
    ledger: {
      events: int(l.events),
      done_7d: int(l.done_7d),
      skipped_7d: int(l.skipped_7d),
      done_30d: int(l.done_30d),
      skipped_30d: int(l.skipped_30d),
      focus_minutes_7d: int(l.focus_minutes_7d)
    }
  };
}

const DATE_RE = /^\d{4}-\d{2}-\d{2}$/;
function normalizeTaskHistory(b: any) {
  return {
    days: arr(b?.days, 30)
      .map((d) => ({
        date: typeof d?.date === 'string' && DATE_RE.test(d.date) ? d.date : null,
        planned: int(d?.planned),
        done: int(d?.done),
        skipped: int(d?.skipped),
        unfinished: int(d?.unfinished),
        focus_minutes: int(d?.focus_minutes),
        tasks: arr(d?.tasks, 80).map((t) => ({
          id: str(t?.id, 80),
          subject: str(t?.subject, 80),
          topic: str(t?.topic, 200),
          task_type: str(t?.task_type, 20),
          state: str(t?.state, 20),
          duration_minutes: int(t?.duration_minutes),
          focus_minutes: int(t?.focus_minutes),
          learning_intent: str(t?.learning_intent, 40),
          completed_status: str(t?.completed_status, 10)
        }))
      }))
      .filter((d) => d.date)
  };
}

export function isProjectionKind(v: unknown): v is ProjectionKind {
  return typeof v === 'string' && (PROJECTION_KINDS as readonly string[]).includes(v);
}

export async function putProjection(sb: SupabaseClient, userId: string, kind: ProjectionKind, body: any) {
  if (!body || typeof body !== 'object') throw new ProjectionError('Body must be a JSON object');
  const payload =
    kind === 'student_model'
      ? normalizeStudentModel(body)
      : kind === 'interventions'
        ? normalizeInterventions(body)
        : kind === 'knowledge_graph'
          ? normalizeKnowledgeGraph(body)
          : kind === 'behavior'
            ? normalizeBehavior(body)
            : kind === 'task_history'
              ? normalizeTaskHistory(body)
              : normalizeTodayPlan(body);

  // Phone clocks drift; refuse nonsense but tolerate a little.
  let generatedMs = typeof body.generated_ms === 'number' ? body.generated_ms : Date.now();
  if (!Number.isFinite(generatedMs) || generatedMs > Date.now() + 5 * 60_000) generatedMs = Date.now();

  // Never let an older snapshot overwrite a newer one (retried/out-of-order pushes).
  const { data: existing } = await sb
    .from('gateway_projection')
    .select('generated_at')
    .eq('user_id', userId)
    .eq('kind', kind)
    .maybeSingle();
  if (existing && new Date(existing.generated_at).getTime() > generatedMs) {
    return { stored: false, reason: 'A newer snapshot is already stored', generated_at: existing.generated_at as string };
  }

  const { error } = await sb.from('gateway_projection').upsert(
    {
      user_id: userId,
      kind,
      payload,
      schema_version: SCHEMA_VERSION,
      generated_at: new Date(generatedMs).toISOString(),
      received_at: new Date().toISOString()
    },
    { onConflict: 'user_id,kind' }
  );
  if (error) throw error;
  if (kind === 'student_model') await recordMasteryRollup(sb, userId, payload as any, generatedMs);
  return { stored: true, generated_at: new Date(generatedMs).toISOString() };
}

// One compact row per student-local day (the day's LAST student_model push wins), so
// checkmate_get_mastery_trends can show movement without keeping every snapshot.
// Best-effort: a failure here (e.g. migration 020 not applied yet) never fails the push.
async function recordMasteryRollup(sb: SupabaseClient, userId: string, payload: any, generatedMs: number): Promise<void> {
  try {
    const concepts: any[] = Array.isArray(payload?.concepts) ? payload.concepts : [];
    if (concepts.length === 0) return;
    const { model } = fillSubjects({ concepts }, await getSubjectByChapter(sb));
    const bySubject = new Map<string, { n: number; sum: number; weak: number }>();
    for (const c of model.concepts as any[]) {
      const k = c.subject ?? 'Unclassified';
      const s = bySubject.get(k) ?? { n: 0, sum: 0, weak: 0 };
      s.n += 1;
      s.sum += typeof c.mastery === 'number' ? c.mastery : 0;
      if (c.weak) s.weak += 1;
      bySubject.set(k, s);
    }
    const { error } = await sb.from('gateway_mastery_history').upsert(
      {
        user_id: userId,
        day: localDateString(new Date(generatedMs)),
        generated_at: new Date(generatedMs).toISOString(),
        concepts_tracked: payload.overall?.concepts_tracked ?? concepts.length,
        concepts_weak: payload.overall?.concepts_weak ?? concepts.filter((c) => c.weak).length,
        average_mastery: payload.overall?.average_mastery ?? null,
        by_subject: Array.from(bySubject.entries()).map(([subject, s]) => ({
          subject,
          concepts: s.n,
          average_mastery: s.n ? Math.round((s.sum / s.n) * 1000) / 1000 : null,
          weak_concepts: s.weak
        }))
      },
      { onConflict: 'user_id,day' }
    );
    if (error) console.error('[gateway/projection] mastery rollup failed:', error.message);
  } catch (err: any) {
    console.error('[gateway/projection] mastery rollup failed:', err?.message ?? err);
  }
}

export interface StoredProjection<T = any> {
  payload: T;
  generated_at: string;
  received_at: string;
}

export async function getProjection<T = any>(
  sb: SupabaseClient,
  userId: string,
  kind: ProjectionKind
): Promise<StoredProjection<T> | null> {
  const { data, error } = await sb
    .from('gateway_projection')
    .select('payload, generated_at, received_at')
    .eq('user_id', userId)
    .eq('kind', kind)
    .maybeSingle();
  if (error || !data) return null;
  return data as StoredProjection<T>;
}

export function freshness(generatedAt: string, staleAfterMinutes = 360) {
  const ageMin = Math.max(0, Math.round((Date.now() - new Date(generatedAt).getTime()) / 60_000));
  return { generated_at: generatedAt, age_minutes: ageMin, stale: ageMin > staleAfterMinutes };
}

export async function projectionStatus(sb: SupabaseClient, userId: string) {
  const { data } = await sb
    .from('gateway_projection')
    .select('kind, generated_at, received_at, schema_version')
    .eq('user_id', userId);
  return (data ?? []).map((r: any) => ({
    kind: r.kind,
    generated_at: r.generated_at,
    received_at: r.received_at,
    schema_version: r.schema_version
  }));
}

function optStr(v: unknown): string | undefined {
  return typeof v === 'string' && v.trim() ? v.trim() : undefined;
}
const eq = (a?: string | null, b?: string | null) => !!a && !!b && a.trim().toLowerCase() === b.trim().toLowerCase();

function requireModel(m: any) {
  if (!m) {
    throw new ToolDataUnavailable(
      "Checkmate hasn't synced its student model to the gateway yet. Open the Checkmate app (it syncs in the background roughly every 30 minutes), then retry. Question-level tools still work."
    );
  }
  return m;
}

// ── Concept detail ───────────────────────────────────────────────────────────
export function getConcept(model: any, args: Record<string, unknown>) {
  const m = requireModel(model);
  const id = optStr(args.concept_id);
  const subject = optStr(args.subject);
  const chapter = optStr(args.chapter);
  const topic = optStr(args.topic);
  const limit = clampInt(args.limit, 5, 1, 20);
  if (!id && !chapter && !topic) {
    throw new ToolInputError('Pass concept_id, or a chapter (optionally with topic).');
  }

  let concepts = (m.concepts as any[]).filter((c) => {
    if (id) return c.concept_id === id;
    return (!subject || eq(c.subject, subject)) && (!chapter || chapterMatches(c.chapter, chapter)) && (!topic || eq(c.topic, topic));
  });
  if (concepts.length === 0) {
    throw new ToolInputError(id ? 'No concept with that id. Take concept_id from checkmate_get_mastery or checkmate_get_weak_concepts.' : 'No concepts match those filters.');
  }
  concepts = concepts.sort((a, b) => (a.mastery ?? 1) - (b.mastery ?? 1));
  const errors = m.unresolved_errors as any[];
  return {
    source: 'checkmate_student_model',
    freshness: m._freshness,
    matching: concepts.length,
    concepts: concepts.slice(0, limit).map((c) => ({
      ...c,
      unresolved_error_patterns: errors
        .filter((e) => (c.concept_id && e.concept_id === c.concept_id) || (!e.concept_id && chapterMatches(e.chapter, c.chapter) && (!c.topic || !e.topic || eq(e.topic, c.topic))))
        .sort((a, b) => b.occurrences - a.occurrences)
        .slice(0, 8)
        .map((e) => ({ error_type: e.error_type, occurrences: e.occurrences, first_seen_at: e.first_seen_at, last_seen_at: e.last_seen_at }))
    })),
    note: 'prerequisite_issues lists weak prerequisite concepts (Checkmate only syncs the weak ones, not the full dependency graph).'
  };
}

// ── Interventions that did not work ──────────────────────────────────────────
export async function getFailedInterventions(sb: SupabaseClient, userId: string, args: Record<string, unknown>) {
  const days = clampInt(args.days, 30, 1, 365);
  const limit = clampInt(args.limit, 25, 1, 100);
  const all: any = await getInterventionHistory(sb, userId, { days, limit: 100 });
  const failed = (all.interventions as any[]).filter((e) => String(e.provenance ?? '').toUpperCase() !== 'SUCCESS');
  const byTrigger: Record<string, number> = {};
  for (const e of failed) {
    const k = e.trigger_type ?? 'UNKNOWN';
    byTrigger[k] = (byTrigger[k] ?? 0) + 1;
  }
  return {
    source: all.source,
    freshness: all.freshness,
    window_days: days,
    note: 'Everything whose provenance is not SUCCESS: the student did not act (USER_ABORTED / TTL_EXPIRED), Checkmate refused or failed (POLICY_REJECTED / EXECUTION_FAILED), or its deterministic fallback ran (NETWORK_FALLBACK / LLM_INVALID).',
    total_failed: failed.length,
    total_in_window: all.total,
    by_trigger: byTrigger,
    by_provenance: failed.reduce((t: Record<string, number>, e) => ((t[e.provenance ?? 'UNKNOWN'] = (t[e.provenance ?? 'UNKNOWN'] ?? 0) + 1), t), {}),
    interventions: failed.slice(0, limit)
  };
}

// ── Learning timeline ────────────────────────────────────────────────────────
export async function getLearningTimeline(sb: SupabaseClient, userId: string, args: Record<string, unknown>) {
  const days = clampInt(args.days, 7, 1, 90);
  const limit = clampInt(args.limit, 40, 1, 100);
  const subject = optStr(args.subject);
  const chapter = optStr(args.chapter);
  const which = optStr(args.events) ?? 'all';
  if (!['all', 'attempts', 'interventions'].includes(which)) {
    throw new ToolInputError("events must be one of: all, attempts, interventions");
  }
  const source = (optStr(args.source) ?? 'all') as SourceFilter;
  const filtered = !!(subject || chapter);
  const events: any[] = [];

  if (which !== 'interventions') {
    const [attempts, subjectByChapter] = await Promise.all([
      loadAttempts(sb, userId, { sinceIso: new Date(Date.now() - days * 86_400_000).toISOString(), maxRows: 2500 }),
      getSubjectByChapter(sb)
    ]);
    for (const a of attempts as Attempt[]) {
      if (!a.answered_at || a.status === 'UNSCORED') continue;
      if (source !== 'all' && a.provenance !== source) continue;
      const ch = a.question.chapter ?? null;
      const subj = ch ? subjectByChapter.get(ch) ?? null : null;
      if (chapter && !chapterMatches(ch, chapter)) continue;
      if (subject && !eq(subj, subject)) continue;
      events.push({
        timestamp: a.answered_at,
        type: `QUESTION_${a.status}`,
        subject: subj,
        chapter: ch,
        topic: a.question.topic ?? null,
        source: a.provenance,
        question_id: a.question.id,
        canonical_question_id: a.canonical_id,
        session_id: a.session_id,
        test_title: a.test_title,
        time_spent_seconds: a.time_spent_seconds
      });
    }
  }

  let interventionsNote: string | undefined;
  if (which !== 'attempts') {
    if (filtered) {
      interventionsNote = 'Interventions are omitted because they carry no chapter/subject; remove the filter to include them.';
    } else {
      try {
        const h: any = await getInterventionHistory(sb, userId, { days, limit: 100 });
        for (const e of h.interventions as any[]) {
          events.push({
            timestamp: e.resolved_at,
            type: `INTERVENTION_${String(e.provenance ?? 'UNKNOWN').toUpperCase()}`,
            source: 'checkmate_outcome_ledger',
            trigger_type: e.trigger_type,
            terminal_state: e.terminal_state,
            outcome: e.outcome,
            failure_reason: e.failure_reason
          });
        }
      } catch (err: any) {
        interventionsNote = err?.message ?? 'Intervention history unavailable.';
      }
    }
  }

  events.sort((x, y) => String(y.timestamp).localeCompare(String(x.timestamp)));
  const tally: Record<string, number> = {};
  for (const e of events) tally[e.type] = (tally[e.type] ?? 0) + 1;
  return {
    window_days: days,
    total_events: events.length,
    by_type: tally,
    ...(interventionsNote ? { note: interventionsNote } : {}),
    events: events.slice(0, limit)
  };
}


const round = (n: number, d = 3) => Math.round(n * 10 ** d) / 10 ** d;
const DAY_MS = 86_400_000;

// ── Knowledge graph ──────────────────────────────────────────────────────────
interface GraphNode {
  concept_id: string;
  subject: string | null;
  chapter: string | null;
  topic: string | null;
}
interface Graph {
  nodes: Map<string, GraphNode>;
  prereqs: Map<string, string[]>;
  dependents: Map<string, string[]>;
  edges: { concept_id: string; prerequisite_id: string }[];
  generated_at: string;
}

async function loadGraph(sb: SupabaseClient, userId: string): Promise<Graph> {
  const p = await getProjection(sb, userId, 'knowledge_graph');
  if (!p) {
    throw new ToolDataUnavailable(
      "Checkmate hasn't synced its prerequisite graph yet. Open the Checkmate app (the graph is re-sent at most once a day), then retry."
    );
  }
  const subjects = await getSubjectByChapter(sb);
  const nodeRows = fillSubjects({ concepts: p.payload.nodes as GraphNode[] }, subjects).model.concepts as GraphNode[];
  const nodes = new Map(nodeRows.map((n) => [n.concept_id, n]));
  const prereqs = new Map<string, string[]>();
  const dependents = new Map<string, string[]>();
  for (const e of p.payload.edges as { concept_id: string; prerequisite_id: string }[]) {
    (prereqs.get(e.concept_id) ?? prereqs.set(e.concept_id, []).get(e.concept_id)!).push(e.prerequisite_id);
    (dependents.get(e.prerequisite_id) ?? dependents.set(e.prerequisite_id, []).get(e.prerequisite_id)!).push(e.concept_id);
  }
  return { nodes, prereqs, dependents, edges: p.payload.edges, generated_at: p.generated_at };
}

function masteryIndex(model: any | null): Map<string, any> {
  const m = new Map<string, any>();
  for (const c of (model?.concepts as any[]) ?? []) m.set(c.concept_id, c);
  return m;
}

// attempted=false means the student has no mastery row for it: unknown, NOT weak.
function nodeView(n: GraphNode, mi: Map<string, any>) {
  const c = mi.get(n.concept_id);
  return {
    concept_id: n.concept_id,
    subject: n.subject,
    chapter: n.chapter,
    topic: n.topic,
    attempted: !!c,
    mastery: c?.mastery ?? null,
    weak: c ? !!c.weak : null,
    retention_decision: c?.retention_decision ?? null,
    forgetting_risk: c?.forgetting_risk ?? null
  };
}

function findStart(g: Graph, args: Record<string, unknown>): string[] {
  const id = optStr(args.concept_id);
  const subject = optStr(args.subject);
  const chapter = optStr(args.chapter);
  const topic = optStr(args.topic);
  if (!id && !chapter && !subject && !topic) {
    throw new ToolInputError('Pass concept_id, or a chapter / subject (optionally with topic).');
  }
  if (id) {
    if (!g.nodes.has(id)) {
      throw new ToolInputError('That concept is not in the prerequisite graph (it has no prerequisite edges). Take concept_id from checkmate_get_mastery or checkmate_get_concept_graph.');
    }
    return [id];
  }
  const ids = Array.from(g.nodes.values())
    .filter((n) => (!chapter || chapterMatches(n.chapter, chapter)) && (!subject || eq(n.subject, subject)) && (!topic || eq(n.topic, topic)))
    .map((n) => n.concept_id);
  if (ids.length === 0) throw new ToolInputError('No concepts in the prerequisite graph match those filters.');
  return ids;
}

/** BFS up (prerequisites) or down (dependents); returns node id -> hop distance. */
function walk(g: Graph, startId: string, dir: 'up' | 'down', depth: number): Map<string, number> {
  const adj = dir === 'up' ? g.prereqs : g.dependents;
  const seen = new Map<string, number>();
  let frontier = [startId];
  for (let d = 1; d <= depth && frontier.length; d += 1) {
    const next: string[] = [];
    for (const id of frontier) {
      for (const n of adj.get(id) ?? []) {
        if (n === startId || seen.has(n)) continue;
        seen.set(n, d);
        next.push(n);
      }
    }
    frontier = next;
  }
  return seen;
}

const GRAPH_NOTE =
  'Edges mean "prerequisite must be understood before the concept". The graph is Checkmate\'s seeded syllabus order (chapter/topic level), not a full pedagogical model; attempted=false means no data yet, not weakness.';

export async function getGraphNeighbors(sb: SupabaseClient, userId: string, model: any | null, args: Record<string, unknown>, dir: 'up' | 'down') {
  const g = await loadGraph(sb, userId);
  const mi = masteryIndex(model);
  const depth = clampInt(args.depth, 1, 1, 4);
  const limit = clampInt(args.limit, 5, 1, 10);
  const starts = findStart(g, args);
  const results = starts.slice(0, limit).map((id) => {
    const related = Array.from(walk(g, id, dir, depth).entries())
      .map(([nid, d]) => ({ depth: d, ...nodeView(g.nodes.get(nid)!, mi) }))
      .sort((a, b) => a.depth - b.depth || (a.mastery ?? 2) - (b.mastery ?? 2));
    const concept = nodeView(g.nodes.get(id)!, mi);
    return dir === 'up'
      ? {
          concept,
          prerequisites: related.slice(0, 60),
          weak_prerequisites: related.filter((r) => r.weak === true).slice(0, 10),
          prerequisites_without_data: related.filter((r) => !r.attempted).length
        }
      : { concept, dependents: related.slice(0, 60), weak_dependents: related.filter((r) => r.weak === true).slice(0, 10) };
  });
  return {
    source: 'checkmate_knowledge_graph',
    freshness: freshness(g.generated_at, 60 * 24 * 3),
    depth,
    matching: starts.length,
    ...(starts.length > limit ? { note: `${starts.length} concepts matched; showing the first ${limit}. Narrow with topic or concept_id.` } : { note: GRAPH_NOTE }),
    results
  };
}

export async function getConceptGraph(sb: SupabaseClient, userId: string, model: any | null, args: Record<string, unknown>) {
  const g = await loadGraph(sb, userId);
  const mi = masteryIndex(model);
  const fresh = freshness(g.generated_at, 60 * 24 * 3);
  const filtered = !!(optStr(args.concept_id) || optStr(args.chapter) || optStr(args.subject) || optStr(args.topic));

  if (!filtered) {
    const bySubject: Record<string, number> = {};
    for (const n of Array.from(g.nodes.values())) bySubject[n.subject ?? 'Unclassified'] = (bySubject[n.subject ?? 'Unclassified'] ?? 0) + 1;
    const dependentCount = (id: string) => g.dependents.get(id)?.length ?? 0;
    const all = Array.from(g.nodes.values());
    return {
      source: 'checkmate_knowledge_graph',
      freshness: fresh,
      note: GRAPH_NOTE + ' Pass chapter, subject, topic or concept_id (and depth) to get the actual nodes and edges.',
      concepts_in_graph: g.nodes.size,
      edges: g.edges.length,
      concepts_by_subject: bySubject,
      foundation_concepts: all.filter((n) => (g.prereqs.get(n.concept_id)?.length ?? 0) === 0).length,
      most_depended_on: all
        .filter((n) => dependentCount(n.concept_id) > 0)
        .sort((a, b) => dependentCount(b.concept_id) - dependentCount(a.concept_id))
        .slice(0, 10)
        .map((n) => ({ ...nodeView(n, mi), dependents: dependentCount(n.concept_id) })),
      weak_foundations: all
        .filter((n) => mi.get(n.concept_id)?.weak === true && dependentCount(n.concept_id) > 0)
        .sort((a, b) => dependentCount(b.concept_id) - dependentCount(a.concept_id))
        .slice(0, 10)
        .map((n) => ({ ...nodeView(n, mi), dependents: dependentCount(n.concept_id) }))
    };
  }

  const depth = clampInt(args.depth, 1, 1, 3);
  const MAX_NODES = 120;
  const starts = findStart(g, args);
  const inScope = new Set(starts);
  const dist = new Map<string, number>(starts.map((s) => [s, 0]));
  for (const s of starts) {
    for (const dir of ['up', 'down'] as const) {
      for (const [nid, d] of Array.from(walk(g, s, dir, depth).entries())) if (!dist.has(nid) || (dist.get(nid) as number) > d) dist.set(nid, d);
    }
  }
  const ordered = Array.from(dist.entries()).sort((a, b) => a[1] - b[1]);
  const keep = new Set(ordered.slice(0, MAX_NODES).map(([id]) => id));
  const edges = g.edges.filter((e) => keep.has(e.concept_id) && keep.has(e.prerequisite_id));
  const label = (id: string) => `${g.nodes.get(id)?.chapter ?? '?'} / ${g.nodes.get(id)?.topic ?? '?'}`;
  return {
    source: 'checkmate_knowledge_graph',
    freshness: fresh,
    note: GRAPH_NOTE,
    depth,
    focus_concepts: starts.length,
    ...(ordered.length > MAX_NODES ? { truncated: true, omitted_nodes: ordered.length - MAX_NODES } : {}),
    nodes: Array.from(keep).map((id) => ({ ...nodeView(g.nodes.get(id)!, mi), in_focus: inScope.has(id) })),
    edges: edges.map((e) => ({ prerequisite_id: e.prerequisite_id, concept_id: e.concept_id, reads: `${label(e.prerequisite_id)} -> ${label(e.concept_id)}` }))
  };
}

// ── Task history (multi-day, from the phone's saved daily plans) ─────────────
async function loadTaskHistory(sb: SupabaseClient, userId: string) {
  const p = await getProjection(sb, userId, 'task_history');
  if (!p) {
    throw new ToolDataUnavailable("Checkmate hasn't synced its task history yet. Open the Checkmate app (it syncs in the background roughly every 30 minutes), then retry.");
  }
  const days = (p.payload.days as any[]).slice().sort((a, b) => String(b.date).localeCompare(String(a.date)));
  return { days, fresh: freshness(p.generated_at, 180), generated_at: p.generated_at, today: localDateString(new Date(p.generated_at)) };
}

function cutoffDate(today: string, days: number): string {
  return new Date(Date.parse(`${today}T00:00:00Z`) - (days - 1) * DAY_MS).toISOString().slice(0, 10);
}

export async function getTaskHistory(sb: SupabaseClient, userId: string, args: Record<string, unknown>) {
  const h = await loadTaskHistory(sb, userId);
  const days = clampInt(args.days, 7, 1, 30);
  const subject = optStr(args.subject);
  const state = optStr(args.state)?.toUpperCase();
  const includeTasks = args.include_tasks !== false;
  const limit = clampInt(args.limit, 60, 1, 150);
  const since = cutoffDate(h.today, days);
  const rows = h.days.filter((d) => d.date >= since);

  const matches = (t: any) => (!subject || eq(t.subject, subject)) && (!state || String(t.state).toUpperCase() === state);
  const tasks: any[] = [];
  for (const d of rows) for (const t of d.tasks as any[]) if (matches(t)) tasks.push({ date: d.date, ...t });
  const sum = (k: string) => rows.reduce((s, d) => s + (d[k] ?? 0), 0);
  return {
    source: 'checkmate_planner',
    freshness: h.fresh,
    window_days: days,
    note: 'Built from the phone\'s saved daily plans (up to 30 days). A day with no tasks counts as a day with no plan.',
    totals: { planned: sum('planned'), done: sum('done'), skipped: sum('skipped'), unfinished: sum('unfinished'), focus_minutes: sum('focus_minutes') },
    by_day: rows.map(({ tasks: _t, ...d }) => d),
    ...(includeTasks ? { matching_tasks: tasks.length, tasks: tasks.slice(0, limit) } : {})
  };
}

export async function getCompletedTasks(sb: SupabaseClient, userId: string, args: Record<string, unknown>) {
  const h = await loadTaskHistory(sb, userId);
  const days = clampInt(args.days, 7, 1, 30);
  const subject = optStr(args.subject);
  const limit = clampInt(args.limit, 60, 1, 150);
  const since = cutoffDate(h.today, days);
  const done: any[] = [];
  for (const d of h.days.filter((x) => x.date >= since)) {
    for (const t of d.tasks as any[]) if (t.state === 'DONE' && (!subject || eq(t.subject, subject))) done.push({ date: d.date, ...t });
  }
  const bySubject: Record<string, number> = {};
  for (const t of done) bySubject[t.subject ?? 'Unknown'] = (bySubject[t.subject ?? 'Unknown'] ?? 0) + 1;
  return {
    source: 'checkmate_planner',
    freshness: h.fresh,
    window_days: days,
    total_completed: done.length,
    focus_minutes: done.reduce((s, t) => s + (t.focus_minutes ?? 0), 0),
    by_subject: bySubject,
    tasks: done.slice(0, limit)
  };
}

function computeAdherence(h: { days: any[]; today: string }, days: number) {
  const since = cutoffDate(h.today, days);
  const rows = h.days.filter((d) => d.date >= since).sort((a, b) => String(a.date).localeCompare(String(b.date)));
  const planned = rows.reduce((s, d) => s + d.planned, 0);
  const completed = rows.reduce((s, d) => s + d.done, 0);
  const skipped = rows.reduce((s, d) => s + d.skipped, 0);
  const abandoned = rows.filter((d) => d.date < h.today).reduce((s, d) => s + d.unfinished, 0);
  const openToday = rows.filter((d) => d.date === h.today).reduce((s, d) => s + d.unfinished, 0);
  const tally = (key: 'subject' | 'task_type') => {
    const m = new Map<string, { planned: number; done: number; skipped: number }>();
    for (const d of rows) {
      for (const t of d.tasks as any[]) {
        const k = t[key] ?? 'Unknown';
        const s = m.get(k) ?? { planned: 0, done: 0, skipped: 0 };
        s.planned += 1;
        if (t.state === 'DONE') s.done += 1;
        if (t.state === 'SKIPPED') s.skipped += 1;
        m.set(k, s);
      }
    }
    return Array.from(m.entries())
      .map(([name, s]) => ({ [key]: name, ...s, adherence: s.planned ? round(s.done / s.planned) : null }))
      .sort((a, b) => b.planned - a.planned);
  };
  return {
    window_days: days,
    planned,
    completed,
    skipped,
    abandoned,
    open_today: openToday,
    adherence: planned ? round(completed / planned) : null,
    days_with_a_plan: rows.filter((d) => d.planned > 0).length,
    days_without_a_plan: rows.filter((d) => d.planned === 0).length,
    by_day: rows.map((d) => ({ date: d.date, planned: d.planned, done: d.done, skipped: d.skipped, unfinished: d.unfinished })),
    by_subject: tally('subject'),
    by_task_type: tally('task_type'),
    definitions: 'adherence = completed / planned. abandoned = tasks left unfinished (never done or skipped) on a day that has ended; open_today = still-open tasks today.'
  };
}

export async function getAttentionSummary(sb: SupabaseClient, userId: string, args: Record<string, unknown>) {
  const { b, fresh } = await loadBehavior(sb, userId);
  const a = b.attention as { checks_passed: number; checks_missed: number; avg_focus_minutes: number };
  const total = a.checks_passed + a.checks_missed;
  let focus: unknown;
  try {
    const h = await loadTaskHistory(sb, userId);
    const days = clampInt(args.days, 7, 1, 30);
    const rows = h.days.filter((d) => d.date >= cutoffDate(h.today, days));
    const done = rows.reduce((s, d) => s + d.done, 0);
    const minutes = rows.reduce((s, d) => s + d.focus_minutes, 0);
    focus = {
      window_days: days,
      focus_minutes: minutes,
      completed_tasks: done,
      avg_focus_minutes_per_completed_task: done ? round(minutes / done, 1) : null,
      by_day: rows.map((d) => ({ date: d.date, focus_minutes: d.focus_minutes, done: d.done })).sort((x, y) => x.date.localeCompare(y.date))
    };
  } catch (err: any) {
    focus = { unavailable: err?.message ?? 'failed' };
  }
  return {
    source: 'checkmate_behavior_ledger',
    freshness: fresh,
    note: 'Attention checks are the random "are you still studying" prompts during focus sessions; focus minutes come from the daily plans.',
    attention_checks: { passed: a.checks_passed, missed: a.checks_missed, pass_rate: total ? round(a.checks_passed / total) : null },
    avg_focus_minutes_in_checked_sessions: a.avg_focus_minutes,
    focus
  };
}

// ── Mastery trends (daily rollups kept by the gateway) ───────────────────────
export async function getMasteryTrends(sb: SupabaseClient, userId: string, args: Record<string, unknown>) {
  const days = clampInt(args.days, 30, 1, 180);
  const subject = optStr(args.subject);
  const sinceDay = localDateString(new Date(Date.now() - days * DAY_MS));
  const { data, error } = await sb
    .from('gateway_mastery_history')
    .select('day, generated_at, concepts_tracked, concepts_weak, average_mastery, by_subject')
    .eq('user_id', userId)
    .gte('day', sinceDay)
    .order('day', { ascending: true })
    .limit(400);
  if (error) {
    throw new ToolDataUnavailable('Mastery history is not available yet (run migration 020_gateway_history.sql in Supabase, then let Checkmate sync).');
  }
  const points = (data ?? []).slice().sort((x: any, y: any) => String(x.day).localeCompare(String(y.day))).map((r: any) => {
    if (!subject) return { day: r.day, average_mastery: r.average_mastery, concepts_tracked: r.concepts_tracked, weak_concepts: r.concepts_weak };
    const s = ((r.by_subject as any[]) ?? []).find((x) => eq(x.subject, subject));
    return { day: r.day, average_mastery: s?.average_mastery ?? null, concepts_tracked: s?.concepts ?? 0, weak_concepts: s?.weak_concepts ?? 0 };
  }).filter((p: any) => p.average_mastery !== null);

  if (points.length < 2) {
    return {
      source: 'gateway_mastery_history',
      window_days: days,
      ...(subject ? { subject } : {}),
      points,
      note: 'Not enough history to show a trend yet. The gateway keeps one rollup per day, starting from the first Checkmate sync after history was enabled, so trends fill in over the coming days.'
    };
  }
  const first = points[0];
  const last = points[points.length - 1];
  const delta = round(last.average_mastery - first.average_mastery);
  return {
    source: 'gateway_mastery_history',
    window_days: days,
    ...(subject ? { subject } : {}),
    note: 'Mastery averages every tracked concept equally, so newly attempted concepts can move it without real change in older ones; compare weak_concepts too.',
    from: first.day,
    to: last.day,
    change: { average_mastery: delta, weak_concepts: last.weak_concepts - first.weak_concepts, concepts_tracked: last.concepts_tracked - first.concepts_tracked },
    direction: delta > 0.02 ? 'improving' : delta < -0.02 ? 'declining' : 'flat',
    points
  };
}

  'Question-level tools return the real questions, the student\'s answers, correct answers and explanations — use them to ground any diagnosis ' +
  'instead of guessing from counts. Data may be stale if the phone has not synced (see freshness fields). ' +
  'This gateway cannot change learning data. A token with the checkmate:plan scope additionally gets request tools ' +
  '(checkmate_request_repair, checkmate_create_task) that only QUEUE a request: the student\'s Checkmate app ' +
  'decides whether to accept it. Only make a request when the student has asked for it in this conversation. ' +
  'A token with the checkmate:behavior scope also gets aggregate behavior, attention and adherence summaries (never individual events).';

type JsonRpcId = string | number | null;
interface JsonRpcRequest {
  jsonrpc: '2.0';
  id?: JsonRpcId;
  method: string;
  params?: Record<string, any>;
}

const ERR = { parse: -32700, invalidRequest: -32600, methodNotFound: -32601, invalidParams: -32602, internal: -32603 };
// -32020..-32099 are reserved for the MCP spec (2026-07-28).
const ERR_MODERN = { headerMismatch: -32020, unsupportedVersion: -32022 };

function rpcError(id: JsonRpcId | undefined, code: number, message: string) {
  return { jsonrpc: '2.0', id: id ?? null, error: { code, message } };
}
function rpcResult(id: JsonRpcId | undefined, result: unknown) {
  return { jsonrpc: '2.0', id: id ?? null, result };
}

function originAllowed(req: NextRequest): boolean {
  const origin = req.headers.get('origin');
  if (!origin) return true; // server-to-server hosts send none
  const allowed = (process.env.GATEWAY_ALLOWED_ORIGINS ?? '')
    .split(',')
    .map((s) => s.trim())
    .filter(Boolean);
  // Default: refuse browser origins entirely (DNS-rebinding defense). Opt specific
  // origins in with GATEWAY_ALLOWED_ORIGINS if a browser-based MCP client is ever needed.
  return allowed.includes(origin);
}

const tools = buildTools();
const toolByName = new Map(tools.map((t) => [t.name, t]));
const resources = buildResources();
const prompts = buildPrompts();

function toolResult(payload: unknown, isError = false) {
  const { text } = capResponse(isError ? payload : minimize(payload));
  return { content: [{ type: 'text', text }], isError };
}

async function handleOne(msg: JsonRpcRequest, principal: GatewayPrincipal): Promise<unknown | null> {
  const isNotification = msg.id === undefined;
  const started = Date.now();
  const supabase = getServiceClient();
  const ctx = { principal, supabase };

  const audit = (extra: { tool?: string; args?: unknown; ok: boolean; error?: string; bytes?: number }) =>
    recordAudit({ principal, method: msg.method, tool: extra.tool, args: extra.args, ok: extra.ok, error: extra.error, durationMs: Date.now() - started, responseBytes: extra.bytes });

  try {
    switch (msg.method) {
      case 'initialize': {
        const requested = msg.params?.protocolVersion;
        const version = SUPPORTED_VERSIONS.includes(requested) ? requested : SUPPORTED_VERSIONS[0];
        void audit({ ok: true });
        return rpcResult(msg.id, {
          protocolVersion: version,
          capabilities: { tools: { listChanged: false }, resources: { listChanged: false, subscribe: false }, prompts: { listChanged: false } },
          serverInfo: SERVER_INFO,
          instructions: INSTRUCTIONS
        });
      }
      case 'notifications/initialized':
      case 'notifications/cancelled':
        return null;
      case 'ping':
        return rpcResult(msg.id, {});

      case 'tools/list': {
        const visible = tools.filter((t) => hasScope(principal, t.scope));
        return rpcResult(msg.id, {
          tools: visible.map((t) => ({
            name: t.name,
            title: t.title,
            description: t.description,
            inputSchema: t.inputSchema,
            // Request tools must NOT claim to be read-only, or a host may auto-approve them.
            annotations: { readOnlyHint: !t.mutates, destructiveHint: !!t.destructive, idempotentHint: !t.mutates, openWorldHint: false }
          }))
        });
      }

      case 'tools/call': {
        const name = msg.params?.name;
        const args = (msg.params?.arguments ?? {}) as Record<string, unknown>;
        const tool = typeof name === 'string' ? toolByName.get(name) : undefined;
        if (!tool) return rpcError(msg.id, ERR.invalidParams, `Unknown tool: ${String(name)}`);
        if (typeof args !== 'object' || Array.isArray(args)) return rpcError(msg.id, ERR.invalidParams, 'arguments must be an object');
        if (!hasScope(principal, tool.scope)) {
          void audit({ tool: tool.name, args, ok: false, error: `missing scope ${tool.scope}` });
          return rpcResult(msg.id, toolResult({ error: `This token does not have the ${tool.scope} scope.` }, true));
        }
        try {
          const result = await tool.handler(args, ctx);
          const out = toolResult(result);
          void audit({ tool: tool.name, args, ok: true, bytes: Buffer.byteLength(out.content[0].text, 'utf8') });
          return rpcResult(msg.id, out);
        } catch (err: any) {
          // Tool-level failures are returned as readable tool errors so the model can
          // self-correct; only genuine bugs get a generic message (details stay in logs).
          const known = err instanceof ToolInputError || err instanceof ToolDataUnavailable;
          if (!known) console.error(`[gateway] ${tool.name} failed:`, err?.message ?? err);
          const message = known ? err.message : 'Internal error while reading learning data. Try narrowing the filters or retry shortly.';
          void audit({ tool: tool.name, args, ok: false, error: known ? message : `internal: ${String(err?.message ?? err)}` });
          return rpcResult(msg.id, toolResult({ error: message }, true));
        }
      }

      case 'resources/list': {
        const visible = resources.filter((r) => hasScope(principal, r.scope));
        return rpcResult(msg.id, {
          resources: visible.map((r) => ({ uri: r.uri, name: r.name, title: r.title, description: r.description, mimeType: 'application/json' }))
        });
      }
      case 'resources/read': {
        const uri = msg.params?.uri;
        const res = resources.find((r) => r.uri === uri);
        if (!res || !hasScope(principal, res.scope)) return rpcError(msg.id, ERR.invalidParams, `Unknown resource: ${String(uri)}`);
        try {
          const { text } = capResponse(minimize(await res.read(ctx)));
          void audit({ tool: `resource:${res.uri}`, ok: true, bytes: Buffer.byteLength(text, 'utf8') });
          return rpcResult(msg.id, { contents: [{ uri: res.uri, mimeType: 'application/json', text }] });
        } catch (err: any) {
          const known = err instanceof ToolInputError || err instanceof ToolDataUnavailable;
          void audit({ tool: `resource:${res.uri}`, ok: false, error: known ? err.message : `internal: ${String(err?.message ?? err)}` });
          return rpcError(msg.id, ERR.internal, known ? err.message : 'Failed to read resource');
        }
      }

      case 'prompts/list':
        return rpcResult(msg.id, { prompts: prompts.map((p) => ({ name: p.name, title: p.title, description: p.description, arguments: p.arguments })) });
      case 'prompts/get': {
        const p = prompts.find((x) => x.name === msg.params?.name);
        if (!p) return rpcError(msg.id, ERR.invalidParams, `Unknown prompt: ${String(msg.params?.name)}`);
        const args: Record<string, string> = {};
        for (const [k, v] of Object.entries((msg.params?.arguments ?? {}) as Record<string, unknown>)) if (typeof v === 'string') args[k] = v.slice(0, 200);
        return rpcResult(msg.id, { description: p.description, messages: [{ role: 'user', content: { type: 'text', text: p.render(args) } }] });
      }

      default:
        void audit({ ok: false, error: 'method not found' });
        return isNotification ? null : rpcError(msg.id, ERR.methodNotFound, `Method not found: ${msg.method}`);
    }
  } catch (err: any) {
    console.error('[gateway] unhandled:', err?.message ?? err);
    return isNotification ? null : rpcError(msg.id, ERR.internal, 'Internal error');
  }
}

// ─────────────────────────────────────────────────────────────────────────────
// MCP 2026-07-28 (stateless) support. Served alongside the legacy handshake path on the
// same endpoint: a request is "modern" when it carries the 2026-07-28 per-request
// `_meta` version or the matching MCP-Protocol-Version header. Everything else goes
// through the unchanged legacy code above.
// ─────────────────────────────────────────────────────────────────────────────

const MODERN_METHODS = new Set([
  'server/discover',
  'tools/list',
  'tools/call',
  'resources/list',
  'resources/read',
  'prompts/list',
  'prompts/get'
]);

// Cache hints required on list/read results. Everything here is behind a per-student
// token (and tools/list varies by scope), so nothing is shareable: scope is always private.
const CACHE_HINTS: Record<string, { ttlMs: number; cacheScope: 'private' }> = {
  'server/discover': { ttlMs: 300_000, cacheScope: 'private' },
  'tools/list': { ttlMs: 60_000, cacheScope: 'private' },
  'prompts/list': { ttlMs: 300_000, cacheScope: 'private' },
  'resources/list': { ttlMs: 300_000, cacheScope: 'private' },
  'resources/read': { ttlMs: 0, cacheScope: 'private' } // live student data
};

function isModernRequest(req: NextRequest, body: unknown): boolean {
  if (req.headers.get('mcp-protocol-version') === MODERN_VERSION) return true;
  const first: any = Array.isArray(body) ? body[0] : body;
  const metaVersion = first?.params?._meta?.[META_VERSION];
  return typeof metaVersion === 'string';
}

/** Mcp-Name / Mcp-Param values that are not plain ASCII arrive as =?base64?...?= (spec: Value Encoding). */
function decodeHeaderValue(raw: string): string {
  const m = /^=\?base64\?(.*)\?=$/.exec(raw);
  if (!m) return raw;
  try {
    return Buffer.from(m[1], 'base64').toString('utf8');
  } catch {
    return raw;
  }
}

function modernError(status: number, id: JsonRpcId | undefined, code: number, message: string, data?: unknown) {
  const error: Record<string, unknown> = { code, message };
  if (data !== undefined) error.data = data;
  return NextResponse.json({ jsonrpc: '2.0', id: id ?? null, error }, { status, headers: { 'Cache-Control': 'no-store' } });
}

async function handleModern(req: NextRequest, principal: GatewayPrincipal, body: unknown): Promise<NextResponse> {
  if (Array.isArray(body)) {
    return modernError(400, null, ERR.invalidRequest, 'Batches are not supported in 2026-07-28: send one JSON-RPC message per POST.');
  }
  const msg = body as JsonRpcRequest;
  if (!msg || msg.jsonrpc !== '2.0' || typeof msg.method !== 'string') {
    return modernError(400, (msg as any)?.id, ERR.invalidRequest, 'Invalid JSON-RPC request');
  }
  const id = msg.id;

  // Version: header and body `_meta` must agree, then must be one we serve.
  const headerVersion = req.headers.get('mcp-protocol-version');
  const metaVersion = msg.params?._meta?.[META_VERSION];
  if (typeof metaVersion !== 'string' || headerVersion !== metaVersion) {
    return modernError(
      400,
      id,
      ERR_MODERN.headerMismatch,
      `Header mismatch: MCP-Protocol-Version header (${headerVersion ?? 'missing'}) must match _meta["${META_VERSION}"] (${typeof metaVersion === 'string' ? metaVersion : 'missing'}).`
    );
  }
  if (metaVersion !== MODERN_VERSION) {
    return modernError(400, id, ERR_MODERN.unsupportedVersion, 'Unsupported protocol version', { supported: ALL_VERSIONS, requested: metaVersion });
  }

  // This revision defines no client notifications over HTTP; accept and ignore any that arrive.
  if (id === undefined) return new NextResponse(null, { status: 202 });

  // Routing headers must mirror the body (a gateway may have routed on them).
  const methodHeader = req.headers.get('mcp-method');
  if (methodHeader !== msg.method) {
    return modernError(400, id, ERR_MODERN.headerMismatch, `Header mismatch: Mcp-Method header (${methodHeader ?? 'missing'}) does not match body method (${msg.method}).`);
  }
  const nameSource =
    msg.method === 'tools/call' || msg.method === 'prompts/get' ? msg.params?.name : msg.method === 'resources/read' ? msg.params?.uri : undefined;
  if (msg.method === 'tools/call' || msg.method === 'prompts/get' || msg.method === 'resources/read') {
    const nameHeader = req.headers.get('mcp-name');
    if (nameHeader === null || typeof nameSource !== 'string' || decodeHeaderValue(nameHeader) !== nameSource) {
      return modernError(400, id, ERR_MODERN.headerMismatch, 'Header mismatch: Mcp-Name header is missing or does not match the request body.');
    }
  }

  if (!MODERN_METHODS.has(msg.method)) {
    return modernError(404, id, ERR.methodNotFound, `Method not found: ${msg.method}`);
  }

  const baseMeta = { [META_SERVER_INFO]: { name: SERVER_INFO.name, version: SERVER_INFO.version } };
  const finish = (result: Record<string, unknown>) => {
    const hint = CACHE_HINTS[msg.method];
    return {
      ...result,
      resultType: 'complete',
      ...(hint ?? {}),
      _meta: { ...((result._meta as Record<string, unknown>) ?? {}), ...baseMeta }
    };
  };

  if (msg.method === 'server/discover') {
    void recordAudit({ principal, method: msg.method, ok: true, durationMs: 0 });
    return NextResponse.json(
      rpcResult(id, finish({ supportedVersions: ALL_VERSIONS, capabilities: { tools: {}, resources: {}, prompts: {} }, instructions: INSTRUCTIONS })),
      { headers: { 'Cache-Control': 'no-store' } }
    );
  }

  const reply: any = await handleOne(msg, principal);
  if (reply && typeof reply === 'object' && 'result' in reply && reply.result && typeof reply.result === 'object') {
    reply.result = finish(reply.result as Record<string, unknown>);
  }
  return NextResponse.json(reply, { headers: { 'Cache-Control': 'no-store' } });
}

export async function handleMcpPost(req: NextRequest): Promise<NextResponse> {
  if (!originAllowed(req)) return NextResponse.json({ error: 'Origin not allowed' }, { status: 403 });

  const auth = await authenticateGateway(req);
  if (!auth.ok) {
    return NextResponse.json(rpcError(null, ERR.invalidRequest, auth.message), {
      status: auth.status,
      headers: { 'WWW-Authenticate': 'Bearer realm="checkmate-gateway"' }
    });
  }
  const { principal } = auth;

  if (await isRateLimited(principal.tokenId)) {
    return NextResponse.json(rpcError(null, ERR.internal, 'Rate limit exceeded — slow down and retry in a minute.'), {
      status: 429,
      headers: { 'Retry-After': '30' }
    });
  }

  let body: unknown;
  try {
    body = await req.json();
  } catch {
    return NextResponse.json(rpcError(null, ERR.parse, 'Parse error'), { status: 400 });
  }

  if (isModernRequest(req, body)) return handleModern(req, principal, body);

  const isBatch = Array.isArray(body);
  const messages = (isBatch ? body : [body]) as JsonRpcRequest[];
  if (messages.length === 0 || messages.length > 10) {
    return NextResponse.json(rpcError(null, ERR.invalidRequest, 'Batch must contain 1-10 messages'), { status: 400 });
  }

  const replies: unknown[] = [];
  for (const m of messages) {
    if (!m || m.jsonrpc !== '2.0' || typeof m.method !== 'string') {
      replies.push(rpcError((m as any)?.id, ERR.invalidRequest, 'Invalid JSON-RPC request'));
      continue;
    }
    const r = await handleOne(m, principal);
    if (r !== null) replies.push(r);
  }

  if (replies.length === 0) return new NextResponse(null, { status: 202 }); // notifications only
  return NextResponse.json(isBatch ? replies : replies[0], { headers: { 'Cache-Control': 'no-store' } });
}

import { NextRequest, NextResponse } from 'next/server';
import { getCurrentUserFromRequest } from '@/lib/auth';
import { getServiceClient } from '@/lib/supabase';
import { isProjectionKind, MAX_PROJECTION_BYTES, PROJECTION_KINDS, ProjectionError, projectionStatus, putProjection } from '@/lib/gateway/projection';

export const dynamic = 'force-dynamic';
export const maxDuration = 30;

// PUT /api/gateway/projection?kind=student_model|interventions|today_plan|knowledge_graph|behavior|task_history
// Called by the Checkmate Android app (device-token Bearer, same as every other Checkmate
// call). Stores a curated, re-normalized snapshot that the MCP gateway later READS.
// This is the only write path into the gateway's data, and it can only ever write the
// caller's own projection rows.
export async function PUT(req: NextRequest) {
  const user = await getCurrentUserFromRequest(req);
  if (!user) return NextResponse.json({ error: 'Not authenticated' }, { status: 401 });

  const kind = req.nextUrl.searchParams.get('kind');
  if (!isProjectionKind(kind)) {
    return NextResponse.json({ error: `kind must be one of: ${PROJECTION_KINDS.join(', ')}` }, { status: 400 });
  }

  const raw = await req.text();
  if (Buffer.byteLength(raw, 'utf8') > MAX_PROJECTION_BYTES) {
    return NextResponse.json({ error: 'Payload too large' }, { status: 413 });
  }
  let body: unknown;
  try {
    body = JSON.parse(raw);
  } catch {
    return NextResponse.json({ error: 'Invalid JSON body' }, { status: 400 });
  }

  try {
    const result = await putProjection(getServiceClient(), user.id, kind, body);
    return NextResponse.json(result);
  } catch (err: any) {
    if (err instanceof ProjectionError) return NextResponse.json({ error: err.message }, { status: 400 });
    console.error('[gateway/projection] failed:', err?.message);
    return NextResponse.json({ error: 'Failed to store projection' }, { status: 500 });
  }
}

// GET -> which projections exist and how fresh (lets the phone/UI show "last synced").
export async function GET(req: NextRequest) {
  const user = await getCurrentUserFromRequest(req);
  if (!user) return NextResponse.json({ error: 'Not authenticated' }, { status: 401 });
  return NextResponse.json({ projections: await projectionStatus(getServiceClient(), user.id) });
}
