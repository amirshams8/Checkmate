//===== TERMUXFILE: Testmate/db/migrations/018_mcp_gateway.sql ===== 
-- 018_mcp_gateway.sql
-- Checkmate MCP gateway (read-only v1). Run once in the Supabase SQL editor.
--
--  mcp_tokens         scoped bearer tokens for EXTERNAL AI hosts (ChatGPT / Claude).
--                     Deliberately a separate table from device_tokens: a leaked
--                     Checkmate device token must never grant AI-gateway access, and
--                     vice versa. Only the SHA-256 hash is stored.
--  gateway_audit      one row per gateway call (who/what/when/how big). Also backs
--                     the per-token rate limit.
--  gateway_projection curated snapshots PUSHED by the Android app (student model,
--                     intervention history, today's plan). The gateway never reaches
--                     into the phone's Room DB; it only reads these rows.
--
-- RLS is enabled with NO policies on purpose: only the service-role client used by
-- the API routes can touch these tables.

create table if not exists mcp_tokens (
  id            uuid primary key default gen_random_uuid(),
  user_id       uuid not null references auth.users(id) on delete cascade,
  token_hash    text not null unique,
  label         text not null default 'External AI',
  scopes        text[] not null default array['checkmate:read', 'checkmate:analysis'],
  created_at    timestamptz not null default now(),
  last_used_at  timestamptz,
  expires_at    timestamptz,
  revoked_at    timestamptz
);
create index if not exists mcp_tokens_user_idx on mcp_tokens (user_id);

create table if not exists gateway_audit (
  id             bigserial primary key,
  user_id        uuid not null,
  token_id       uuid,
  method         text not null,
  tool           text,
  args           jsonb,
  ok             boolean not null,
  error          text,
  duration_ms    integer,
  response_bytes integer,
  created_at     timestamptz not null default now()
);
create index if not exists gateway_audit_token_time_idx on gateway_audit (token_id, created_at desc);
create index if not exists gateway_audit_user_time_idx on gateway_audit (user_id, created_at desc);

create table if not exists gateway_projection (
  user_id        uuid not null references auth.users(id) on delete cascade,
  kind           text not null,
  payload        jsonb not null,
  schema_version integer not null default 1,
  generated_at   timestamptz not null,
  received_at    timestamptz not null default now(),
  primary key (user_id, kind)
);

alter table mcp_tokens         enable row level security;
alter table gateway_audit      enable row level security;
alter table gateway_projection enable row level security;

//===== TERMUXFILE: Testmate/lib/gateway/auth.ts =====
import { randomBytes, createHash } from 'crypto';
import { NextRequest } from 'next/server';
import { getServiceClient } from '@/lib/supabase';
import { GatewayPrincipal, Scope, isScope } from './types';

// Distinct from device tokens ('tmk_'): the gateway ONLY accepts 'tmcp_' tokens, so a
// Checkmate device token pasted into an AI host cannot open the gateway.
export const MCP_TOKEN_PREFIX = 'tmcp_';

export function generateMcpToken(): { raw: string; hash: string } {
  const raw = MCP_TOKEN_PREFIX + randomBytes(32).toString('hex');
  return { raw, hash: hashMcpToken(raw) };
}

export function hashMcpToken(raw: string): string {
  return createHash('sha256').update(raw).digest('hex');
}

export type GatewayAuthResult =
  | { ok: true; principal: GatewayPrincipal }
  | { ok: false; status: 401 | 403; message: string };

const LAST_USED_WRITE_INTERVAL_MS = 60_000;

/**
 * Bearer-only. Deliberately does NOT fall back to the browser cookie session — an
 * external AI host never has one, and accepting it here would let a logged-in browser
 * tab be driven through the gateway by any page that can reach this origin.
 */
export async function authenticateGateway(req: NextRequest): Promise<GatewayAuthResult> {
  const header = req.headers.get('authorization');
  if (!header?.startsWith('Bearer ')) {
    return { ok: false, status: 401, message: 'Missing bearer token' };
  }
  const raw = header.slice('Bearer '.length).trim();
  if (!raw.startsWith(MCP_TOKEN_PREFIX)) {
    return { ok: false, status: 401, message: 'Invalid token' };
  }

  const supabase = getServiceClient();
  const { data: row, error } = await supabase
    .from('mcp_tokens')
    .select('id, user_id, label, scopes, revoked_at, expires_at, last_used_at')
    .eq('token_hash', hashMcpToken(raw))
    .maybeSingle();

  if (error || !row || row.revoked_at) {
    return { ok: false, status: 401, message: 'Invalid or revoked token' };
  }
  if (row.expires_at && new Date(row.expires_at).getTime() <= Date.now()) {
    return { ok: false, status: 401, message: 'Token expired' };
  }

  const scopes = ((row.scopes as unknown[]) ?? []).filter(isScope) as Scope[];

  const lastUsed = row.last_used_at ? new Date(row.last_used_at).getTime() : 0;
  if (Date.now() - lastUsed > LAST_USED_WRITE_INTERVAL_MS) {
    void supabase
      .from('mcp_tokens')
      .update({ last_used_at: new Date().toISOString() })
      .eq('id', row.id)
      .then(() => {});
  }

  return {
    ok: true,
    principal: { userId: row.user_id as string, tokenId: row.id as string, label: row.label as string, scopes }
  };
}

export function hasScope(principal: GatewayPrincipal, scope: Scope): boolean {
  return principal.scopes.includes(scope);
}

//===== TERMUXFILE: Testmate/lib/gateway/audit.ts =====
import { getServiceClient } from '@/lib/supabase';
import { GatewayPrincipal } from './types';

const MAX_ARGS_CHARS = 2000;

export interface AuditEntry {
  principal: GatewayPrincipal;
  method: string;
  tool?: string;
  args?: unknown;
  ok: boolean;
  error?: string;
  durationMs?: number;
  responseBytes?: number;
}

function boundedArgs(args: unknown): unknown {
  if (args === undefined) return null;
  try {
    const s = JSON.stringify(args);
    if (s.length <= MAX_ARGS_CHARS) return args;
    return { _truncated: true, preview: s.slice(0, MAX_ARGS_CHARS) };
  } catch {
    return { _unserializable: true };
  }
}

/** Best-effort: an audit failure must never fail (or leak into) the tool call itself. */
export async function recordAudit(entry: AuditEntry): Promise<void> {
  try {
    const supabase = getServiceClient();
    await supabase.from('gateway_audit').insert({
      user_id: entry.principal.userId,
      token_id: entry.principal.tokenId,
      method: entry.method,
      tool: entry.tool ?? null,
      args: boundedArgs(entry.args),
      ok: entry.ok,
      error: entry.error ? entry.error.slice(0, 500) : null,
      duration_ms: entry.durationMs ?? null,
      response_bytes: entry.responseBytes ?? null
    });
  } catch (err: any) {
    console.error('[gateway/audit] insert failed:', err?.message);
  }
}

const DEFAULT_LIMIT_PER_MIN = 60;

/**
 * DB-backed sliding window over the audit table (serverless instances don't share
 * memory, so an in-process counter would let a client multiply its limit by the number
 * of warm lambdas). Counts every audited call for this token in the last 60s.
 * Fails OPEN on a DB error — the bearer token is still the real gate.
 */
export async function isRateLimited(tokenId: string): Promise<boolean> {
  const limit = Number(process.env.MCP_RATE_LIMIT_PER_MIN) || DEFAULT_LIMIT_PER_MIN;
  try {
    const supabase = getServiceClient();
    const since = new Date(Date.now() - 60_000).toISOString();
    const { count, error } = await supabase
      .from('gateway_audit')
      .select('id', { count: 'exact', head: true })
      .eq('token_id', tokenId)
      .gte('created_at', since);
    if (error) return false;
    return (count ?? 0) >= limit;
  } catch {
    return false;
  }
}

//===== TERMUXFILE: Testmate/lib/gateway/sanitize.ts =====
// Output minimization. The gateway returns meaning, not database structure — this is the
// last line of defense that keeps internal identifiers and bulk out of what an external
// model sees, whatever a tool handler happened to put in its result.

// Never leave the server: ownership ids and anything credential-shaped. (Session ids,
// question ids and test ids are intentionally allowed — models need them to chain calls.)
const DROP_KEYS = new Set([
  'user_id',
  'owner_user_id',
  'token_hash',
  'token',
  'password',
  'secret',
  'api_key',
  'authorization'
]);

const MAX_STRING_CHARS = 4000;
export const MAX_RESPONSE_BYTES = 90_000;

export function minimize(value: unknown, depth = 0): unknown {
  if (depth > 12) return null;
  if (value === null || value === undefined) return value ?? null;
  if (typeof value === 'string') {
    return value.length > MAX_STRING_CHARS ? value.slice(0, MAX_STRING_CHARS) + '…[truncated]' : value;
  }
  if (typeof value !== 'object') return value;
  if (Array.isArray(value)) return value.map((v) => minimize(v, depth + 1));
  const out: Record<string, unknown> = {};
  for (const [k, v] of Object.entries(value as Record<string, unknown>)) {
    if (DROP_KEYS.has(k.toLowerCase())) continue;
    out[k] = minimize(v, depth + 1);
  }
  return out;
}

/**
 * Hard ceiling on one tool result. If it's too big, pop items off the largest top-level
 * array until it fits and say so, rather than returning a megabyte the host will choke
 * on or silently cutting JSON mid-token.
 */
export function capResponse(value: unknown, maxBytes = MAX_RESPONSE_BYTES): { text: string; truncated: boolean } {
  let text = JSON.stringify(value);
  if (Buffer.byteLength(text, 'utf8') <= maxBytes) return { text, truncated: false };
  if (value === null || typeof value !== 'object' || Array.isArray(value)) {
    return {
      text: JSON.stringify({ _truncated: true, note: 'Result too large; narrow the filters.' }),
      truncated: true
    };
  }
  const obj = { ...(value as Record<string, unknown>) };
  let dropped = 0;
  for (let guard = 0; guard < 5000; guard += 1) {
    let bestKey: string | null = null;
    let bestLen = 0;
    for (const [k, v] of Object.entries(obj)) {
      if (Array.isArray(v) && v.length > bestLen) {
        bestKey = k;
        bestLen = v.length;
      }
    }
    if (!bestKey || bestLen === 0) break;
    const arr = obj[bestKey] as unknown[];
    obj[bestKey] = arr.slice(0, Math.max(0, Math.floor(arr.length * 0.8) - (arr.length > 1 ? 0 : 1)));
    dropped += arr.length - (obj[bestKey] as unknown[]).length;
    obj._truncated = true;
    obj._note = `Result exceeded the size limit; ${dropped} item(s) omitted. Narrow the filters (subject/chapter/days/limit) to see them.`;
    text = JSON.stringify(obj);
    if (Buffer.byteLength(text, 'utf8') <= maxBytes) return { text, truncated: true };
  }
  return {
    text: JSON.stringify({ _truncated: true, note: 'Result too large; narrow the filters.' }),
    truncated: true
  };
}

//===== TERMUXFILE: Testmate/lib/gateway/projection.ts =====
import type { SupabaseClient } from '@supabase/supabase-js';

// ─────────────────────────────────────────────────────────────────────────────
// Gateway read-projection (FINAL_PLAN "Option C — hybrid").
//
// The Android app stays the source of truth for the student model, interventions and
// today's plan. It PUSHES curated snapshots here (PUT /api/gateway/projection, device-token
// auth); the MCP gateway only ever READS these rows. The gateway never gets arbitrary SQL
// against anything, and keeps working while the phone is offline (with a visible
// freshness stamp so the model can say "as of 3h ago").
//
// Every pushed payload is re-normalized field by field below — unknown fields are dropped,
// strings bounded, arrays capped — so the phone can't (accidentally) smuggle internals,
// secrets, or megabytes into what an external model eventually reads.
// ─────────────────────────────────────────────────────────────────────────────

export const PROJECTION_KINDS = ['student_model', 'interventions', 'today_plan'] as const;
export type ProjectionKind = (typeof PROJECTION_KINDS)[number];
export const MAX_PROJECTION_BYTES = 800_000;
const SCHEMA_VERSION = 1;

export class ProjectionError extends Error {}

function str(v: unknown, max = 200): string | null {
  if (typeof v !== 'string') return null;
  const t = v.trim();
  return t ? t.slice(0, max) : null;
}
function num(v: unknown): number | null {
  return typeof v === 'number' && Number.isFinite(v) ? v : null;
}
function int(v: unknown): number {
  const n = num(v);
  return n === null ? 0 : Math.trunc(n);
}
function bool(v: unknown): boolean {
  return v === true;
}
function arr(v: unknown, max: number): any[] {
  return Array.isArray(v) ? v.slice(0, max) : [];
}
function epochToIso(v: unknown): string | null {
  const n = num(v);
  return n === null || n <= 0 ? null : new Date(n).toISOString();
}

function normalizeStudentModel(b: any) {
  const o = b?.overall ?? {};
  const ref = (p: any) => ({ subject: str(p?.subject), chapter: str(p?.chapter), topic: str(p?.topic) });
  return {
    exam: str(b?.exam, 60),
    model_version: int(b?.model_version) || 1,
    overall: {
      concepts_tracked: int(o.concepts_tracked),
      concepts_mastered: int(o.concepts_mastered),
      concepts_weak: int(o.concepts_weak),
      average_mastery: num(o.average_mastery),
      total_attempts: int(o.total_attempts),
      unresolved_error_count: int(o.unresolved_error_count)
    },
    concepts: arr(b?.concepts, 3000).map((c) => ({
      concept_id: str(c?.concept_id, 120),
      subject: str(c?.subject),
      chapter: str(c?.chapter),
      topic: str(c?.topic),
      mastery: num(c?.mastery),
      forgetting_risk: num(c?.forgetting_risk),
      retention_decision: ['REVIEW', 'TEACH', 'MOVE_ON'].includes(c?.retention_decision) ? c.retention_decision : null,
      attempt_count: int(c?.attempt_count),
      recent_accuracy: num(c?.recent_accuracy),
      lifetime_accuracy: num(c?.lifetime_accuracy),
      error_count: int(c?.error_count),
      last_seen_at: epochToIso(c?.last_seen_ms),
      weak: bool(c?.weak),
      prerequisite_issues: arr(c?.prerequisite_issues, 10).map(ref)
    })),
    unresolved_errors: arr(b?.unresolved_errors, 500).map((e) => ({
      concept_id: str(e?.concept_id, 120),
      subject: str(e?.subject),
      chapter: str(e?.chapter),
      topic: str(e?.topic),
      error_type: str(e?.error_type, 40),
      occurrences: int(e?.occurrences),
      first_seen_at: epochToIso(e?.first_seen_ms),
      last_seen_at: epochToIso(e?.last_seen_ms)
    }))
  };
}

function normalizeInterventions(b: any) {
  return {
    entries: arr(b?.entries, 300).map((e) => ({
      transaction_id: str(e?.transaction_id, 80),
      task_id: str(e?.task_id, 80),
      trigger_type: str(e?.trigger_type, 60),
      terminal_state: str(e?.terminal_state, 60),
      provenance: str(e?.provenance, 60),
      attempt_count: int(e?.attempt_count),
      resolved_at: epochToIso(e?.resolved_ms),
      outcome: str(e?.outcome, 300),
      failure_reason: str(e?.failure_reason, 300)
    }))
  };
}

function normalizeTodayPlan(b: any) {
  return {
    day_key: str(b?.day_key, 20),
    completion_percent: num(b?.completion_percent),
    tasks: arr(b?.tasks, 100).map((t) => ({
      id: str(t?.id, 80),
      subject: str(t?.subject, 80),
      topic: str(t?.topic, 200),
      duration_minutes: int(t?.duration_minutes),
      state: str(t?.state, 20),
      task_type: str(t?.task_type, 20),
      scheduled_start_time: str(t?.scheduled_start_time, 10),
      completed_status: str(t?.completed_status, 10),
      learning_intent: str(t?.learning_intent, 40),
      concept_id: str(t?.concept_id, 120),
      rationale: str(t?.rationale, 400),
      focus_minutes: int(t?.focus_minutes)
    }))
  };
}

export function isProjectionKind(v: unknown): v is ProjectionKind {
  return typeof v === 'string' && (PROJECTION_KINDS as readonly string[]).includes(v);
}

export async function putProjection(sb: SupabaseClient, userId: string, kind: ProjectionKind, body: any) {
  if (!body || typeof body !== 'object') throw new ProjectionError('Body must be a JSON object');
  const payload =
    kind === 'student_model'
      ? normalizeStudentModel(body)
      : kind === 'interventions'
        ? normalizeInterventions(body)
        : normalizeTodayPlan(body);

  // Phone clocks drift; refuse nonsense but tolerate a little.
  let generatedMs = typeof body.generated_ms === 'number' ? body.generated_ms : Date.now();
  if (!Number.isFinite(generatedMs) || generatedMs > Date.now() + 5 * 60_000) generatedMs = Date.now();

  // Never let an older snapshot overwrite a newer one (retried/out-of-order pushes).
  const { data: existing } = await sb
    .from('gateway_projection')
    .select('generated_at')
    .eq('user_id', userId)
    .eq('kind', kind)
    .maybeSingle();
  if (existing && new Date(existing.generated_at).getTime() > generatedMs) {
    return { stored: false, reason: 'A newer snapshot is already stored', generated_at: existing.generated_at as string };
  }

  const { error } = await sb.from('gateway_projection').upsert(
    {
      user_id: userId,
      kind,
      payload,
      schema_version: SCHEMA_VERSION,
      generated_at: new Date(generatedMs).toISOString(),
      received_at: new Date().toISOString()
    },
    { onConflict: 'user_id,kind' }
  );
  if (error) throw error;
  return { stored: true, generated_at: new Date(generatedMs).toISOString() };
}

export interface StoredProjection<T = any> {
  payload: T;
  generated_at: string;
  received_at: string;
}

export async function getProjection<T = any>(
  sb: SupabaseClient,
  userId: string,
  kind: ProjectionKind
): Promise<StoredProjection<T> | null> {
  const { data, error } = await sb
    .from('gateway_projection')
    .select('payload, generated_at, received_at')
    .eq('user_id', userId)
    .eq('kind', kind)
    .maybeSingle();
  if (error || !data) return null;
  return data as StoredProjection<T>;
}

export function freshness(generatedAt: string, staleAfterMinutes = 360) {
  const ageMin = Math.max(0, Math.round((Date.now() - new Date(generatedAt).getTime()) / 60_000));
  return { generated_at: generatedAt, age_minutes: ageMin, stale: ageMin > staleAfterMinutes };
}

export async function projectionStatus(sb: SupabaseClient, userId: string) {
  const { data } = await sb
    .from('gateway_projection')
    .select('kind, generated_at, received_at, schema_version')
    .eq('user_id', userId);
  return (data ?? []).map((r: any) => ({
    kind: r.kind,
    generated_at: r.generated_at,
    received_at: r.received_at,
    schema_version: r.schema_version
  }));
}

// ─────────────────────────────────────────────────────────────────────────────
// Learning Evidence layer (FINAL_PLAN: question-level evidence is core to v1).
//
// Everything here is question-centric: the actual question, the student's answer, the
// right answer, the explanation, WHY the question was in front of them (provenance /
// purpose), and what happened. Nothing returns a raw table row — rows are reshaped into
// semantic objects, so the Room/Supabase schema can evolve without breaking AI clients.
//
// Attempts are always resolved to the CANONICAL question (origin_question_id ?? id, same
// rule as lib/retention.ts) so a question copied into three Q-bank / repair tests reads as
// one question with one history, not three.
// ─────────────────────────────────────────────────────────────────────────────

const CHUNK_SIZE = 150;

export const QUESTION_SELECT =
  'id, test_id, question_number, question_text, question_image_url, question_data, options, option_images, correct_answer, marks, negative_marks, chapter, topic, difficulty, question_type, explanation, origin_question_id, tag_source';

export type Provenance = 'full_test' | 'qbank_practice' | 'targeted_repair' | 'retention_check' | 'coaching_bank';
export type AttemptStatus = 'CORRECT' | 'WRONG' | 'SKIPPED' | 'UNSCORED';
export type SourceFilter = 'all' | Provenance;
export const SOURCE_VALUES: SourceFilter[] = [
  'all',
  'full_test',
  'qbank_practice',
  'targeted_repair',
  'retention_check',
  'coaching_bank'
];

function chunk<T>(arr: T[], size = CHUNK_SIZE): T[][] {
  const out: T[][] = [];
  for (let i = 0; i < arr.length; i += size) out.push(arr.slice(i, i + size));
  return out;
}

/** `.in()` in URL-safe batches (same defense as app/api/tests/targeted — see its BUGFIX note). */
async function selectIn(
  sb: SupabaseClient,
  table: string,
  select: string,
  column: string,
  ids: string[],
  opts: { paged?: boolean; decorate?: (q: any) => any } = {}
): Promise<any[]> {
  const out: any[] = [];
  const unique = Array.from(new Set(ids.filter(Boolean)));
  for (const batch of chunk(unique)) {
    if (!opts.paged) {
      let q: any = sb.from(table).select(select).in(column, batch);
      if (opts.decorate) q = opts.decorate(q);
      const { data, error } = await q;
      if (error) throw error;
      out.push(...(data ?? []));
      continue;
    }
    // PostgREST caps one response at 1000 rows — page through with a stable order.
    for (let from = 0; ; from += 1000) {
      let q: any = sb.from(table).select(select).in(column, batch).order('id', { ascending: true }).range(from, from + 999);
      if (opts.decorate) q = opts.decorate(q);
      const { data, error } = await q;
      if (error) throw error;
      const rows = data ?? [];
      out.push(...rows);
      if (rows.length < 1000) break;
    }
  }
  return out;
}

export function provenanceOf(generatedFrom: unknown, sessionInterventionId?: string | null): Provenance {
  const gf = (generatedFrom && typeof generatedFrom === 'object' ? generatedFrom : null) as GeneratedFrom | null;
  const interventionId = gf?.intervention_id ?? sessionInterventionId ?? null;
  if (gf?.kind === 'qbank_practice') return 'qbank_practice';
  if (gf?.kind === 'coaching_bank') return 'coaching_bank';
  if (interventionId) return interventionId.startsWith('retention-') ? 'retention_check' : 'targeted_repair';
  if (gf?.kind === 'targeted_repair') return 'targeted_repair';
  return 'full_test';
}

export function statusOf(selected: string | null | undefined, isCorrect: boolean | null | undefined): AttemptStatus {
  if (selected === null || selected === undefined || selected === '') return 'SKIPPED';
  if (isCorrect === true) return 'CORRECT';
  if (isCorrect === false) return 'WRONG';
  return 'UNSCORED';
}

// ── Time helpers (the student is in India; "today" for Q-bank means their local day) ──
function offsetMinutes(): number {
  const raw = process.env.GATEWAY_TZ_OFFSET || '+05:30';
  const m = /^([+-])(\d{2}):(\d{2})$/.exec(raw);
  if (!m) return 330;
  return (m[1] === '-' ? -1 : 1) * (Number(m[2]) * 60 + Number(m[3]));
}
export function localDateString(at = new Date()): string {
  return new Date(at.getTime() + offsetMinutes() * 60_000).toISOString().slice(0, 10);
}
function localDayBounds(date: string): { start: string; end: string } {
  const off = process.env.GATEWAY_TZ_OFFSET || '+05:30';
  const start = new Date(`${date}T00:00:00${off}`);
  return { start: start.toISOString(), end: new Date(start.getTime() + 86_400_000).toISOString() };
}
function sinceFromDays(days: number): string {
  return new Date(Date.now() - days * 86_400_000).toISOString();
}

export interface Attempt {
  response_id: string;
  session_id: string;
  question: any;
  canonical_id: string;
  status: AttemptStatus;
  selected_answer: string | null;
  answered_at: string | null;
  time_spent_seconds: number | null;
  provenance: Provenance;
  test_id: string | null;
  test_title: string | null;
  intervention_id: string | null;
  skip_kind?: 'left_blank' | 'never_opened';
}

export function shapeAttempt(a: Attempt) {
  return {
    status: a.status,
    selected_answer: a.selected_answer,
    answered_at: a.answered_at,
    time_spent_seconds: a.time_spent_seconds,
    provenance: a.provenance,
    test_title: a.test_title,
    session_id: a.session_id,
    intervention_id: a.intervention_id,
    ...(a.skip_kind ? { skip_kind: a.skip_kind } : {})
  };
}

export function summarizeHistory(group: Attempt[]) {
  const wrong = group.filter((a) => a.status === 'WRONG').length;
  const skipped = group.filter((a) => a.status === 'SKIPPED').length;
  const correct = group.filter((a) => a.status === 'CORRECT').length;
  const lastCorrect = group.find((a) => a.status === 'CORRECT');
  const lastWrong = group.find((a) => a.status === 'WRONG');
  return {
    attempts: group.length,
    correct,
    wrong,
    skipped,
    last_correct_at: lastCorrect?.answered_at ?? null,
    last_wrong_at: lastWrong?.answered_at ?? null
  };
}

export function groupByCanonical(attempts: Attempt[]): Map<string, Attempt[]> {
  const map = new Map<string, Attempt[]>();
  for (const a of attempts) {
    const g = map.get(a.canonical_id);
    if (g) g.push(a);
    else map.set(a.canonical_id, [a]);
  }
  const ts = (a: Attempt) => (a.answered_at ? new Date(a.answered_at).getTime() : 0);
  for (const g of Array.from(map.values())) g.sort((x, y) => ts(y) - ts(x));
  return map;
}

// ── Loading ──────────────────────────────────────────────────────────────────
export async function getSubjectByChapter(sb: SupabaseClient): Promise<Map<string, string>> {
  const map = new Map<string, string>();
  const { data, error } = await sb.from('syllabus_chapters').select('chapter, subject');
  if (error) return map; // best-effort, same as lib/daily-target-engine.ts
  for (const r of data ?? []) if (r.chapter && r.subject && !map.has(r.chapter)) map.set(r.chapter, r.subject);
  return map;
}

/** Every response this student gave since `sinceIso`, enriched with question + provenance. */
export async function loadAttempts(
  sb: SupabaseClient,
  userId: string,
  opts: { sinceIso?: string; maxRows?: number } = {}
): Promise<Attempt[]> {
  const maxRows = opts.maxRows ?? 4000;
  const rows: any[] = [];
  for (let from = 0; rows.length < maxRows; from += 1000) {
    let q: any = sb
      .from('responses')
      .select('id, session_id, question_id, selected_answer, is_correct, answered_at, time_spent_seconds')
      .eq('user_id', userId)
      .order('answered_at', { ascending: false })
      .range(from, from + 999);
    if (opts.sinceIso) q = q.gte('answered_at', opts.sinceIso);
    const { data, error } = await q;
    if (error) throw error;
    const page = data ?? [];
    rows.push(...page);
    if (page.length < 1000) break;
  }
  return enrichResponses(sb, rows);
}

async function enrichResponses(sb: SupabaseClient, rows: any[]): Promise<Attempt[]> {
  if (rows.length === 0) return [];
  const [sessions, questions] = await Promise.all([
    selectIn(sb, 'test_sessions', 'id, test_id, intervention_id', 'id', rows.map((r) => r.session_id)),
    selectIn(sb, 'questions', QUESTION_SELECT, 'id', rows.map((r) => r.question_id))
  ]);
  const sessionById = new Map(sessions.map((s) => [s.id as string, s]));
  const questionById = new Map(questions.map((q) => [q.id as string, q]));
  const tests = await selectIn(
    sb,
    'tests',
    'id, title, generated_from, created_at',
    'id',
    sessions.map((s) => s.test_id as string)
  );
  const testById = new Map(tests.map((t) => [t.id as string, t]));

  const out: Attempt[] = [];
  for (const r of rows) {
    const q = questionById.get(r.question_id);
    if (!q) continue;
    const s = sessionById.get(r.session_id);
    const t = s ? testById.get(s.test_id) : undefined;
    out.push({
      response_id: r.id,
      session_id: r.session_id,
      question: q,
      canonical_id: canonicalQuestionId(q),
      status: statusOf(r.selected_answer, r.is_correct),
      selected_answer: r.selected_answer ?? null,
      answered_at: r.answered_at ?? null,
      time_spent_seconds: typeof r.time_spent_seconds === 'number' ? r.time_spent_seconds : null,
      provenance: provenanceOf(t?.generated_from, s?.intervention_id),
      test_id: s?.test_id ?? null,
      test_title: t?.title ?? null,
      intervention_id: s?.intervention_id ?? (t?.generated_from as any)?.intervention_id ?? null
    });
  }
  return out;
}

/**
 * Questions in a SUBMITTED session that have no response row at all — the student never
 * opened them. scoring.ts counts these as skipped (`!response || !selected`), but they
 * leave no `responses` trail, so left_blank-only logic (lib/qbank-review.ts's KNOWN
 * LIMITATION) misses them. Bounded to the most recent submitted sessions.
 */
async function loadNeverOpened(
  sb: SupabaseClient,
  userId: string,
  sinceIso: string,
  maxSessions = 25
): Promise<Attempt[]> {
  const { data: parts, error } = await sb
    .from('session_participants')
    .select('session_id, submitted_at')
    .eq('user_id', userId)
    .not('submitted_at', 'is', null)
    .gte('submitted_at', sinceIso)
    .order('submitted_at', { ascending: false })
    .limit(maxSessions);
  if (error) throw error;
  if (!parts || parts.length === 0) return [];

  const sessionIds = parts.map((p: any) => p.session_id as string);
  const sessions = await selectIn(sb, 'test_sessions', 'id, test_id, intervention_id', 'id', sessionIds);
  const testIds = sessions.map((s) => s.test_id as string);
  const [tests, questions, responded] = await Promise.all([
    selectIn(sb, 'tests', 'id, title, generated_from', 'id', testIds),
    selectIn(sb, 'questions', QUESTION_SELECT, 'test_id', testIds, { paged: true }),
    selectIn(sb, 'responses', 'id, session_id, question_id', 'session_id', sessionIds, {
      paged: true,
      decorate: (q) => q.eq('user_id', userId)
    })
  ]);

  const testById = new Map(tests.map((t) => [t.id as string, t]));
  const respondedBySession = new Map<string, Set<string>>();
  for (const r of responded) {
    const set = respondedBySession.get(r.session_id) ?? new Set<string>();
    set.add(r.question_id);
    respondedBySession.set(r.session_id, set);
  }
  const questionsByTest = new Map<string, any[]>();
  for (const q of questions) {
    const arr = questionsByTest.get(q.test_id) ?? [];
    arr.push(q);
    questionsByTest.set(q.test_id, arr);
  }
  const submittedAt = new Map(parts.map((p: any) => [p.session_id as string, p.submitted_at as string]));

  const out: Attempt[] = [];
  for (const s of sessions) {
    const t = testById.get(s.test_id);
    const seen = respondedBySession.get(s.id) ?? new Set<string>();
    for (const q of questionsByTest.get(s.test_id) ?? []) {
      if (seen.has(q.id)) continue;
      out.push({
        response_id: `never_opened:${s.id}:${q.id}`,
        session_id: s.id,
        question: q,
        canonical_id: canonicalQuestionId(q),
        status: 'SKIPPED',
        selected_answer: null,
        answered_at: submittedAt.get(s.id) ?? null,
        time_spent_seconds: null,
        provenance: provenanceOf(t?.generated_from, s.intervention_id),
        test_id: s.test_id,
        test_title: t?.title ?? null,
        intervention_id: s.intervention_id ?? (t?.generated_from as any)?.intervention_id ?? null,
        skip_kind: 'never_opened'
      });
    }
  }
  return out;
}

// ── Filters shared by the list tools ─────────────────────────────────────────
export interface AttemptFilters {
  days: number;
  subject?: string;
  chapter?: string;
  topic?: string;
  source: SourceFilter;
}

export function clampInt(v: unknown, def: number, min: number, max: number): number {
  const n = typeof v === 'number' && Number.isFinite(v) ? Math.floor(v) : def;
  return Math.min(max, Math.max(min, n));
}
function optString(v: unknown, name: string): string | undefined {
  if (v === undefined || v === null || v === '') return undefined;
  if (typeof v !== 'string') throw new ToolInputError(`${name} must be a string`);
  return v.trim() || undefined;
}
export function parseFilters(args: Record<string, unknown>, defaultDays = 60): AttemptFilters {
  const source = (optString(args.source, 'source') ?? 'all') as SourceFilter;
  if (!SOURCE_VALUES.includes(source)) {
    throw new ToolInputError(`source must be one of: ${SOURCE_VALUES.join(', ')}`);
  }
  return {
    days: clampInt(args.days, defaultDays, 1, 365),
    subject: optString(args.subject, 'subject'),
    chapter: optString(args.chapter, 'chapter'),
    topic: optString(args.topic, 'topic'),
    source
  };
}

function applyFilters(attempts: Attempt[], f: AttemptFilters, subjectByChapter: Map<string, string>): Attempt[] {
  return attempts.filter((a) => {
    if (f.source !== 'all' && a.provenance !== f.source) return false;
    if (f.chapter && a.question.chapter !== f.chapter) return false;
    if (f.topic && a.question.topic !== f.topic) return false;
    if (f.subject) {
      const subj = a.question.chapter ? subjectByChapter.get(a.question.chapter) : undefined;
      if (!subj || subj.toLowerCase() !== f.subject.toLowerCase()) return false;
    }
    return true;
  });
}

// ── Tool: wrong / skipped questions ──────────────────────────────────────────
export async function getWrongOrSkippedQuestions(
  sb: SupabaseClient,
  userId: string,
  kind: 'WRONG' | 'SKIPPED',
  args: Record<string, unknown>
) {
  const f = parseFilters(args);
  const limit = clampInt(args.limit, 20, 1, 50);
  const includeResolved = args.include_resolved === true;
  const minTimes = clampInt(args.min_times, 1, 1, 50);
  const since = sinceFromDays(f.days);

  const subjectByChapter = f.subject ? await getSubjectByChapter(sb) : new Map<string, string>();
  const attempts = applyFilters(await loadAttempts(sb, userId, { sinceIso: since }), f, subjectByChapter);

  let combined = attempts;
  if (kind === 'SKIPPED' && (f.source === 'all' || f.source === 'full_test' || f.source === 'qbank_practice')) {
    const neverOpened = applyFilters(await loadNeverOpened(sb, userId, since), f, subjectByChapter);
    combined = attempts.concat(neverOpened);
  }

  const groups = groupByCanonical(combined);
  const items: any[] = [];
  for (const [, group] of Array.from(groups.entries())) {
    const latest = group[0];
    const hist = summarizeHistory(group);
    const times = kind === 'WRONG' ? hist.wrong : hist.skipped;
    const qualifies = includeResolved ? times > 0 : latest.status === kind;
    if (!qualifies || times < minTimes) continue;
    items.push({
      question: shapeQuestion(latest.question),
      latest_attempt: shapeAttempt(latest),
      times_wrong: hist.wrong,
      times_skipped: hist.skipped,
      history: hist,
      still_unresolved: latest.status !== 'CORRECT'
    });
  }

  const ts = (x: any) => (x.latest_attempt.answered_at ? new Date(x.latest_attempt.answered_at).getTime() : 0);
  items.sort((a, b) => {
    const ta = kind === 'WRONG' ? a.times_wrong : a.times_skipped;
    const tb = kind === 'WRONG' ? b.times_wrong : b.times_skipped;
    return tb - ta || ts(b) - ts(a);
  });

  return {
    kind,
    window_days: f.days,
    filters: { subject: f.subject ?? null, chapter: f.chapter ?? null, topic: f.topic ?? null, source: f.source },
    note:
      kind === 'SKIPPED'
        ? "skip_kind 'left_blank' = opened but unanswered; 'never_opened' = in a submitted test but never visited."
        : undefined,
    total_matching: items.length,
    returned: Math.min(items.length, limit),
    questions: items.slice(0, limit)
  };
}

// ── Tool: recent attempts (compact stream, no explanations) ──────────────────
export async function getRecentAttempts(sb: SupabaseClient, userId: string, args: Record<string, unknown>) {
  const f = parseFilters(args, 14);
  const limit = clampInt(args.limit, 30, 1, 100);
  const result = (optString(args.result, 'result') ?? 'all').toLowerCase();
  if (!['all', 'correct', 'wrong', 'skipped'].includes(result)) {
    throw new ToolInputError('result must be one of: all, correct, wrong, skipped');
  }
  const subjectByChapter = f.subject ? await getSubjectByChapter(sb) : new Map<string, string>();
  let attempts = applyFilters(await loadAttempts(sb, userId, { sinceIso: sinceFromDays(f.days), maxRows: 2000 }), f, subjectByChapter);
  if (result !== 'all') attempts = attempts.filter((a) => a.status === result.toUpperCase());

  const counts = { correct: 0, wrong: 0, skipped: 0 };
  for (const a of attempts) {
    if (a.status === 'CORRECT') counts.correct += 1;
    else if (a.status === 'WRONG') counts.wrong += 1;
    else if (a.status === 'SKIPPED') counts.skipped += 1;
  }
  return {
    window_days: f.days,
    filters: { subject: f.subject ?? null, chapter: f.chapter ?? null, topic: f.topic ?? null, source: f.source, result },
    totals: counts,
    note: 'Compact stream. Use checkmate_get_question_context / checkmate_get_wrong_questions for the question text and explanation.',
    attempts: attempts.slice(0, limit).map((a) => ({
      ...compactQuestion(a.question),
      ...shapeAttempt(a)
    }))
  };
}

// ── Sessions (Q-bank / tests) ────────────────────────────────────────────────
async function loadSessionDetail(sb: SupabaseClient, userId: string, sessionId: string) {
  const { data: session, error } = await sb
    .from('test_sessions')
    .select('id, test_id, mode, status, intervention_id, server_start_time')
    .eq('id', sessionId)
    .maybeSingle();
  if (error) throw error;
  if (!session) throw new ToolInputError('session_id not found');

  const { data: participant } = await sb
    .from('session_participants')
    .select('submitted_at')
    .eq('session_id', sessionId)
    .eq('user_id', userId)
    .maybeSingle();
  if (!participant) throw new ToolInputError('session_id not found'); // not this student's session

  const [{ data: test }, questions, { data: responses }] = await Promise.all([
    sb.from('tests').select('id, title, generated_from, duration_seconds, total_marks, created_at').eq('id', session.test_id).maybeSingle(),
    selectIn(sb, 'questions', QUESTION_SELECT, 'test_id', [session.test_id], { paged: true }),
    sb
      .from('responses')
      .select('question_id, selected_answer, is_correct, answered_at, time_spent_seconds')
      .eq('session_id', sessionId)
      .eq('user_id', userId)
  ]);
  questions.sort((a, b) => (a.question_number ?? 0) - (b.question_number ?? 0));
  const respByQ = new Map((responses ?? []).map((r: any) => [r.question_id as string, r]));
  const submitted = !!participant.submitted_at;

  const rows = questions.map((q) => {
    const r: any = respByQ.get(q.id);
    let status: AttemptStatus | 'UNANSWERED';
    if (!r) status = submitted ? 'SKIPPED' : 'UNANSWERED';
    else status = statusOf(r.selected_answer, r.is_correct);
    return { q, r, status };
  });
  return { session, test, submitted, submitted_at: participant.submitted_at as string | null, rows };
}

function questionWithOutcome(row: { q: any; r: any; status: string }, full = true) {
  return {
    ...(full ? shapeQuestion(row.q) : compactQuestion(row.q)),
    status: row.status,
    selected_answer: row.r?.selected_answer ?? null,
    answered_at: row.r?.answered_at ?? null,
    time_spent_seconds: typeof row.r?.time_spent_seconds === 'number' ? row.r.time_spent_seconds : null
  };
}

function tally(rows: { status: string }[]) {
  const t = { correct: 0, wrong: 0, skipped: 0, unanswered: 0, unscored: 0 };
  for (const r of rows) {
    if (r.status === 'CORRECT') t.correct += 1;
    else if (r.status === 'WRONG') t.wrong += 1;
    else if (r.status === 'SKIPPED') t.skipped += 1;
    else if (r.status === 'UNANSWERED') t.unanswered += 1;
    else t.unscored += 1;
  }
  return t;
}

export async function getQbankSession(sb: SupabaseClient, userId: string, args: Record<string, unknown>) {
  const sessionId = optString(args.session_id, 'session_id');
  if (!sessionId) throw new ToolInputError('session_id is required');
  const d = await loadSessionDetail(sb, userId, sessionId);
  const prov = provenanceOf(d.test?.generated_from, d.session.intervention_id);
  const gf = (d.test?.generated_from ?? {}) as GeneratedFrom;
  return {
    session_id: d.session.id,
    test_id: d.session.test_id,
    title: d.test?.title ?? null,
    provenance: prov,
    purpose: purposeOf(prov, gf),
    chapter: gf.chapter ?? null,
    topic: gf.topic ?? null,
    pool: gf.pool ?? null,
    intervention_id: d.session.intervention_id ?? gf.intervention_id ?? null,
    session_status: d.session.status,
    submitted: d.submitted,
    submitted_at: d.submitted_at,
    totals: tally(d.rows),
    questions: d.rows.map((r) => questionWithOutcome(r))
  };
}

function purposeOf(prov: Provenance, gf: GeneratedFrom): string {
  if (prov === 'qbank_practice') return gf.pool === 'WRONG_SKIPPED' ? 'qbank_retry_of_misses' : 'coverage';
  if (prov === 'targeted_repair') return 'repair';
  if (prov === 'retention_check') return 'retention';
  if (prov === 'coaching_bank') return 'coaching_bank';
  return 'full_test';
}

// ── Tool: today's Q-bank ─────────────────────────────────────────────────────
export async function getTodayQbank(sb: SupabaseClient, userId: string, args: Record<string, unknown>) {
  const date = optString(args.date, 'date') ?? localDateString();
  if (!/^\d{4}-\d{2}-\d{2}$/.test(date)) throw new ToolInputError('date must be YYYY-MM-DD');
  const { start, end } = localDayBounds(date);

  const { data: tests, error } = await sb
    .from('tests')
    .select('id, title, generated_from, created_at')
    .eq('owner_user_id', userId)
    .eq('generated_from->>kind', 'qbank_practice')
    .gte('created_at', start)
    .lt('created_at', end)
    .order('created_at', { ascending: true });
  if (error) throw error;
  if (!tests || tests.length === 0) {
    return {
      date,
      sessions: [],
      note: 'No Q-bank practice session has been started for this date. Use checkmate_get_daily_target for what Testmate wants done today.'
    };
  }

  const testIds = tests.map((t: any) => t.id as string);
  const sessions = await selectIn(sb, 'test_sessions', 'id, test_id, status, server_start_time', 'test_id', testIds);
  const latestByTest = new Map<string, any>();
  for (const s of sessions.sort((a, b) => String(b.server_start_time).localeCompare(String(a.server_start_time)))) {
    if (!latestByTest.has(s.test_id)) latestByTest.set(s.test_id, s);
  }
  const sessionIds = Array.from(latestByTest.values()).map((s) => s.id as string);
  const [questions, responses, parts] = await Promise.all([
    selectIn(sb, 'questions', QUESTION_SELECT, 'test_id', testIds, { paged: true }),
    selectIn(sb, 'responses', 'session_id, question_id, selected_answer, is_correct, answered_at, time_spent_seconds', 'session_id', sessionIds, {
      paged: true,
      decorate: (q) => q.eq('user_id', userId)
    }),
    selectIn(sb, 'session_participants', 'session_id, submitted_at', 'session_id', sessionIds, {
      decorate: (q) => q.eq('user_id', userId)
    })
  ]);
  const submittedBySession = new Map(parts.map((p) => [p.session_id as string, !!p.submitted_at]));
  const respKey = (sid: string, qid: string) => `${sid}:${qid}`;
  const respMap = new Map(responses.map((r) => [respKey(r.session_id, r.question_id), r]));

  const out = tests.map((t: any) => {
    const s = latestByTest.get(t.id);
    const gf = (t.generated_from ?? {}) as GeneratedFrom;
    const qs = questions.filter((q) => q.test_id === t.id).sort((a, b) => (a.question_number ?? 0) - (b.question_number ?? 0));
    const submitted = s ? submittedBySession.get(s.id) === true : false;
    const rows = qs.map((q) => {
      const r = s ? respMap.get(respKey(s.id, q.id)) : undefined;
      const status = !r ? (submitted ? 'SKIPPED' : 'UNANSWERED') : statusOf(r.selected_answer, r.is_correct);
      return { q, r, status };
    });
    return {
      session_id: s?.id ?? null,
      test_id: t.id,
      title: t.title,
      created_at: t.created_at,
      chapter: gf.chapter ?? null,
      topic: gf.topic ?? null,
      purpose: purposeOf('qbank_practice', gf),
      pool: gf.pool ?? null,
      session_status: s?.status ?? null,
      submitted,
      totals: tally(rows),
      questions: rows.map((r) => questionWithOutcome(r))
    };
  });
  return { date, sessions: out };
}

// ── Tasks → questions → outcomes (repair + retention checks) ─────────────────
async function loadTaskTests(sb: SupabaseClient, userId: string, wanted: Provenance, days: number) {
  const { data: tests, error } = await sb
    .from('tests')
    .select('id, title, generated_from, created_at')
    .eq('owner_user_id', userId)
    .not('generated_from', 'is', null)
    .gte('created_at', sinceFromDays(days))
    .order('created_at', { ascending: false })
    .limit(300);
  if (error) throw error;
  return (tests ?? []).filter((t: any) => provenanceOf(t.generated_from) === wanted);
}

async function taskGroups(
  sb: SupabaseClient,
  userId: string,
  wanted: 'targeted_repair' | 'retention_check',
  args: Record<string, unknown>,
  extraReason?: (chapter: string | null, topic: string | null) => unknown
) {
  const days = clampInt(args.days, 60, 1, 365);
  const limit = clampInt(args.limit, 10, 1, 30);
  const chapter = optString(args.chapter, 'chapter');
  const onlyOpen = args.only_unresolved === true;

  let tests = await loadTaskTests(sb, userId, wanted, days);
  if (chapter) tests = tests.filter((t: any) => (t.generated_from?.chapter ?? null) === chapter);
  tests = tests.slice(0, limit * 2);
  if (tests.length === 0) return { window_days: days, tasks: [] as any[] };

  const testIds = tests.map((t: any) => t.id as string);
  const sessions = await selectIn(sb, 'test_sessions', 'id, test_id, status, intervention_id, server_start_time', 'test_id', testIds);
  const latestByTest = new Map<string, any>();
  for (const s of sessions.sort((a, b) => String(b.server_start_time).localeCompare(String(a.server_start_time)))) {
    if (!latestByTest.has(s.test_id)) latestByTest.set(s.test_id, s);
  }
  const sessionIds = Array.from(latestByTest.values()).map((s) => s.id as string);
  const [questions, responses, parts] = await Promise.all([
    selectIn(sb, 'questions', QUESTION_SELECT, 'test_id', testIds, { paged: true }),
    selectIn(sb, 'responses', 'session_id, question_id, selected_answer, is_correct, answered_at, time_spent_seconds', 'session_id', sessionIds, {
      paged: true,
      decorate: (q) => q.eq('user_id', userId)
    }),
    selectIn(sb, 'session_participants', 'session_id, submitted_at', 'session_id', sessionIds, {
      decorate: (q) => q.eq('user_id', userId)
    })
  ]);
  const submittedBySession = new Map(parts.map((p) => [p.session_id as string, !!p.submitted_at]));
  const respMap = new Map(responses.map((r) => [`${r.session_id}:${r.question_id}`, r]));

  // Prior history per canonical question, so each question can say "you'd already
  // missed this N times before this task was built" (the `reason` the plan asks for).
  const allAttempts = await loadAttempts(sb, userId, { sinceIso: sinceFromDays(Math.max(days, 180)), maxRows: 4000 });
  const historyByCanonical = groupByCanonical(allAttempts);

  const tasks = tests.map((t: any) => {
    const s = latestByTest.get(t.id);
    const gf = (t.generated_from ?? {}) as GeneratedFrom;
    const createdMs = new Date(t.created_at).getTime();
    const submitted = s ? submittedBySession.get(s.id) === true : false;
    const qs = questions.filter((q) => q.test_id === t.id).sort((a, b) => (a.question_number ?? 0) - (b.question_number ?? 0));
    const rows = qs.map((q) => {
      const r = s ? respMap.get(`${s.id}:${q.id}`) : undefined;
      const status = !r ? (submitted ? 'SKIPPED' : 'UNANSWERED') : statusOf(r.selected_answer, r.is_correct);
      const prior = (historyByCanonical.get(canonicalQuestionId(q)) ?? []).filter(
        (a) => a.answered_at && new Date(a.answered_at).getTime() < createdMs && a.test_id !== t.id
      );
      const priorHist = summarizeHistory(prior);
      return {
        ...questionWithOutcome({ q, r, status }),
        purpose: wanted === 'targeted_repair' ? 'repair' : 'retention',
        reason: {
          concept: q.topic ?? q.chapter ?? null,
          selection_pool: gf.pool ?? null,
          previous_attempts: priorHist.attempts,
          previous_wrong: priorHist.wrong,
          previous_skipped: priorHist.skipped,
          last_correct_before: priorHist.last_correct_at
        }
      };
    });
    const totals = tally(rows);
    return {
      intervention_id: gf.intervention_id ?? s?.intervention_id ?? null,
      test_id: t.id,
      session_id: s?.id ?? null,
      title: t.title,
      created_at: t.created_at,
      chapter: gf.chapter ?? null,
      topic: gf.topic ?? null,
      selection_pool: gf.pool ?? null,
      submitted,
      totals,
      unresolved: totals.wrong + totals.skipped + totals.unanswered,
      checkmate_context: extraReason ? extraReason(gf.chapter ?? null, gf.topic ?? null) : undefined,
      questions: rows
    };
  });

  const filtered = onlyOpen ? tasks.filter((t) => t.unresolved > 0) : tasks;
  return { window_days: days, tasks: filtered.slice(0, limit) };
}

export async function getRetentionQuestions(
  sb: SupabaseClient,
  userId: string,
  args: Record<string, unknown>,
  extraReason?: (chapter: string | null, topic: string | null) => unknown
) {
  const mode = (optString(args.mode, 'mode') ?? 'both').toLowerCase();
  if (!['due', 'checks', 'both'].includes(mode)) throw new ToolInputError('mode must be one of: due, checks, both');
  const limit = clampInt(args.limit, 20, 1, 50);
  const chapter = optString(args.chapter, 'chapter');

  const result: Record<string, unknown> = {};

  if (mode === 'due' || mode === 'both') {
    let q: any = sb
      .from('question_mastery')
      .select('canonical_question_id, chapter, topic, last_result, last_wrong_source, correct_streak, retention_box, next_review_at, last_attempted_at, times_attempted, times_correct')
      .eq('user_id', userId)
      .lte('next_review_at', new Date().toISOString())
      .order('next_review_at', { ascending: true })
      .limit(limit);
    if (chapter) q = q.eq('chapter', chapter);
    const { data: due, error } = await q;
    if (error) throw error;
    const qs = await selectIn(sb, 'questions', QUESTION_SELECT, 'id', (due ?? []).map((d: any) => d.canonical_question_id as string));
    const qById = new Map(qs.map((x) => [x.id as string, x]));
    const nowMs = Date.now();
    result.due = (due ?? [])
      .map((d: any) => {
        const q0 = qById.get(d.canonical_question_id);
        if (!q0) return null;
        const weak = d.last_result === 'wrong' || (d.correct_streak ?? 0) === 0;
        return {
          ...shapeQuestion(q0),
          purpose: 'retention',
          // The distinction that matters: "weak at this" vs "strong, but due for spaced recall".
          strength: weak ? 'weak' : 'strong_but_due',
          retention: {
            last_result: d.last_result,
            last_attempted_at: d.last_attempted_at,
            next_review_at: d.next_review_at,
            overdue_days: Math.max(0, Math.floor((nowMs - new Date(d.next_review_at).getTime()) / 86_400_000)),
            retention_box: d.retention_box,
            correct_streak: d.correct_streak,
            times_attempted: d.times_attempted,
            times_correct: d.times_correct
          }
        };
      })
      .filter(Boolean);
  }

  if (mode === 'checks' || mode === 'both') {
    const checks = await taskGroups(sb, userId, 'retention_check', { ...args, limit: Math.min(limit, 10) }, extraReason);
    result.checks = checks.tasks;
  }

  return {
    ...result,
    note: "'due' = Testmate's spaced-repetition queue (Leitner boxes). 'checks' = retention-check sessions Checkmate scheduled, with per-question outcomes."
  };
}

// ── Tool: one question's whole story ─────────────────────────────────────────
async function resolveOwnedQuestion(sb: SupabaseClient, userId: string, questionId: string) {
  const { data: q, error } = await sb.from('questions').select(QUESTION_SELECT).eq('id', questionId).maybeSingle();
  if (error) throw error;
  if (!q) throw new ToolInputError('question_id not found');
  // A copy living inside another student's private repair/Q-bank test must not be readable.
  const { data: test } = await sb.from('tests').select('owner_user_id').eq('id', q.test_id).maybeSingle();
  if (test?.owner_user_id && test.owner_user_id !== userId) throw new ToolInputError('question_id not found');
  const canonicalId = canonicalQuestionId(q);
  const canonical =
    canonicalId === q.id
      ? q
      : (await sb.from('questions').select(QUESTION_SELECT).eq('id', canonicalId).maybeSingle()).data ?? q;
  return { requested: q, canonical, canonicalId };
}

async function loadQuestionAttempts(sb: SupabaseClient, userId: string, canonicalId: string): Promise<Attempt[]> {
  const { data: copies, error } = await sb
    .from('questions')
    .select('id')
    .or(`id.eq.${canonicalId},origin_question_id.eq.${canonicalId}`);
  if (error) throw error;
  const ids = (copies ?? []).map((c: any) => c.id as string);
  const rows = await selectIn(
    sb,
    'responses',
    'id, session_id, question_id, selected_answer, is_correct, answered_at, time_spent_seconds',
    'question_id',
    ids,
    { decorate: (q) => q.eq('user_id', userId).order('answered_at', { ascending: false }) }
  );
  return (await enrichResponses(sb, rows)).sort(
    (a, b) => (b.answered_at ? new Date(b.answered_at).getTime() : 0) - (a.answered_at ? new Date(a.answered_at).getTime() : 0)
  );
}

async function loadMastery(sb: SupabaseClient, userId: string, canonicalId: string) {
  const { data } = await sb
    .from('question_mastery')
    .select('last_result, correct_streak, retention_box, next_review_at, last_attempted_at, times_attempted, times_correct, last_wrong_source')
    .eq('user_id', userId)
    .eq('canonical_question_id', canonicalId)
    .maybeSingle();
  return data ?? null;
}

export async function getQuestionHistory(sb: SupabaseClient, userId: string, args: Record<string, unknown>) {
  const questionId = optString(args.question_id, 'question_id');

  const { canonical, canonicalId } = await resolveOwnedQuestion(sb, userId, questionId);

    question: shapeQuestion(canonical),

  };
}

export async function getQuestionContext(
  sb: SupabaseClient,
  userId: string,
  args: Record<string, unknown>,
  conceptLookup?: (chapter: string | null, topic: string | null) => unknown
) {
  const questionId = optString(args.question_id, 'question_id');
  if (!questionId) throw new ToolInputError('question_id is required');
  const { canonical, canonicalId } = await resolveOwnedQuestion(sb, userId, questionId);
  const [attempts, mastery] = await Promise.all([loadQuestionAttempts(sb, userId, canonicalId), loadMastery(sb, userId, canonicalId)]);

  const bySource: Record<string, ReturnType<typeof summarizeHistory>> = {};
  for (const p of ['full_test', 'qbank_practice', 'targeted_repair', 'retention_check'] as Provenance[]) {
    const g = attempts.filter((a) => a.provenance === p);
    if (g.length) bySource[p] = summarizeHistory(g);
  }
  const interventions = Array.from(
    new Map(
      attempts
        .filter((a) => a.intervention_id)
        .map((a) => [`${a.intervention_id}`, { intervention_id: a.intervention_id, provenance: a.provenance, status: a.status, answered_at: a.answered_at }])
    ).values()
  );
  const latest = attempts[0] ?? null;
  return {
    question: shapeQuestion(canonical),
    current_state: latest ? { status: latest.status, as_of: latest.answered_at, provenance: latest.provenance } : null,
    history: summarizeHistory(attempts),
    history_by_source: bySource,
    appeared_in: Array.from(new Set(attempts.map((a) => a.test_title).filter(Boolean))).slice(0, 15),
    repair_and_retention_tasks: interventions.slice(0, 15),
    retention_state: mastery,
    checkmate_concept: conceptLookup ? conceptLookup(canonical.chapter ?? null, canonical.topic ?? null) ?? null : null,
    recent_attempts: attempts.slice(0, 10).map((a) => shapeAttempt(a))
  };
}

// ── Tool: a test / session breakdown ─────────────────────────────────────────
export async function getTestBreakdown(sb: SupabaseClient, userId: string, args: Record<string, unknown>) {
  let sessionId = optString(args.session_id, 'session_id');
  const testId = optString(args.test_id, 'test_id');
  if (!sessionId && !testId) throw new ToolInputError('Provide session_id or test_id');

  if (!sessionId && testId) {
    const sessions = await selectIn(sb, 'test_sessions', 'id, server_start_time', 'test_id', [testId]);
    const mine = await selectIn(sb, 'session_participants', 'session_id, submitted_at', 'session_id', sessions.map((s) => s.id as string), {
      decorate: (q) => q.eq('user_id', userId)
    });
    const mineIds = new Set(mine.map((m) => m.session_id as string));
    const candidate = sessions
      .filter((s) => mineIds.has(s.id))
      .sort((a, b) => String(b.server_start_time).localeCompare(String(a.server_start_time)))[0];
    if (!candidate) throw new ToolInputError('No session of yours exists for that test_id');
    sessionId = candidate.id as string;
  }

  const d = await loadSessionDetail(sb, userId, sessionId!);
  const { data: summary } = await sb
    .from('result_summaries')
    .select('*')
    .eq('session_id', sessionId!)
    .eq('user_id', userId)
    .maybeSingle();

  const withText = args.include_question_text === true;
  const chapterStats = new Map<string, { correct: number; wrong: number; skipped: number; time: number; timed: number }>();
  for (const r of d.rows) {
    const key = r.q.chapter ?? 'Uncategorized';
    const s = chapterStats.get(key) ?? { correct: 0, wrong: 0, skipped: 0, time: 0, timed: 0 };
    if (r.status === 'CORRECT') s.correct += 1;
    else if (r.status === 'WRONG') s.wrong += 1;
    else s.skipped += 1;
    if (typeof r.r?.time_spent_seconds === 'number') {
      s.time += r.r.time_spent_seconds;
      s.timed += 1;
    }
    chapterStats.set(key, s);
  }
  const prov = provenanceOf(d.test?.generated_from, d.session.intervention_id);

  return {
    session_id: d.session.id,
    test_id: d.session.test_id,
    title: d.test?.title ?? null,
    provenance: prov,
    submitted: d.submitted,
    submitted_at: d.submitted_at,
    result: summary
      ? {
          score: summary.score,
          total_marks: summary.total_marks,
          accuracy_pct: summary.accuracy_pct,
          attempted: summary.attempted_count,
          correct: summary.correct_count,
          incorrect: summary.incorrect_count,
          skipped: summary.skipped_count,
          avg_time_per_question: summary.avg_time_per_question,
          rank_in_session: summary.rank_in_session
        }
      : null,
    by_chapter: Array.from(chapterStats.entries())
      .map(([chapter, s]) => ({
        chapter,
        correct: s.correct,
        wrong: s.wrong,
        skipped: s.skipped,
        accuracy_pct: s.correct + s.wrong > 0 ? Math.round((s.correct / (s.correct + s.wrong)) * 1000) / 10 : null,
        avg_time_seconds: s.timed ? Math.round(s.time / s.timed) : null
      }))
      .sort((a, b) => b.wrong + b.skipped - (a.wrong + a.skipped)),
    questions: d.rows.map((r) => {
      const base = { ...questionWithOutcome(r, false), correct_answer: r.q.correct_answer ?? null };
      // Full text/options/explanation only for the questions worth diagnosing.
      return withText && r.status !== 'CORRECT' ? { ...questionWithOutcome(r, true) } : base;
    })
  };
}

  return typeof v === 'string' && v.trim() ? v.trim() : undefined;

const eq = (a?: string | null, b?: string | null) => !!a && !!b && a.trim().toLowerCase() === b.trim().toLowerCase();

function requireModel(m: any) {
  if (!m) {

export function conceptLookupFor(model: any | null) {
  if (!model) return undefined;
  return (chapter: string | null, topic: string | null) => {
    const concepts = (model.concepts as any[]).filter((c) => eq(c.chapter, chapter) && (!topic || !c.topic || eq(c.topic, topic)));
    if (concepts.length === 0) return null;
    const errors = (model.unresolved_errors as any[]).filter((e) => eq(e.chapter, chapter) && (!topic || !e.topic || eq(e.topic, topic)));
    return {
      concepts: concepts.slice(0, 5).map((c) => ({
        subject: c.subject,
        chapter: c.chapter,
        topic: c.topic,
        mastery: c.mastery,
        retention_decision: c.retention_decision,
        forgetting_risk: c.forgetting_risk,
        weak: c.weak
      })),
      unresolved_error_patterns: errors.slice(0, 5).map((e) => ({ error_type: e.error_type, occurrences: e.occurrences, last_seen_at: e.last_seen_at }))
    };
  };
}

export function getMastery(model: any, extras: { questionLevel: unknown }, args: Record<string, unknown>) {
  const m = requireModel(model);
  const subject = optStr(args.subject);
  const chapter = optStr(args.chapter);
  const topic = optStr(args.topic);
  const limit = clampInt(args.limit, 40, 1, 150);
  let concepts = (m.concepts as any[]).filter(
    (c) => (!subject || eq(c.subject, subject)) && (!chapter || eq(c.chapter, chapter)) && (!topic || eq(c.topic, topic))
  );
  concepts = concepts.sort((a, b) => (a.mastery ?? 1) - (b.mastery ?? 1));
  const bySubject = new Map<string, { n: number; sum: number; weak: number }>();
  for (const c of concepts) {
    const key = c.subject ?? 'Unclassified';
    const s = bySubject.get(key) ?? { n: 0, sum: 0, weak: 0 };
    s.n += 1;
    s.sum += c.mastery ?? 0;
    if (c.weak) s.weak += 1;
    bySubject.set(key, s);
  }
  return {
    source: 'checkmate_student_model',
    freshness: m._freshness,
    overall: m.overall,
    by_subject: Array.from(bySubject.entries()).map(([name, s]) => ({
      subject: name,
      concepts: s.n,
      average_mastery: s.n ? round(s.sum / s.n) : null,
      weak_concepts: s.weak
    })),
    matching: concepts.length,
    concepts: concepts.slice(0, limit).map(({ concept_id, ...rest }) => ({ concept_id, ...rest })),
    question_level: extras.questionLevel
  };
}

    .from('session_participants')
    .select('session_id, submitted_at')
    .eq('user_id', userId)
    .not('submitted_at', 'is', null)
    .order('submitted_at', { ascending: false })
    .limit(limit * 4);
  if (error) throw error;
  const sessionIds = (parts ?? []).map((p: any) => p.session_id as string);
  if (sessionIds.length === 0) return { tests: [] };

  const [{ data: sessions }, { data: summaries }] = await Promise.all([
    sb.from('test_sessions').select('id, test_id, intervention_id').in('id', sessionIds.slice(0, 150)),
    sb.from('result_summaries').select('*').eq('user_id', userId).in('session_id', sessionIds.slice(0, 150))
  ]);
  const testIds = Array.from(new Set((sessions ?? []).map((s: any) => s.test_id as string)));
  const { data: tests } = testIds.length
    ? await sb.from('tests').select('id, title, generated_from').in('id', testIds.slice(0, 150))
    : { data: [] as any[] };
  const testById = new Map((tests ?? []).map((t: any) => [t.id as string, t]));
  const sessionById = new Map((sessions ?? []).map((s: any) => [s.id as string, s]));
  const summaryBySession = new Map((summaries ?? []).map((r: any) => [r.session_id as string, r]));

  const out: any[] = [];
  for (const p of parts ?? []) {
    const s: any = sessionById.get(p.session_id);
    if (!s) continue;
    const t: any = testById.get(s.test_id);
    const prov = provenanceOf(t?.generated_from, s.intervention_id);
    if (!includePractice && prov !== 'full_test') continue;
    const r: any = summaryBySession.get(p.session_id);
    out.push({
      session_id: p.session_id,
      test_id: s.test_id,
      title: t?.title ?? null,
      provenance: prov,
      submitted_at: p.submitted_at,
      score: r?.score ?? null,
      total_marks: r?.total_marks ?? null,
      accuracy_pct: r?.accuracy_pct ?? null,
      attempted: r?.attempted_count ?? null,
      correct: r?.correct_count ?? null,
      incorrect: r?.incorrect_count ?? null,
      skipped: r?.skipped_count ?? null,
      avg_time_per_question: r?.avg_time_per_question ?? null,
      weakest_chapters: ((r?.weak_chapters as any[]) ?? []).slice(0, 5),
      weakest_topics: ((r?.weak_topics as any[]) ?? []).slice(0, 5)
    });
    if (out.length >= limit) break;
  }
  return {
    note: includePractice ? undefined : 'Full tests only. Pass include_practice=true to include Q-bank / repair / retention sessions.',
    tests: out
  };
}

export function dailyTargetView(b: DailyTargetBreakdown) {
  if (!b.configured) {
    return {
      configured: false,
      note: 'No exam date configured in Testmate yet (Admin → Settings → Exam & study targets), so there is no daily target.'
    };
  }
  const why: string[] = [];
  if (b.days_left !== null) {
    why.push(`${b.days_left} day(s) left to the ${b.deadline_source === 'syllabus_deadline' ? 'syllabus deadline' : 'exam date'}; ${b.remaining_questions} of ${b.total_questions} questions remain.`);
  }
  if (b.ft_pressure?.applied) {
    why.push(`Full-test pressure raised the coverage target from ${b.ft_pressure.base_target} to ${b.ft_pressure.boosted_target} (binding test: ${b.ft_pressure.binding_test}).`);
  } else if (b.ft_pressure) {
    why.push('Full-test deadlines did not demand more than the global pace, so no boost was applied.');
  }
  if (b.capacity_capped) why.push('The target was capped by your daily capacity setting.');
  why.push(`Today: ${b.today_target.coverage} coverage + ${b.today_target.repair} repair + ${b.today_target.retention} retention = ${b.today_target.total}; ${b.today_completed} done so far.`);
  return {
    configured: true,
    exam_date: b.exam_date,
    syllabus_deadline: b.syllabus_deadline,
    deadline_source: b.deadline_source,
    days_left: b.days_left,
    total_questions: b.total_questions,
    completed_questions: b.completed_questions,
    remaining_questions: b.remaining_questions,
    today_target: b.today_target,
    today_completed: b.today_completed,
    capacity_capped: b.capacity_capped,
    subject_breakdown: b.subject_breakdown,
    ft_pressure: b.ft_pressure,
    why,
    computed_at: b.computed_at
  };
}

// ── Question-level mastery + retention + Q-bank (Testmate side) ──────────────
async function fetchMasteryRows(sb: SupabaseClient, userId: string): Promise<any[]> {
  const out: any[] = [];
  for (let from = 0; ; from += 1000) {
    const { data, error } = await sb
      .from('question_mastery')
      .select('canonical_question_id, chapter, topic, last_result, last_wrong_source, correct_streak, retention_box, next_review_at, times_attempted, times_correct')
      .eq('user_id', userId)
      .order('canonical_question_id', { ascending: true })
      .range(from, from + 999);
    if (error) throw error;
    out.push(...(data ?? []));
    if ((data ?? []).length < 1000) break;
  }
  return out;
}

export async function questionLevelMastery(sb: SupabaseClient, userId: string, args: Record<string, unknown>) {
  const subject = optStr(args.subject);
  const chapter = optStr(args.chapter);
  const subjectByChapter = subject ? await getSubjectByChapter(sb) : new Map<string, string>();
  const rows = (await fetchMasteryRows(sb, userId)).filter(
    (r) => (!chapter || eq(r.chapter, chapter)) && (!subject || eq(subjectByChapter.get(r.chapter ?? ''), subject))
  );
  const now = Date.now();
  const byChapter = new Map<string, any>();
  for (const r of rows) {
    const key = r.chapter ?? 'Uncategorized';
    const s = byChapter.get(key) ?? { tracked: 0, correct_now: 0, wrong_now: 0, box: 0, due: 0, att: 0, cor: 0 };
    s.tracked += 1;
    if (r.last_result === 'correct') s.correct_now += 1;
    if (r.last_result === 'wrong') s.wrong_now += 1;
    s.box += r.retention_box ?? 0;
    if (r.next_review_at && new Date(r.next_review_at).getTime() <= now) s.due += 1;
    s.att += r.times_attempted ?? 0;
    s.cor += r.times_correct ?? 0;
    byChapter.set(key, s);
  }
  return {
    source: 'testmate_question_mastery',
    questions_tracked: rows.length,
    chapters: Array.from(byChapter.entries())
      .map(([name, s]) => ({
        chapter: name,
        questions_tracked: s.tracked,
        currently_correct: s.correct_now,
        currently_wrong: s.wrong_now,
        lifetime_accuracy_pct: s.att ? round((s.cor / s.att) * 100, 1) : null,
        avg_retention_box: s.tracked ? round(s.box / s.tracked, 2) : null,
        due_for_review: s.due
      }))
      .sort((a, b) => b.currently_wrong - a.currently_wrong)
      .slice(0, 60)
  };
}

export async function getRetentionDue(sb: SupabaseClient, userId: string, model: any | null, args: Record<string, unknown>) {
  const limit = clampInt(args.limit, 10, 1, 50);
  const chapter = optStr(args.chapter);
  const rows = (await fetchMasteryRows(sb, userId)).filter((r) => !chapter || eq(r.chapter, chapter));
  const now = Date.now();
  const due = rows
    .filter((r) => r.next_review_at && new Date(r.next_review_at).getTime() <= now)
    .map((r) => ({ ...r, overdue_days: Math.floor((now - new Date(r.next_review_at).getTime()) / 86_400_000) }));
  const weak = due.filter((r) => r.last_result === 'wrong' || (r.correct_streak ?? 0) === 0);
  const byChapter = new Map<string, number>();
  for (const r of due) byChapter.set(r.chapter ?? 'Uncategorized', (byChapter.get(r.chapter ?? 'Uncategorized') ?? 0) + 1);

  return {
    question_level: {
      source: 'testmate_spaced_repetition',
      due_now: due.length,
      overdue_over_3_days: due.filter((r) => r.overdue_days > 3).length,
      weak_and_due: weak.length,
      strong_but_due: due.length - weak.length,
      by_chapter: Array.from(byChapter.entries())
        .map(([name, n]) => ({ chapter: name, due: n }))
        .sort((a, b) => b.due - a.due)
        .slice(0, 15),
      next_step: 'Call checkmate_get_retention_questions to see the actual questions.'
    },
    concept_level: model
      ? {
          source: 'checkmate_student_model',
          freshness: model._freshness,
          needs_review: (model.concepts as any[])
            .filter((c) => c.retention_decision === 'REVIEW' && (!chapter || eq(c.chapter, chapter)))
            .sort((a, b) => (b.forgetting_risk ?? 0) - (a.forgetting_risk ?? 0))
            .slice(0, limit)
            .map((c) => ({ subject: c.subject, chapter: c.chapter, topic: c.topic, mastery: c.mastery, forgetting_risk: c.forgetting_risk, last_seen_at: c.last_seen_at }))
        }
      : { available: false, note: "Checkmate hasn't synced its student model yet." }
  };
}

export async function getQbankProgress(sb: SupabaseClient, userId: string, args: Record<string, unknown>) {
  const subject = optStr(args.subject);
  const chapter = optStr(args.chapter);
  const subjectByChapter = await getSubjectByChapter(sb);

  // Canonical Q-bank supply = tag_source 'coaching_module' (see app/api/qbank/practice's POOL SOURCE note).
  const bank: { id: string; chapter: string | null }[] = [];
  for (let from = 0; ; from += 1000) {
    const { data, error } = await sb
      .from('questions')
      .select('id, chapter')
      .eq('tag_source', 'coaching_module')
      .order('id', { ascending: true })
      .range(from, from + 999);
    if (error) throw error;
    bank.push(...((data ?? []) as any[]));
    if ((data ?? []).length < 1000) break;
  }
  const mastery = await fetchMasteryRows(sb, userId);
  const masteryByCanonical = new Map(mastery.map((m) => [m.canonical_question_id as string, m]));
  const wrongSkipped = new Map((await getQbankWrongSkippedByChapter(sb, userId)).map((r) => [r.chapter, r]));

  const byChapter = new Map<string, { total: number; attempted: number; correct_now: number }>();
  for (const q of bank) {
    const key = q.chapter ?? 'Uncategorized';
    if (chapter && !eq(key, chapter)) continue;
    if (subject && !eq(subjectByChapter.get(key), subject)) continue;
    const s = byChapter.get(key) ?? { total: 0, attempted: 0, correct_now: 0 };
    s.total += 1;
    const m = masteryByCanonical.get(q.id);
    if (m) {
      s.attempted += 1;
      if (m.last_result === 'correct') s.correct_now += 1;
    }
    byChapter.set(key, s);
  }
  const chapters = Array.from(byChapter.entries())
    .map(([name, s]) => ({
      chapter: name,
      subject: subjectByChapter.get(name) ?? null,
      bank_questions: s.total,
      attempted: s.attempted,
      remaining_unattempted: s.total - s.attempted,
      currently_correct: s.correct_now,
      qbank_wrong: wrongSkipped.get(name)?.wrong ?? 0,
      qbank_skipped: wrongSkipped.get(name)?.skipped ?? 0
    }))
    .sort((a, b) => b.remaining_unattempted - a.remaining_unattempted);
  const totals = chapters.reduce(
    (t, c) => ({
      bank_questions: t.bank_questions + c.bank_questions,
      attempted: t.attempted + c.attempted,
      qbank_wrong: t.qbank_wrong + c.qbank_wrong,
      qbank_skipped: t.qbank_skipped + c.qbank_skipped
    }),
    { bank_questions: 0, attempted: 0, qbank_wrong: 0, qbank_skipped: 0 }
  );
  return {
    note: 'Wrong/skipped here are Q-bank practice only (never mixed with full tests or repairs).',
    totals: { ...totals, remaining_unattempted: totals.bank_questions - totals.attempted },
    chapters: chapters.slice(0, 40)
  };
}

// ── Snapshot + composite context ─────────────────────────────────────────────
async function attemptsSummary(sb: SupabaseClient, userId: string, days: number) {
  const attempts: Attempt[] = await loadAttempts(sb, userId, { sinceIso: new Date(Date.now() - days * 86_400_000).toISOString(), maxRows: 2500 });
  const subjectByChapter = await getSubjectByChapter(sb);
  const tally = () => ({ correct: 0, wrong: 0, skipped: 0 });
  const overall = tally();
  const per = new Map<string, ReturnType<typeof tally>>();
  for (const a of attempts) {
    const key = a.question.chapter ? subjectByChapter.get(a.question.chapter) ?? 'Unclassified' : 'Unclassified';
    const s = per.get(key) ?? tally();
    const bucket = a.status === 'CORRECT' ? 'correct' : a.status === 'WRONG' ? 'wrong' : a.status === 'SKIPPED' ? 'skipped' : null;
    if (bucket) {
      s[bucket] += 1;
      overall[bucket] += 1;
    }
    per.set(key, s);
  }
  const acc = (t: { correct: number; wrong: number }) => (t.correct + t.wrong ? round((t.correct / (t.correct + t.wrong)) * 100, 1) : null);
  return {
    attempts,
    summary: {
      window_days: days,
      overall: { ...overall, accuracy_pct: acc(overall) },
      by_subject: Array.from(per.entries()).map(([subject, t]) => ({ subject, ...t, accuracy_pct: acc(t) }))
    }
  };
}

function stillWrongTop(attempts: Attempt[], n: number) {
  const groups = groupByCanonical(attempts);
  const out: any[] = [];
  for (const [id, g] of Array.from(groups.entries())) {
    if (g[0].status !== 'WRONG') continue;
    const h = summarizeHistory(g);
    out.push({ question_id: g[0].question.id, canonical_question_id: id, chapter: g[0].question.chapter ?? null, topic: g[0].question.topic ?? null, times_wrong: h.wrong, last_wrong_at: h.last_wrong_at });
  }
  return out.sort((a, b) => b.times_wrong - a.times_wrong).slice(0, n);
}

export async function getStudentSnapshot(sb: SupabaseClient, userId: string, model: any | null) {
  const [recent, target, tests] = await Promise.all([
    attemptsSummary(sb, userId, 14),
    loadDailyTarget(userId).catch(() => null),
    getRecentTests(sb, userId, { limit: 3 }).catch(() => ({ tests: [] as any[] }))
  ]);
  const retention = await getRetentionDue(sb, userId, model, { limit: 3 });
  const bySubject = new Map<string, { n: number; sum: number; weak: number }>();
  for (const c of (model?.concepts as any[]) ?? []) {
    const k = c.subject ?? 'Unclassified';
    const s = bySubject.get(k) ?? { n: 0, sum: 0, weak: 0 };
    s.n += 1;
    s.sum += c.mastery ?? 0;
    if (c.weak) s.weak += 1;
    bySubject.set(k, s);
  }
  const view = target ? dailyTargetView(target) : null;
  return {
    exam: model?.exam ?? null,
    checkmate: model
      ? {
          synced: true,
          freshness: model._freshness,
          overall: model.overall,
          mastery_by_subject: Array.from(bySubject.entries()).map(([subject, s]) => ({ subject, concepts: s.n, average_mastery: s.n ? round(s.sum / s.n) : null, weak_concepts: s.weak }))
        }
      : { synced: false, note: "Checkmate hasn't synced its student model yet — concept-level mastery and error patterns are unavailable until it does." },
    recent_accuracy: recent.summary,
    recent_full_tests: tests.tests,
    still_wrong_questions: stillWrongTop(recent.attempts, 5),
    retention: { question_level: (retention as any).question_level, concepts_needing_review: (retention as any).concept_level?.needs_review?.length ?? null },
    schedule: view && (view as any).configured
      ? {
          days_left: (view as any).days_left,
          deadline_source: (view as any).deadline_source,
          today_target: (view as any).today_target,
          today_completed: (view as any).today_completed,
          next_test: target!.upcoming_tests[0] ?? null
        }
      : view
  };
}

const PURPOSES = ['diagnose', 'plan', 'review_test', 'prepare_for_test', 'analyze_performance', 'repair_weakness', 'explain_progress'] as const;
export const CONTEXT_PURPOSES = PURPOSES as unknown as string[];

/**
 * The composite read tool. The gateway gathers what a purpose needs so the external model
 * doesn't have to orchestrate 15 calls — and every section degrades independently
 * (`{ unavailable: reason }`) instead of failing the whole context.
 */
export async function getContext(
  sb: SupabaseClient,
  userId: string,
  model: any | null,
  args: Record<string, unknown>,
  deps: { testBreakdown: (a: Record<string, unknown>) => Promise<unknown>; repairQuestions: (a: Record<string, unknown>) => Promise<unknown>; wrongQuestions: (a: Record<string, unknown>) => Promise<unknown> }
) {
  const purpose = optStr(args.purpose);
  if (!purpose || !CONTEXT_PURPOSES.includes(purpose)) {
    throw new ToolInputError(`purpose must be one of: ${CONTEXT_PURPOSES.join(', ')}`);
  }
  const chapter = optStr(args.chapter);
  const subject = optStr(args.subject);

  const section = async (fn: () => unknown | Promise<unknown>) => {
    try {
      return await fn();
    } catch (err: any) {
      return { unavailable: err?.message ?? 'failed' };
    }
  };

  let target: DailyTargetBreakdown | null = null;
  const getTarget = async () => (target ??= await loadDailyTarget(userId));

  const sections: Record<string, unknown> = { purpose };

  switch (purpose) {
    case 'diagnose': {
      sections.snapshot = await section(() => getStudentSnapshot(sb, userId, model));
      sections.weak_concepts = await section(() => getWeakConcepts(model, { subject, limit: 8 }));
      sections.error_patterns = await section(() => getErrorPatterns(model, { subject, chapter, limit: 8 }));
      sections.questions_you_keep_missing = await section(() => deps.wrongQuestions({ subject, chapter, days: 60, min_times: 2, limit: 8 }));
      sections.what_checkmate_already_tried = await section(() => getInterventionHistory(sb, userId, { days: 30, limit: 8 }));
      break;
    }
    case 'plan': {
      sections.today_plan = await section(() => getTodayPlan(sb, userId));
      sections.daily_target = await section(async () => dailyTargetView(await getTarget()));
      sections.upcoming_tests = await section(async () => upcomingTestsView(await getTarget(), { limit: 3 }));
      sections.retention_due = await section(() => getRetentionDue(sb, userId, model, { limit: 5 }));
      sections.open_repairs = await section(() => deps.repairQuestions({ only_unresolved: true, limit: 5 }));
      break;
    }
    case 'review_test': {
      const args2 = { session_id: args.session_id, test_id: args.test_id };
      if (!args2.session_id && !args2.test_id) {
        const recent: any = await getRecentTests(sb, userId, { limit: 1 });
        const first = recent.tests?.[0];
        if (!first) throw new ToolDataUnavailable('No submitted full test found to review.');
        args2.session_id = first.session_id;
      }
      sections.test = await section(() => deps.testBreakdown({ ...args2, include_question_text: true }));
      sections.recent_tests = await section(() => getRecentTests(sb, userId, { limit: 4 }));
      break;
    }
    case 'prepare_for_test': {
      sections.upcoming_tests = await section(async () => upcomingTestsView(await getTarget(), { limit: 3 }));
      sections.coverage_gaps = await section(async () => coverageGapsView(await getTarget(), { limit: 10, subject }));
      sections.weak_concepts = await section(() => getWeakConcepts(model, { subject, limit: 8 }));
      sections.retention_due = await section(() => getRetentionDue(sb, userId, model, { limit: 5 }));
      sections.questions_you_keep_missing = await section(() => deps.wrongQuestions({ subject, chapter, days: 90, limit: 8 }));
      sections.daily_target = await section(async () => dailyTargetView(await getTarget()));
      break;
    }
    case 'analyze_performance': {
      sections.recent_tests = await section(() => getRecentTests(sb, userId, { limit: 6 }));
      sections.accuracy = await section(async () => (await attemptsSummary(sb, userId, 30)).summary);
      sections.question_level_mastery = await section(() => questionLevelMastery(sb, userId, { subject, chapter }));
      sections.error_patterns = await section(() => getErrorPatterns(model, { subject, chapter, limit: 8 }));
      break;
    }
    case 'repair_weakness': {
      sections.weak_concepts = await section(() => getWeakConcepts(model, { subject, limit: 6 }));
      sections.repair_tasks = await section(() => deps.repairQuestions({ chapter, limit: 5 }));
      sections.wrong_questions = await section(() => deps.wrongQuestions({ subject, chapter, days: 90, limit: 10 }));
      sections.error_patterns = await section(() => getErrorPatterns(model, { subject, chapter, limit: 8 }));
      sections.what_checkmate_already_tried = await section(() => getInterventionHistory(sb, userId, { days: 45, limit: 8 }));
      break;
    }
    case 'explain_progress': {
      sections.snapshot = await section(() => getStudentSnapshot(sb, userId, model));
      sections.recent_tests = await section(() => getRecentTests(sb, userId, { limit: 6 }));
      sections.mastery = await section(() => getMastery(model, { questionLevel: null }, { subject, limit: 15 }));
      sections.what_checkmate_already_tried = await section(() => getInterventionHistory(sb, userId, { days: 60, limit: 10 }));
      break;
    }
  }
  return sections;
}

// A deliberately compact context — the "minimum useful state" tool.
export async function getLearningContext(sb: SupabaseClient, userId: string, model: any | null) {
  const section = async (fn: () => unknown | Promise<unknown>) => {
    try {
      return await fn();
    } catch (err: any) {
      return { unavailable: err?.message ?? 'failed' };
    }
  };
  const target = await loadDailyTarget(userId).catch(() => null);
  const recent = await attemptsSummary(sb, userId, 14).catch(() => null);
  return {
    priorities: await section(() => {
      const w: any = getWeakConcepts(model, { limit: 5 });
      return w.concepts.map((c: any) => ({ subject: c.subject, chapter: c.chapter, topic: c.topic, mastery: c.mastery }));
    }),
    recent_errors: await section(() => (getErrorPatterns(model, { limit: 5 }) as any).patterns),
    retention_due: await section(async () => (await getRetentionDue(sb, userId, model, { limit: 5 })) as any),
    upcoming_test: target ? target.upcoming_tests[0] ?? null : null,
    todays_target: target ? dailyTargetView(target) : { unavailable: 'daily target could not be computed' },
    recent_performance: recent ? recent.summary : null,
    still_wrong_questions: recent ? stillWrongTop(recent.attempts, 5) : [],
    recent_interventions: await section(async () => ((await getInterventionHistory(sb, userId, { days: 14, limit: 5 })) as any).interventions),
    todays_plan: await section(async () => {
      const p: any = await getTodayPlan(sb, userId);
      return { freshness: p.freshness, totals: p.totals, tasks: p.tasks.map((t: any) => ({ subject: t.subject, topic: t.topic, state: t.state, learning_intent: t.learning_intent })) };
    }),
    next_step: 'Drill down with checkmate_get_wrong_questions, checkmate_get_repair_questions, checkmate_get_retention_questions, or checkmate_get_context(purpose=...).'
  };
}

// ─────────────────────────────────────────────────────────────────────────────
// Tool registry — semantic tools only. There is intentionally NO execute_sql,
// get_table, dump_database, get_all_*_events, or any write tool (FINAL_PLAN: "Absolutely
// don't expose" + "Writes: V2, not V1"). Names use underscores (not dots) because several
// MCP hosts validate tool names against ^[a-zA-Z0-9_-]{1,64}$.
// ─────────────────────────────────────────────────────────────────────────────

type Prop = Record<string, unknown>;
const S = (description: string): Prop => ({ type: 'string', description });
const I = (description: string, min?: number, max?: number): Prop => ({ type: 'integer', description, ...(min !== undefined ? { minimum: min } : {}), ...(max !== undefined ? { maximum: max } : {}) });
const B = (description: string): Prop => ({ type: 'boolean', description });
const schema = (properties: Record<string, Prop> = {}, required: string[] = []) => ({
  type: 'object',
  properties,
  required,
  additionalProperties: false
});

const SUBJECT = S('Optional subject filter, e.g. Physics, Chemistry, Botany, Zoology.');
const CHAPTER = S('Optional exact chapter name (as Testmate tags it).');
const TOPIC = S('Optional exact topic name.');
const DAYS = (def: number) => I(`Look-back window in days (default ${def}, max 365).`, 1, 365);
const SOURCE: Prop = {
  type: 'string',
  enum: SOURCE_VALUES,
  description:
    "Where the attempt came from: full_test (real mocks/FTs), qbank_practice, targeted_repair (Checkmate repair tasks), retention_check. Default 'all'."
};

function tool(name: string, title: string, scope: Scope, description: string, inputSchema: Record<string, unknown>, handler: ToolDef['handler']): ToolDef {
  return { name: `checkmate_${name}`, title, scope, description, inputSchema, handler };
}

import { NextRequest, NextResponse } from 'next/server';
import { getServiceClient } from '@/lib/supabase';
import { authenticateGateway, hasScope } from './auth';
import { isRateLimited, recordAudit } from './audit';
import { buildPrompts, buildResources, buildTools } from './registry';
import { capResponse, minimize } from './sanitize';
import { GatewayPrincipal, ToolDataUnavailable, ToolInputError } from './types';

// Minimal MCP server over Streamable HTTP (POST JSON-RPC, JSON responses, no SSE, no
// sessions — every call is independent, which is what a stateless serverless host wants).

const SUPPORTED_VERSIONS = ['2025-06-18', '2025-03-26', '2024-11-05'];
const SERVER_INFO = { name: 'checkmate-gateway', title: 'Checkmate Learning Gateway', version: '1.0.0' };
const INSTRUCTIONS =
  "Read-only access to one student's learning data from Checkmate (planner/mastery) and Testmate (tests, Q-bank). " +
  'Start with checkmate_get_student_snapshot, or checkmate_get_context(purpose=...) for a task-specific bundle. ' +
  'Question-level tools return the real questions, the student\'s answers, correct answers and explanations — use them to ground any diagnosis ' +
  'instead of guessing from counts. Data may be stale if the phone has not synced (see freshness fields). This gateway cannot change anything.';

type JsonRpcId = string | number | null;
interface JsonRpcRequest {
  jsonrpc: '2.0';
  id?: JsonRpcId;
  method: string;
  params?: Record<string, any>;
}

const ERR = { parse: -32700, invalidRequest: -32600, methodNotFound: -32601, invalidParams: -32602, internal: -32603 };

function rpcError(id: JsonRpcId | undefined, code: number, message: string) {
  return { jsonrpc: '2.0', id: id ?? null, error: { code, message } };
}
function rpcResult(id: JsonRpcId | undefined, result: unknown) {
  return { jsonrpc: '2.0', id: id ?? null, result };
}

function originAllowed(req: NextRequest): boolean {
  const origin = req.headers.get('origin');
  if (!origin) return true; // server-to-server hosts send none
  const allowed = (process.env.GATEWAY_ALLOWED_ORIGINS ?? '')
    .split(',')
    .map((s) => s.trim())
    .filter(Boolean);
  // Default: refuse browser origins entirely (DNS-rebinding defense). Opt specific
  // origins in with GATEWAY_ALLOWED_ORIGINS if a browser-based MCP client is ever needed.
  return allowed.includes(origin);
}

const tools = buildTools();
const toolByName = new Map(tools.map((t) => [t.name, t]));
const resources = buildResources();
const prompts = buildPrompts();

function toolResult(payload: unknown, isError = false) {
  const { text } = capResponse(isError ? payload : minimize(payload));
  return { content: [{ type: 'text', text }], isError };
}

async function handleOne(msg: JsonRpcRequest, principal: GatewayPrincipal): Promise<unknown | null> {
  const isNotification = msg.id === undefined;
  const started = Date.now();
  const supabase = getServiceClient();
  const ctx = { principal, supabase };

  const audit = (extra: { tool?: string; args?: unknown; ok: boolean; error?: string; bytes?: number }) =>
    recordAudit({ principal, method: msg.method, tool: extra.tool, args: extra.args, ok: extra.ok, error: extra.error, durationMs: Date.now() - started, responseBytes: extra.bytes });

  try {
    switch (msg.method) {
      case 'initialize': {
        const requested = msg.params?.protocolVersion;
        const version = SUPPORTED_VERSIONS.includes(requested) ? requested : SUPPORTED_VERSIONS[0];
        void audit({ ok: true });
        return rpcResult(msg.id, {
          protocolVersion: version,
          capabilities: { tools: { listChanged: false }, resources: { listChanged: false, subscribe: false }, prompts: { listChanged: false } },
          serverInfo: SERVER_INFO,
          instructions: INSTRUCTIONS
        });
      }
      case 'notifications/initialized':
      case 'notifications/cancelled':
        return null;
      case 'ping':
        return rpcResult(msg.id, {});

      case 'tools/list': {
        const visible = tools.filter((t) => hasScope(principal, t.scope));
        return rpcResult(msg.id, {
          tools: visible.map((t) => ({
            name: t.name,
            title: t.title,
            description: t.description,
            inputSchema: t.inputSchema,
            annotations: { readOnlyHint: true, destructiveHint: false, idempotentHint: true, openWorldHint: false }
          }))
        });
      }

      case 'tools/call': {
        const name = msg.params?.name;
        const args = (msg.params?.arguments ?? {}) as Record<string, unknown>;
        const tool = typeof name === 'string' ? toolByName.get(name) : undefined;
        if (!tool) return rpcError(msg.id, ERR.invalidParams, `Unknown tool: ${String(name)}`);
        if (typeof args !== 'object' || Array.isArray(args)) return rpcError(msg.id, ERR.invalidParams, 'arguments must be an object');
        if (!hasScope(principal, tool.scope)) {
          void audit({ tool: tool.name, args, ok: false, error: `missing scope ${tool.scope}` });
          return rpcResult(msg.id, toolResult({ error: `This token does not have the ${tool.scope} scope.` }, true));
        }
        try {
          const result = await tool.handler(args, ctx);
          const out = toolResult(result);
          void audit({ tool: tool.name, args, ok: true, bytes: Buffer.byteLength(out.content[0].text, 'utf8') });
          return rpcResult(msg.id, out);
        } catch (err: any) {
          // Tool-level failures are returned as readable tool errors so the model can
          // self-correct; only genuine bugs get a generic message (details stay in logs).
          const known = err instanceof ToolInputError || err instanceof ToolDataUnavailable;
          if (!known) console.error(`[gateway] ${tool.name} failed:`, err?.message ?? err);
          const message = known ? err.message : 'Internal error while reading learning data. Try narrowing the filters or retry shortly.';
          void audit({ tool: tool.name, args, ok: false, error: known ? message : `internal: ${String(err?.message ?? err)}` });
          return rpcResult(msg.id, toolResult({ error: message }, true));
        }
      }

      case 'resources/list': {
        const visible = resources.filter((r) => hasScope(principal, r.scope));
        return rpcResult(msg.id, {
          resources: visible.map((r) => ({ uri: r.uri, name: r.name, title: r.title, description: r.description, mimeType: 'application/json' }))
        });
      }
      case 'resources/read': {
        const uri = msg.params?.uri;
        const res = resources.find((r) => r.uri === uri);
        if (!res || !hasScope(principal, res.scope)) return rpcError(msg.id, ERR.invalidParams, `Unknown resource: ${String(uri)}`);
        try {
          const { text } = capResponse(minimize(await res.read(ctx)));
          void audit({ tool: `resource:${res.uri}`, ok: true, bytes: Buffer.byteLength(text, 'utf8') });
          return rpcResult(msg.id, { contents: [{ uri: res.uri, mimeType: 'application/json', text }] });
        } catch (err: any) {
          const known = err instanceof ToolInputError || err instanceof ToolDataUnavailable;
          void audit({ tool: `resource:${res.uri}`, ok: false, error: known ? err.message : `internal: ${String(err?.message ?? err)}` });
          return rpcError(msg.id, ERR.internal, known ? err.message : 'Failed to read resource');
        }
      }

      case 'prompts/list':
        return rpcResult(msg.id, { prompts: prompts.map((p) => ({ name: p.name, title: p.title, description: p.description, arguments: p.arguments })) });
      case 'prompts/get': {
        const p = prompts.find((x) => x.name === msg.params?.name);
        if (!p) return rpcError(msg.id, ERR.invalidParams, `Unknown prompt: ${String(msg.params?.name)}`);
        const args: Record<string, string> = {};
        for (const [k, v] of Object.entries((msg.params?.arguments ?? {}) as Record<string, unknown>)) if (typeof v === 'string') args[k] = v.slice(0, 200);
        return rpcResult(msg.id, { description: p.description, messages: [{ role: 'user', content: { type: 'text', text: p.render(args) } }] });
      }

      default:
        void audit({ ok: false, error: 'method not found' });
        return isNotification ? null : rpcError(msg.id, ERR.methodNotFound, `Method not found: ${msg.method}`);
    }
  } catch (err: any) {
    console.error('[gateway] unhandled:', err?.message ?? err);
    return isNotification ? null : rpcError(msg.id, ERR.internal, 'Internal error');
  }
}

export async function handleMcpPost(req: NextRequest): Promise<NextResponse> {
  if (!originAllowed(req)) return NextResponse.json({ error: 'Origin not allowed' }, { status: 403 });

  const auth = await authenticateGateway(req);
  if (!auth.ok) {
    return NextResponse.json(rpcError(null, ERR.invalidRequest, auth.message), {
      status: auth.status,
      headers: { 'WWW-Authenticate': 'Bearer realm="checkmate-gateway"' }
    });
  }
  const { principal } = auth;

  if (await isRateLimited(principal.tokenId)) {
    return NextResponse.json(rpcError(null, ERR.internal, 'Rate limit exceeded — slow down and retry in a minute.'), {
      status: 429,
      headers: { 'Retry-After': '30' }
    });
  }

  let body: unknown;
  try {
    body = await req.json();
  } catch {
    return NextResponse.json(rpcError(null, ERR.parse, 'Parse error'), { status: 400 });
  }

  const isBatch = Array.isArray(body);
  const messages = (isBatch ? body : [body]) as JsonRpcRequest[];
  if (messages.length === 0 || messages.length > 10) {
    return NextResponse.json(rpcError(null, ERR.invalidRequest, 'Batch must contain 1-10 messages'), { status: 400 });
  }

  const replies: unknown[] = [];
  for (const m of messages) {
    if (!m || m.jsonrpc !== '2.0' || typeof m.method !== 'string') {
      replies.push(rpcError((m as any)?.id, ERR.invalidRequest, 'Invalid JSON-RPC request'));
      continue;
    }
    const r = await handleOne(m, principal);
    if (r !== null) replies.push(r);
  }

  if (replies.length === 0) return new NextResponse(null, { status: 202 }); // notifications only
  return NextResponse.json(isBatch ? replies : replies[0], { headers: { 'Cache-Control': 'no-store' } });
}

// POST { label?, scopes?: string[], expires_in_days?: number }  -> raw token shown ONCE
export async function POST(req: NextRequest) {
  const user = await getCurrentUser();
  if (!user) return NextResponse.json({ error: 'Not authenticated' }, { status: 401 });

  let body: any = {};
  try {
    body = await req.json();
  } catch {
    // defaults
  }
  const label = typeof body?.label === 'string' && body.label.trim() ? body.label.trim().slice(0, 60) : 'External AI';

  let scopes: Scope[] = V1_MINTABLE_SCOPES;
  if (Array.isArray(body?.scopes)) {
    const requested = body.scopes.filter(isScope) as Scope[];
    const bad = body.scopes.filter((s: unknown) => !isScope(s) || !V1_MINTABLE_SCOPES.includes(s as Scope));
    if (bad.length > 0) {
      return NextResponse.json({ error: `Scopes not available in v1: ${bad.join(', ')}. Allowed: ${V1_MINTABLE_SCOPES.join(', ')}` }, { status: 400 });
    }
    scopes = Array.from(new Set(requested));
    if (scopes.length === 0) return NextResponse.json({ error: 'At least one scope is required' }, { status: 400 });
  }

  let expiresAt: string | null = null;
  if (body?.expires_in_days !== undefined && body.expires_in_days !== null) {
    const days = Number(body.expires_in_days);
    if (!Number.isFinite(days) || days < 1 || days > 365) {
      return NextResponse.json({ error: 'expires_in_days must be between 1 and 365' }, { status: 400 });
    }
    expiresAt = new Date(Date.now() + days * 86_400_000).toISOString();
  }

  const { raw, hash } = generateMcpToken();
  const supabase = getServiceClient();
  const { data, error } = await supabase
    .from('mcp_tokens')
    .insert({ user_id: user.id, token_hash: hash, label, scopes, expires_at: expiresAt })
    .select('id, label, scopes, created_at, expires_at')
    .single();
  if (error || !data) return NextResponse.json({ error: error?.message ?? 'Failed to create token' }, { status: 500 });

  return NextResponse.json({ token: raw, ...data });
}

import { NextRequest, NextResponse } from 'next/server';
import { getCurrentUserFromRequest } from '@/lib/auth';
import { getServiceClient } from '@/lib/supabase';
import { isProjectionKind, MAX_PROJECTION_BYTES, ProjectionError, projectionStatus, putProjection } from '@/lib/gateway/projection';

export const dynamic = 'force-dynamic';
export const maxDuration = 30;

// PUT /api/gateway/projection?kind=student_model|interventions|today_plan
// Called by the Checkmate Android app (device-token Bearer, same as every other Checkmate
// call). Stores a curated, re-normalized snapshot that the MCP gateway later READS.
// This is the only write path into the gateway's data, and it can only ever write the
// caller's own projection rows.
export async function PUT(req: NextRequest) {
  const user = await getCurrentUserFromRequest(req);
  if (!user) return NextResponse.json({ error: 'Not authenticated' }, { status: 401 });

  const kind = req.nextUrl.searchParams.get('kind');
  if (!isProjectionKind(kind)) {
    return NextResponse.json({ error: 'kind must be one of: student_model, interventions, today_plan' }, { status: 400 });
  }

  const raw = await req.text();
  if (Buffer.byteLength(raw, 'utf8') > MAX_PROJECTION_BYTES) {
    return NextResponse.json({ error: 'Payload too large' }, { status: 413 });
  }
  let body: unknown;
  try {
    body = JSON.parse(raw);
  } catch {
    return NextResponse.json({ error: 'Invalid JSON body' }, { status: 400 });
  }

  try {
    const result = await putProjection(getServiceClient(), user.id, kind, body);
    return NextResponse.json(result);
  } catch (err: any) {
    if (err instanceof ProjectionError) return NextResponse.json({ error: err.message }, { status: 400 });
    console.error('[gateway/projection] failed:', err?.message);
    return NextResponse.json({ error: 'Failed to store projection' }, { status: 500 });
  }
}

/**
 * Checkmate MCP gateway — read-projection push (FINAL_PLAN "Option C — hybrid").
 *
 * Checkmate stays the source of truth for the student model, the intervention history and
 * today's plan. This pushes curated SNAPSHOTS of those three things to Testmate's
 * `PUT /api/gateway/projection`, using the SAME base URL + device token the rest of
 * Checkmate already uses for Testmate (Settings -> Test Platform). The MCP gateway on the
 * Testmate side only ever reads those stored snapshots — it never reaches into this phone,
 * and keeps answering (with a freshness stamp) while the phone is offline.
 *
 * What is deliberately NOT pushed: the raw behavior event ledger, screen/app-usage data,
 * profile/consultation text, API keys, anything outside the three snapshots below. The
 * server re-normalizes every field it accepts, so this file is the first filter, not the
 * only one.
 *
 * Fire-and-forget background sync: a periodic WorkManager job (network required) plus one
 * immediate run at app start. Each snapshot is pushed independently — one failing (e.g. an
 * empty student model on a fresh install) never blocks the others.
 */
object GatewaySync {

    private const val TAG = "GatewaySync"
    private const val MAX_INTERVENTIONS = 300

    private val client = OkHttpClient.Builder()
        .connectTimeout(15, TimeUnit.SECONDS)
        .writeTimeout(30, TimeUnit.SECONDS)
        .readTimeout(30, TimeUnit.SECONDS)
        .build()

    enum class PushResult { OK, SKIPPED_NOT_CONFIGURED, REJECTED, RETRYABLE_FAILURE }

    /** Pushes all three snapshots. Returns the worst outcome so the Worker can decide to retry. */
    suspend fun pushAll(context: Context): PushResult = withContext(Dispatchers.IO) {
        val base = baseUrl()
        val token = token()
        if (base == null || token == null) return@withContext PushResult.SKIPPED_NOT_CONFIGURED

        val results = listOf(
            "student_model" to { buildStudentModel(context) },
            "interventions" to { buildInterventions(context) },
            "today_plan" to { buildTodayPlan() }
        ).map { (kind, build) ->
            try {
                push(base, token, kind, build())
            } catch (e: Exception) {
                Log.w(TAG, "push $kind failed to build/send: ${e.message}")
                PushResult.RETRYABLE_FAILURE
            }
        }
        when {
            PushResult.REJECTED in results -> PushResult.REJECTED
            PushResult.RETRYABLE_FAILURE in results -> PushResult.RETRYABLE_FAILURE
            else -> PushResult.OK
        }
    }

    // ── config (same prefs + allow-list as TestmateApi; never accepts an arbitrary host) ──
    private fun baseUrl(): String? {
        val saved = CheckmatePrefs.getString(TestmateApi.PREF_BASE_URL, null)?.trim()?.trimEnd('/')
            ?.takeIf { it.isNotBlank() } ?: return null
        if (!TestmateApi.isAllowedBaseUrl(saved)) {
            Log.w(TAG, "Stored Testmate base URL failed the allow-list check — not syncing")
            return null
        }
        return saved
    }

    private fun token(): String? =
        CheckmatePrefs.getString(TestmateApi.PREF_TOKEN, null)?.trim()?.takeIf { it.isNotBlank() }

    private fun push(base: String, token: String, kind: String, payload: JSONObject): PushResult {
        payload.put("generated_ms", System.currentTimeMillis())
        val body = payload.toString().toRequestBody("application/json".toMediaType())
        val request = Request.Builder()
            .url("$base/api/gateway/projection?kind=$kind")
            .addHeader("Authorization", "Bearer $token")
            .put(body)
            .build()
        client.newCall(request).execute().use { resp ->
            Log.d(TAG, "push $kind: HTTP ${resp.code}")
            return when {
                resp.isSuccessful -> PushResult.OK
                // 400/401/403/413: retrying the identical payload/credentials can't succeed.
                resp.code in 400..499 && resp.code != 408 && resp.code != 429 -> PushResult.REJECTED
                else -> PushResult.RETRYABLE_FAILURE
            }
        }
    }

    // org.json throws on NaN/Infinity — skip non-finite numbers instead of failing the whole push.
    private fun JSONObject.putNum(key: String, v: Double?): JSONObject {
        if (v != null && v.isFinite()) put(key, v)
        return this
    }

    private fun JSONObject.putStr(key: String, v: String?): JSONObject {
        if (!v.isNullOrBlank() && v != "null") put(key, v)
        return this
    }

    // ── student_model ─────────────────────────────────────────────────────────
    private suspend fun buildStudentModel(context: Context): JSONObject {
        val model: StudentModel = StudentModelBuilder.build(context)
        val root = JSONObject()
        root.putStr("exam", runCatching { ConsultationProfile.load().examTarget }.getOrNull())
        root.put("model_version", model.modelVersion)
        root.put(
            "overall",
            JSONObject().apply {
                put("concepts_tracked", model.overall.conceptsTracked)
                put("concepts_mastered", model.overall.conceptsMastered)
                put("concepts_weak", model.overall.conceptsWeak)
                putNum("average_mastery", model.overall.averageMastery)
                put("total_attempts", model.overall.totalAttempts)
                put("unresolved_error_count", model.overall.unresolvedErrorCount)
            }
        )

        val concepts = JSONArray()
        for (c in model.concepts.values) {
            val issues = JSONArray()
            for (p in c.prerequisiteIssues) {
                issues.put(JSONObject().putStr("subject", p.subject).putStr("chapter", p.chapter).putStr("topic", p.topic))
            }
            concepts.put(
                JSONObject().apply {
                    putStr("concept_id", c.conceptId)
                    putStr("subject", c.subject)
                    putStr("chapter", c.chapter)
                    putStr("topic", c.topic)
                    putNum("mastery", c.mastery)
                    putNum("forgetting_risk", c.forgettingRisk)
                    put("retention_decision", c.retentionDecision.name)
                    put("attempt_count", c.attemptCount)
                    putNum("recent_accuracy", c.recentAccuracy)
                    putNum("lifetime_accuracy", c.lifetimeAccuracy)
                    put("error_count", c.errorCount)
                    c.lastSeen?.let { put("last_seen_ms", it) }
                    put("weak", c.mastery < MasteryEngine.MASTERY_THRESHOLD)
                    put("prerequisite_issues", issues)
                }
            )
        }
        root.put("concepts", concepts)

        val errors = JSONArray()
        for (e in model.unresolvedErrors) {
            val concept = model.concepts[e.conceptId]
            errors.put(
                JSONObject().apply {
                    putStr("concept_id", e.conceptId)
                    putStr("subject", concept?.subject)
                    putStr("chapter", concept?.chapter)
                    putStr("topic", concept?.topic)
                    putStr("error_type", e.errorType)
                    put("occurrences", e.occurrences)
                    put("first_seen_ms", e.firstSeen)
                    put("last_seen_ms", e.lastSeen)
                }

    }

    private suspend fun buildInterventions(context: Context): JSONObject {

                }

        }

    }

/** WorkManager entry point. Network-constrained; only retries failures that could succeed on retry. */
class GatewaySyncWorker(
    context: Context,
    params: WorkerParameters
) : CoroutineWorker(context, params) {

    override suspend fun doWork(): Result =
        when (GatewaySync.pushAll(applicationContext)) {
            GatewaySync.PushResult.RETRYABLE_FAILURE -> if (runAttemptCount < 3) Result.retry() else Result.success()
            // Not configured / rejected: nothing to gain from retrying; the next periodic run re-checks.
            else -> Result.success()
        }
}

object GatewaySyncScheduler {

    private const val PERIODIC_WORK = "checkmate_gateway_sync"
    private const val STARTUP_WORK = "checkmate_gateway_sync_startup"

    /** WorkManager's periodic floor is 15 min; 30 keeps the snapshot fresh without wasting battery. */
    private const val INTERVAL_MINUTES = 30L

    fun schedule(context: Context) {
        val constraints = Constraints.Builder()
            .setRequiredNetworkType(NetworkType.CONNECTED)
            .build()
        val wm = WorkManager.getInstance(context)

        wm.enqueueUniquePeriodicWork(
            PERIODIC_WORK,
            ExistingPeriodicWorkPolicy.KEEP,
            PeriodicWorkRequestBuilder<GatewaySyncWorker>(INTERVAL_MINUTES, TimeUnit.MINUTES)
                .setConstraints(constraints)
                .setBackoffCriteria(BackoffPolicy.EXPONENTIAL, 5, TimeUnit.MINUTES)
                .build()
        )
        // One immediate run per app start so a fresh install / new token has data right away.
        wm.enqueueUniqueWork(
            STARTUP_WORK,
            ExistingWorkPolicy.KEEP,
            OneTimeWorkRequestBuilder<GatewaySyncWorker>()
                .setConstraints(constraints)
                .setBackoffCriteria(BackoffPolicy.EXPONENTIAL, 5, TimeUnit.MINUTES)
                .build()
        )
    }

    fun cancel(context: Context) {
        val wm = WorkManager.getInstance(context)
        wm.cancelUniqueWork(PERIODIC_WORK)
        wm.cancelUniqueWork(STARTUP_WORK)
    }
}
