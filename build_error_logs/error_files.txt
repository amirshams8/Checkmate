 473    /**
 474     * Hands a WebView download off to the system DownloadManager so it lands
 475     * in the device's normal Downloads folder with a notification, progress
 476     * bar, and retry-on-failure — same as any browser download. Cookies are
 477     * copied over explicitly because DownloadManager makes its own network
 478     * request outside the WebView, so it doesn't automatically inherit the
 479     * WebView's session.
 480     */
 481    private fun downloadFile(
 482        context: Context,
 483        url: String,
 484        userAgent: String?,
 485        contentDisposition: String?,
 486        mimeType: String?
 487    ) {
 488        try {
 489            val fileName = URLUtil.guessFileName(url, contentDisposition, mimeType)
 490            val cookie = CookieManager.getInstance().getCookie(url)
 491            val resolvedMimeType = mimeType?.takeIf { it.isNotBlank() }
 492                ?: MimeTypeMap.getSingleton()
 493                    .getMimeTypeFromExtension(MimeTypeMap.getFileExtensionFromUrl(url))
 494                ?: "application/octet-stream"
 495    
 496            val request = DownloadManager.Request(Uri.parse(url)).apply {
 497                if (!cookie.isNullOrBlank()) addRequestHeader("cookie", cookie)
 498                if (!userAgent.isNullOrBlank()) addRequestHeader("User-Agent", userAgent)
 499                setMimeType(resolvedMimeType)
 500                setNotificationVisibility(DownloadManager.Request.VISIBILITY_VISIBLE_NOTIFY_COMPLETED)
 501                setDestinationInExternalPublicDir(Environment.DIRECTORY_DOWNLOADS, fileName)
 502                setAllowedOverRoaming(true)
 503            }
 504    
 505            val downloadManager = context.getSystemService(Context.DOWNLOAD_SERVICE) as DownloadManager
 506            downloadManager.enqueue(request)
 507            Toast.makeText(context, "Downloading $fileName…", Toast.LENGTH_SHORT).show()
 508        } catch (e: Exception) {
 509            Toast.makeText(context, "Download failed: ${e.message}", Toast.LENGTH_LONG).show()
 510        }
 511    }
