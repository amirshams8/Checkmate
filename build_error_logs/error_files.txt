 496                val cycleState = AttentionCycleManager.currentState()
 497                AttentionCycleService.stop(context)
 498                FloatingAttentionService.stop(context)
 499                WorkModeManager.deactivate(context)
 500                ScreenCaptureManager.release()
 501                PlanStore.markTask(task.id, TaskState.SKIPPED)
 502                // BUGFIX (Blueprint 2.6): same focus-time fix as confirmCompletion() — a skipped
 503                // session still did real focus minutes before the skip, worth recording rather
 504                // than leaving at the StudyTask default of 0.
 505                val focusMin = (cycleState.totalSessionSeconds / 60).toInt().coerceAtLeast(0)
 506                PlanStore.setFocusMinutes(task.id, focusMin)
 507                val skippedTask = task.copy(focusMinutes = focusMin, actualMinutes = focusMin)
 508                // Mentor v2 (spec 3.6): capture whatever app was most recently foregrounded so the
 509                // skip event — and any guardian alert built from it — can name the actual cause.
 510                val distractionApp = try {
 511                    AppUsageTracker.getMostRecentForegroundApp(context)
 512                } catch (_: Exception) { null }
 513                PsycheEngine.onTaskSkipped(skippedTask, cycleState.checksPassed, cycleState.checksMissed, distractionApp)
 514                // Mentor v2 (spec 3.7): keep the planner's cached behavior summary current on
 515                // skips too, not just completions, so a same-day regenerate reflects the skip.
 516                PsycheEngine.refreshBehaviorSummaryCache()
 517                // Mentor v2 (spec 3.4): opens the escalation-watchlist lockdown window — checked by
 518                // AppAutomationService alongside the normal blocklist for the configured duration.
 519                WorkModeManager.startPostSkipLockdown(context)
 520                val msg = PsycheEngine.getSkipReaction(task)
 521                CheckmateTTS.speak(context, msg)
 522                // Mentor v2 (spec 3.2): same reaction text, also logged into Mentor's persisted
 523                // chat so it's a running log, not just a spoken line that's gone once said.
 524                ProactiveMentor.onSkip(context, msg)
 525                val newSkips = _state.value.consecutiveSkips + 1
 526                _state.update { it.copy(activeTaskId = null, psycheMessage = msg, consecutiveSkips = newSkips) }
 527                if (newSkips >= 3) {
 528                    GuardianNotifier.notifySkipStreak(context, newSkips, "${task.subject}: ${task.topic}", distractionApp)
 529                }
 530            }
 531        }
