 129        /**
 130         * Adds a full-screen, invisible, touch-swallowing overlay so nothing under it
 131         * can be tapped, then auto-removes itself after [durationMs]. Safe to call
 132         * repeatedly while a guarded screen keeps re-triggering the watchdog — it just
 133         * extends the existing overlay's lifetime instead of stacking a second window,
 134         * and (see [touchBlockUntil]) a shorter subsequent call can never cut short a
 135         * longer block already in flight.
 136         */
 137        private fun blockTouchesBriefly(durationMs: Long = 700L) {
 138            val requestedUntil = System.currentTimeMillis() + durationMs
 139            if (requestedUntil <= touchBlockUntil) {
 140                // A longer block (POST_DETECT_BLOCK_MS or the hard lock) is already
 141                // scheduled to release later than this request would — ignore, don't
 142                // shorten it.
 143                return
 144            }
 145            touchBlockUntil = requestedUntil
 146    
 147            touchBlockerHandler.removeCallbacks(removeTouchBlockerRunnable)
 148            if (touchBlockerView == null) {
 149                val params = WindowManager.LayoutParams(
 150                    WindowManager.LayoutParams.MATCH_PARENT,
 151                    WindowManager.LayoutParams.MATCH_PARENT,
 152                    WindowManager.LayoutParams.TYPE_ACCESSIBILITY_OVERLAY,
 153                    WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE or
 154                        WindowManager.LayoutParams.FLAG_NOT_TOUCH_MODAL,
 155                    PixelFormat.TRANSLUCENT
 156                )
 157                val view = View(this)
 158                try {
 159                    windowManager.addView(view, params)
 160                    touchBlockerView = view
 161                } catch (e: Exception) {
 162                    Log.w(GUARD_TAG, "touch blocker overlay failed to add", e)
 163                    touchBlockUntil = 0L
 164                    return
 165                }
 166            }
 167            touchBlockerHandler.postDelayed(removeTouchBlockerRunnable, durationMs)
 168        }

 387        /** Depth-first collection of all text + content-description on screen, space-joined. */
 388        private fun collectAllText(root: AccessibilityNodeInfo): String {
 389            val sb = StringBuilder()
 390            val stack = ArrayDeque<AccessibilityNodeInfo>()
 391            stack.add(root)
 392            var visited = 0
 393            while (stack.isNotEmpty() && visited < 400) { // cap traversal — Settings trees are small
 394                val node = stack.removeFirst()
 395                visited++
 396                node.text?.let { sb.append(it).append(' ') }
 397                node.contentDescription?.let { sb.append(it).append(' ') }
 398                for (i in 0 until node.childCount) node.getChild(i)?.let { stack.add(it) }
 399            }
 400            return sb.toString()
 401        }

 427        /**
 428         * Tries known view IDs for the browser first, then falls back to
 429         * scanning all text nodes whose content-description suggests a URL bar.
 430         */
 431        private fun extractUrl(root: AccessibilityNodeInfo, pkg: String): String? {
 432            // Try known IDs for this browser
 433            val ids = URL_BAR_IDS[pkg] ?: emptyList()
 434            for (id in ids) {
 435                val node = root.findAccessibilityNodeInfosByViewId(id)?.firstOrNull()
 436                val text = node?.text?.toString()?.trim()
 437                if (!text.isNullOrBlank()) return text
 438            }
 439    
 440            // Generic fallback: find any node whose description sounds like an address bar
 441            // and whose text looks like a URL
 442            val stack = ArrayDeque<AccessibilityNodeInfo>()
 443            stack.add(root)
 444            while (stack.isNotEmpty()) {
 445                val node = stack.removeFirst()
 446                val desc = node.contentDescription?.toString() ?: ""
 447                val text = node.text?.toString()?.trim() ?: ""
 448                if (URL_BAR_DESCRIPTIONS.any { desc.contains(it, ignoreCase = true) }
 449                    && looksLikeUrl(text)) {
 450                    return text
 451                }
 452                for (i in 0 until node.childCount) node.getChild(i)?.let { stack.add(it) }
 453            }
 454            return null
 455        }

 457        private fun looksLikeUrl(text: String): Boolean {
 458            if (text.isBlank() || text.length < 4) return false
 459            return text.contains('.') &&
 460                   !text.contains(' ') &&
 461                   (text.startsWith("http") || text.startsWith("www.") || text.matches(Regex(".*\\.[a-z]{2,}.*")))
 462        }

 464        /**
 465         * Parses the registered hostname from a raw URL string.
 466         * "https://www.youtube.com/watch?v=..." → "youtube.com"
 467         *
 468         * Strips "www." so the stored domain "youtube.com" matches both
 469         * "youtube.com" and "www.youtube.com".
 470         */
 471        private fun parseHostname(raw: String): String? {
 472            return try {
 473                val withScheme = if (raw.startsWith("http")) raw else "https://$raw"
 474                val host = Uri.parse(withScheme).host ?: return null
 475                host.removePrefix("www.").lowercase().trim()
 476            } catch (_: Exception) { null }
 477        }

 520            return true
 521        }

 525        private fun findNodeById(root: AccessibilityNodeInfo, id: String): AccessibilityNodeInfo? =
 526            root.findAccessibilityNodeInfosByViewId(id)?.firstOrNull()
 527    
 528        private fun findNodeByClass(root: AccessibilityNodeInfo, className: String): AccessibilityNodeInfo? {
 529            val stack = ArrayDeque<AccessibilityNodeInfo>()
 530            stack.add(root)
 531            while (stack.isNotEmpty()) {
 532                val node = stack.removeFirst()
 533                if (node.className?.toString() == className && node.isEditable) return node
 534                for (i in 0 until node.childCount) node.getChild(i)?.let { stack.add(it) }
 535            }
 536            return null
 537        }
 538    
 539        private fun findNodeByDesc(root: AccessibilityNodeInfo, desc: String): AccessibilityNodeInfo? {
 540            val stack = ArrayDeque<AccessibilityNodeInfo>()
 541            stack.add(root)
 542            while (stack.isNotEmpty()) {
 543                val node = stack.removeFirst()
 544                if (node.contentDescription?.toString()?.contains(desc, ignoreCase = true) == true
 545                    && node.isClickable) return node
 546                for (i in 0 until node.childCount) node.getChild(i)?.let { stack.add(it) }
 547            }
 548            return null
 549        }
