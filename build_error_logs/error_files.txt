  33    /**
  34     * Checkmate MCP gateway — read-projection push (FINAL_PLAN "Option C — hybrid").
  35     *
  36     * Checkmate stays the source of truth for the student model, the intervention history and
  37     * today's plan. This pushes curated SNAPSHOTS of those three things to Testmate's
  38     * `PUT /api/gateway/projection`, using the SAME base URL + device token the rest of
  39     * Checkmate already uses for Testmate (Settings -> Test Platform). The MCP gateway on the
  40     * Testmate side only ever reads those stored snapshots — it never reaches into this phone,
  41     * and keeps answering (with a freshness stamp) while the phone is offline.
  42     *
  43     * What is deliberately NOT pushed: the raw behavior event ledger, screen/app-usage data,
  44     * profile/consultation text, API keys, anything outside the three snapshots below. The
  45     * server re-normalizes every field it accepts, so this file is the first filter, not the
  46     * only one.
  47     *
  48     * Fire-and-forget background sync: a periodic WorkManager job (network required) plus one
  49     * immediate run at app start. Each snapshot is pushed independently — one failing (e.g. an
  50     * empty student model on a fresh install) never blocks the others.

 104        private fun push(base: String, token: String, kind: String, payload: JSONObject): PushResult {
 105            payload.put("generated_ms", System.currentTimeMillis())
 106            val body = payload.toString().toRequestBody("application/json".toMediaType())
 107            val request = Request.Builder()
 108                .url("$base/api/gateway/projection?kind=$kind")
 109                .addHeader("Authorization", "Bearer $token")
 110                .put(body)
 111                .build()
 112            client.newCall(request).execute().use { resp ->
 113                Log.d(TAG, "push $kind: HTTP ${resp.code}")
 114                return when {
 115                    resp.isSuccessful -> PushResult.OK
 116                    // 400/401/403/413: retrying the identical payload/credentials can't succeed.
 117                    resp.code in 400..499 && resp.code != 408 && resp.code != 429 -> PushResult.REJECTED
 118                    else -> PushResult.RETRYABLE_FAILURE
 119                }
 120            }
 121        }

 264    object GatewaySyncScheduler {
 265    
 266        private const val PERIODIC_WORK = "checkmate_gateway_sync"
 267        private const val STARTUP_WORK = "checkmate_gateway_sync_startup"
 268    
 269        /** WorkManager's periodic floor is 15 min; 30 keeps the snapshot fresh without wasting battery. */
 270        private const val INTERVAL_MINUTES = 30L
 271    
 272        fun schedule(context: Context) {
 273            val constraints = Constraints.Builder()
 274                .setRequiredNetworkType(NetworkType.CONNECTED)
 275                .build()
 276            val wm = WorkManager.getInstance(context)
 277    
 278            wm.enqueueUniquePeriodicWork(
 279                PERIODIC_WORK,
 280                ExistingPeriodicWorkPolicy.KEEP,
 281                PeriodicWorkRequestBuilder<GatewaySyncWorker>(INTERVAL_MINUTES, TimeUnit.MINUTES)
 282                    .setConstraints(constraints)
 283                    .setBackoffCriteria(BackoffPolicy.EXPONENTIAL, 5, TimeUnit.MINUTES)
 284                    .build()
 285            )
 286            // One immediate run per app start so a fresh install / new token has data right away.
 287            wm.enqueueUniqueWork(
 288                STARTUP_WORK,
 289                ExistingWorkPolicy.KEEP,
 290                OneTimeWorkRequestBuilder<GatewaySyncWorker>()
 291                    .setConstraints(constraints)
 292                    .setBackoffCriteria(BackoffPolicy.EXPONENTIAL, 5, TimeUnit.MINUTES)
 293                    .build()
 294            )
 295        }
 296    
 297        fun cancel(context: Context) {
 298            val wm = WorkManager.getInstance(context)
 299            wm.cancelUniqueWork(PERIODIC_WORK)
 300            wm.cancelUniqueWork(STARTUP_WORK)
 301        }
