 268            val studyStart = CheckmatePrefs.getString("study_start", "06:00") ?: "06:00"
 269            val studyEnd   = CheckmatePrefs.getString("study_end", "22:00") ?: "22:00"
 270            val profile    = ConsultationProfile.load()

 495                // this is genuinely "focus time, not wall clock." Previously nothing ever wrote it
 496                // onto the task, so StudyTask.focusMinutes/actualMinutes — and everything downstream
 497                // that reads them (BehaviorLedger's "Avg Focus" stat, Stats' "Focus Today") — always
 498                // saw 0 regardless of how long the student actually focused.
 499                val focusMin = (cycleState.totalSessionSeconds / 60).toInt().coerceAtLeast(0)
 500                PlanStore.setFocusMinutes(task.id, focusMin)
 501                val completedTask = task.copy(focusMinutes = focusMin, actualMinutes = focusMin)
 502                PsycheEngine.onTaskCompleted(completedTask, cycleState.checksPassed, cycleState.checksMissed)
 503                // Mentor v2 (spec 3.7): log the completion into TodayContext so AdaptivePlanner
 504                // sees what's actually been finished today, then refresh the cached summary
 505                // AdaptivePlanner reads (previously-dead "behavior_summary" pref — see PsycheEngine).
 506                TodayContext.appendUpdate("Completed ${task.subject}: ${task.topic} (${task.taskType})")
 507                PsycheEngine.refreshBehaviorSummaryCache()
 508                val spoken = when (status) {
 509                    "NO"      -> "Noted. Every session still counts — reset for the next one."
 510                    "PARTIAL" -> "Task marked complete. Partial progress logged."
 511                    else      -> "Task complete. Well done."
 512                }
 513                CheckmateTTS.speak(context, spoken)
 514                val msg = PsycheEngine.getDailyMorningMessage()
 515                _state.update { it.copy(
 516                    activeTaskId         = null,
 517                    psycheMessage        = msg,
 518                    consecutiveSkips     = 0,
 519                    completionPromptTask = null
 520                ) }
 521            }
 522        }

 524        fun markSkip(context: Context, task: StudyTask) {
 525            viewModelScope.launch {
 526                // Same fix as confirmCompletion() — capture before stop()/reset() wipes the counts.
 527                val cycleState = AttentionCycleManager.currentState()
 528                AttentionCycleService.stop(context)
 529                FloatingAttentionService.stop(context)
 530                WorkModeManager.deactivate(context)
 531                ScreenCaptureManager.release()
 532                PlanStore.markTask(task.id, TaskState.SKIPPED)
 533                // BUGFIX (Blueprint 2.6): same focus-time fix as confirmCompletion() — a skipped
 534                // session still did real focus minutes before the skip, worth recording rather
 535                // than leaving at the StudyTask default of 0.
 536                val focusMin = (cycleState.totalSessionSeconds / 60).toInt().coerceAtLeast(0)
 537                PlanStore.setFocusMinutes(task.id, focusMin)
 538                val skippedTask = task.copy(focusMinutes = focusMin, actualMinutes = focusMin)
 539                // Mentor v2 (spec 3.6): capture whatever app was most recently foregrounded so the
 540                // skip event — and any guardian alert built from it — can name the actual cause.
 541                val distractionApp = try {
 542                    AppUsageTracker.getMostRecentForegroundApp(context)
 543                } catch (_: Exception) { null }
 544                PsycheEngine.onTaskSkipped(skippedTask, cycleState.checksPassed, cycleState.checksMissed, distractionApp)
 545                // Mentor v2 (spec 3.7): keep the planner's cached behavior summary current on
 546                // skips too, not just completions, so a same-day regenerate reflects the skip.
 547                PsycheEngine.refreshBehaviorSummaryCache()
 548                // Mentor v2 (spec 3.4): opens the escalation-watchlist lockdown window — checked by
 549                // AppAutomationService alongside the normal blocklist for the configured duration.
 550                WorkModeManager.startPostSkipLockdown(context)
 551                val msg = PsycheEngine.getSkipReaction(task)
 552                CheckmateTTS.speak(context, msg)
 553                // Mentor v2 (spec 3.2): same reaction text, also logged into Mentor's persisted
 554                // chat so it's a running log, not just a spoken line that's gone once said.
 555                ProactiveMentor.onSkip(context, msg)
 556                val newSkips = _state.value.consecutiveSkips + 1
 557                _state.update { it.copy(activeTaskId = null, psycheMessage = msg, consecutiveSkips = newSkips) }
 558                if (newSkips >= 3) {
 559                    GuardianNotifier.notifySkipStreak(context, newSkips, "${task.subject}: ${task.topic}", distractionApp)
 560                }
 561            }
 562        }
