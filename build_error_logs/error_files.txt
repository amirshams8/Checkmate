1047    /** "3h ago" / "2d ago" style label for [GapTaskManager.PREF_TESTMATE_LAST_ERROR_AT]. */
1048    private fun relativeTimeAgo(epochMillis: Long): String {
1049        val deltaMin = (System.currentTimeMillis() - epochMillis) / 60_000L
1050        return when {
1051            deltaMin < 1L  -> "just now"
1052            deltaMin < 60L -> "${deltaMin}m ago"
1053            deltaMin < 1440L -> "${deltaMin / 60L}h ago"
1054            else -> "${deltaMin / 1440L}d ago"
1055        }
1056    }
