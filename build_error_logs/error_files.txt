 178        /** Status for a single calendar day. Days after today are FUTURE (not yet knowable). */
 179        fun getDayStatus(cal: Calendar): DayStudyStatus {
 180            val today = Calendar.getInstance()
 181            val isFuture = cal.get(Calendar.YEAR) > today.get(Calendar.YEAR) ||
 182                (cal.get(Calendar.YEAR) == today.get(Calendar.YEAR) &&
 183                    cal.get(Calendar.DAY_OF_YEAR) > today.get(Calendar.DAY_OF_YEAR))
 184            if (isFuture) return DayStudyStatus.FUTURE
 185    
 186            val tasks = loadDay(keyForDay(cal))
 187            if (tasks.isEmpty()) return DayStudyStatus.MISSED // no tasks in plan == no study
 188            val done = tasks.count { it.state == TaskState.DONE }
 189            return when {
 190                done == tasks.size -> DayStudyStatus.COMPLETE
 191                done > 0            -> DayStudyStatus.PARTIAL
 192                else                 -> DayStudyStatus.MISSED
 193            }
 194        }

 264        fun getPauseStats(): PauseStats {
 265            val actualFocusToday = _todayTasks.value
 266                .filter { it.state == TaskState.DONE }
 267                .sumOf { it.actualMinutes }
 268    
 269            val cal = Calendar.getInstance()
 270            val weekSessions = mutableListOf<StudyTask>()
 271            repeat(7) {
 272                weekSessions += loadDay(keyForDay(cal)).filter { it.state == TaskState.DONE || it.state == TaskState.SKIPPED }
 273                cal.add(Calendar.DAY_OF_YEAR, -1)
 274            }
 275    
 276            if (weekSessions.isEmpty()) return PauseStats(actualFocusMinutesToday = actualFocusToday)
 277    
 278            val avgPauses  = weekSessions.sumOf { it.pauseCount }.toFloat() / weekSessions.size
 279            val pauseRate  = weekSessions.count { it.pauseCount > 0 } * 100 / weekSessions.size
 280    
 281            return PauseStats(
 282                actualFocusMinutesToday = actualFocusToday,
 283                avgPausesPerSession     = avgPauses,
 284                pauseRatePercent        = pauseRate
 285            )
 286        }
