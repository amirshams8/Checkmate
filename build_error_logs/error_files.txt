 181            val dayStart = (now.clone() as Calendar).apply {
 182                set(Calendar.HOUR_OF_DAY, 0); set(Calendar.MINUTE, 0)
 183                set(Calendar.SECOND, 0); set(Calendar.MILLISECOND, 0)
 184            }
 185            val nowMillis = now.timeInMillis
 186    
 187            fun rangeMillis(startHour: Int, startMinute: Int, endHour: Int, endMinute: Int): Pair<Long, Long>? {
 188                val start = (dayStart.clone() as Calendar).apply {
 189                    set(Calendar.HOUR_OF_DAY, startHour); set(Calendar.MINUTE, startMinute)
 190                }.timeInMillis
 191                val end = (dayStart.clone() as Calendar).apply {
 192                    set(Calendar.HOUR_OF_DAY, endHour); set(Calendar.MINUTE, endMinute)
 193                }.timeInMillis
 194                val clippedEnd = end.coerceAtMost(nowMillis)
 195                return if (clippedEnd > start) start to clippedEnd else null
 196            }
 197    
 198            val ranges = mutableListOf<Pair<Long, Long>>()
 199    
 200            // Tail of last night's usual 19:00-05:00 window, always present every day.
 201            rangeMillis(0, 0, WorkModeSchedule.END_HOUR, 0)?.let { ranges.add(it) }
 202    
 203            // Sun/Wed extra window: 01:00-15:10 — only the part beyond END_HOUR is new
 204            // (00:00-END_HOUR is already covered by the range above).
 205            if (now.get(Calendar.DAY_OF_WEEK) == Calendar.SUNDAY || now.get(Calendar.DAY_OF_WEEK) == Calendar.WEDNESDAY) {
 206                rangeMillis(
 207                    WorkModeSchedule.END_HOUR, 0,
 208                    WorkModeSchedule.SPECIAL_END_HOUR, WorkModeSchedule.SPECIAL_END_MINUTE
 209                )?.let { ranges.add(it) }
 210            }
 211    
 212            // Start of tonight's usual window: 19:00 onward, clipped to now.
 213            rangeMillis(WorkModeSchedule.START_HOUR, 0, 24, 0)?.let { ranges.add(it) }
 214    
 215            val lockedElapsedMillis = ranges.sumOf { it.second - it.first }
 216            if (lockedElapsedMillis <= 0L) return 100 // no locked window has elapsed yet today
 217    
 218            val usedDuringLockedMillis = ranges.sumOf { (s, e) -> AppUsageTracker.getUsageMillisInRange(context, s, e) }
 219            val adherence = 1f - (usedDuringLockedMillis.toFloat() / lockedElapsedMillis.toFloat())
 220            return (adherence.coerceIn(0f, 1f) * 100).roundToInt()
 221        }

 223        /**
 224         * Step 14 (Blueprint §26 "Baseline intervention statistics"). Separate load function,
 225         * same pattern as [loadAppUsage] just above — needs a Context (for
 226         * [InterventionDatabase.getInstance]) that only the Composable has, so it's called from
 227         * its own LaunchedEffect in StatsScreen rather than folded into [loadStats]'s init-time
 228         * call. Safe to call repeatedly (e.g. on resume), same as loadAppUsage.
 229         *
 230         * This is purely a read model — see [OutcomeLedgerStats]'s own doc for why "Learning
 231         * comes after measurement" (§25 principle 7) means nothing here feeds back into
 232         * trigger sensitivity, policy, or strategy selection. That's Step 15, not this.
 233         */
 234        fun loadInterventionStats(context: Context) {
 235            viewModelScope.launch {
 236                val stats = withContext(Dispatchers.IO) {
 237                    val dao = InterventionDatabase.getInstance(context).outcomeLedgerDao()
 238                    OutcomeLedgerStats.compute(dao)
 239                }
 240                _state.update { it.copy(interventionStats = stats) }
 241            }
 242        }
