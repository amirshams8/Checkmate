 181    /**
 182     * Hands a WebView download off to the system DownloadManager so it lands
 183     * in the device's normal Downloads folder with a notification, progress
 184     * bar, and retry-on-failure — same as any browser download. Cookies are
 185     * copied over explicitly because DownloadManager makes its own network
 186     * request outside the WebView, so it doesn't automatically inherit the
 187     * WebView's session.
 188     */
 189    private fun downloadFile(
 190        context: Context,
 191        url: String,
 192        userAgent: String?,
 193        contentDisposition: String?,
 194        mimeType: String?
 195    ) {
 196        try {
 197            val fileName = URLUtil.guessFileName(url, contentDisposition, mimeType)
 198            val cookie = CookieManager.getInstance().getCookie(url)
 199            val resolvedMimeType = mimeType?.takeIf { it.isNotBlank() }
 200                ?: MimeTypeMap.getSingleton()
 201                    .getMimeTypeFromExtension(MimeTypeMap.getFileExtensionFromUrl(url))
 202                ?: "application/octet-stream"
 203    
 204            val request = DownloadManager.Request(Uri.parse(url)).apply {
 205                if (!cookie.isNullOrBlank()) addRequestHeader("cookie", cookie)
 206                if (!userAgent.isNullOrBlank()) addRequestHeader("User-Agent", userAgent)
 207                setMimeType(resolvedMimeType)
 208                setNotificationVisibility(DownloadManager.Request.VISIBILITY_VISIBLE_NOTIFY_COMPLETED)
 209                setDestinationInExternalPublicDir(Environment.DIRECTORY_DOWNLOADS, fileName)
 210                setAllowedOverRoaming(true)
 211            }
 212    
 213            val downloadManager = context.getSystemService(Context.DOWNLOAD_SERVICE) as DownloadManager
 214            downloadManager.enqueue(request)
 215            Toast.makeText(context, "Downloading $fileName…", Toast.LENGTH_SHORT).show()
 216        } catch (e: Exception) {
 217            Toast.makeText(context, "Download failed: ${e.message}", Toast.LENGTH_LONG).show()
 218        }
 219    }
