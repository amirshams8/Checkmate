 190                }.timeInMillis
 191                val end = (dayStart.clone() as Calendar).apply {
 192                    set(Calendar.HOUR_OF_DAY, endHour); set(Calendar.MINUTE, endMinute)
 193                }.timeInMillis
 194                val clippedEnd = end.coerceAtMost(nowMillis)
 195                return if (clippedEnd > start) start to clippedEnd else null
 196            }

 199
